    43→const PROGRESS_KEY = 'energohero-progress-v1'

    47→interface Progress {
    48→  unlocked: number // сколько уровней открыто (1…20)
    49→  stars: number[]
    50→  best: number[]
    51→}
    52→
    53→const EMPTY_PROGRESS: Progress = {
    54→  unlocked: 1,
    55→  stars: Array(LEVELS.length).fill(0),
    56→  best: Array(LEVELS.length).fill(0),
    57→}
    58→
    59→function loadProgress(): Progress {
    60→  try {
    61→    const raw = localStorage.getItem(PROGRESS_KEY)
    62→    if (!raw) return EMPTY_PROGRESS
    63→    const p = JSON.parse(raw) as Partial<Progress>
    64→    const stars = Array(LEVELS.length).fill(0)
    65→    const best = Array(LEVELS.length).fill(0)
    66→    ;(p.stars ?? []).forEach((v, i) => (stars[i] = Number(v) || 0))
    67→    ;(p.best ?? []).forEach((v, i) => (best[i] = Number(v) || 0))
    68→    return {
    69→      unlocked: Math.min(LEVELS.length, Math.max(1, Number(p.unlocked) || 1)),
    70→      stars,
    71→      best,
    72→    }
    73→  } catch {
    74→    return EMPTY_PROGRESS
    75→  }
    76→}
    77→
    78→function starsFromResult(r: LevelResult): number {
    79→  const got = r.bulbs + r.switches + r.fuses
    80→  const tot = r.bulbsTotal + r.switchesTotal + r.fusesTotal
    81→  const pct = tot > 0 ? got / tot : 1
    82→  return pct >= 0.9 ? 3 : pct >= 0.55 ? 2 : 1
    83→}
    84→
    85→function saveLevelProgress(
    86→  levelIndex: number,
    87→  result: LevelResult | null,
    88→  summary?: RunSummary | null,
    89→) {
    90→  try {
    91→    const p = loadProgress()
    92→    const stars = result ? starsFromResult(result) : (summary?.stars ?? 1)
    93→    const score = result ? result.score : (summary?.score ?? 0)
    94→    p.stars[levelIndex] = Math.max(p.stars[levelIndex] ?? 0, stars)
    95→    p.best[levelIndex] = Math.max(p.best[levelIndex] ?? 0, score)
    96→    p.unlocked = Math.min(
    97→      LEVELS.length,
    98→      Math.max(p.unlocked, levelIndex + 2),
    99→    )
   100→    localStorage.setItem(PROGRESS_KEY, JSON.stringify(p))
   101→  } catch {
   102→    /* localStorage недоступен */
   103→  }
   104→}

   218→export default function EnergoHeroGame() {
   219→  const canvasRef = useRef<HTMLCanvasElement>(null)
   220→  const gameRef = useRef<EnergoGame | null>(null)
   221→  const musicRef = useRef<Music | null>(null)
   222→  const [phase, setPhase] = useState<Phase>('menu')
   223→  const [snap, setSnap] = useState<Snapshot | null>(null)
   224→  const [musicOn, setMusicOn] = useState(true)
   225→  const [sfxOn, setSfxOn] = useState(true)
   226→  const [isTouch, setIsTouch] = useState(false)
   227→  const [best, setBest] = useState(0)
   228→  const [progress, setProgress] = useState<Progress>(EMPTY_PROGRESS)
   229→  const [menuView, setMenuView] = useState<
   230→    'main' | 'levels' | 'howto' | 'story'
   231→  >('main')
   232→  // ---------- Сюжет: катсцены ----------
   233→  const [cutscene, setCutscene] = useState<(typeof STORY)[string] | null>(null)
   234→  const [seenScenes, setSeenScenes] = useState<Set<string>>(new Set())
   235→  const seenRef = useRef<Set<string>>(new Set())
   236→  const afterSceneRef = useRef<(() => void) | null>(null)
   237→
   238→  /** Показать главу хроники; повторно — только по запросу (force). */
   239→  const playScene = useCallback(
   240→    (id: string, after: () => void, force = false) => {
   241→      if (!STORY[id]) {
   242→        after()
   243→        return
   244→      }
   245→      if (!force && seenRef.current.has(id)) {
   246→        after()
   247→        return
   248→      }
   249→      seenRef.current.add(id)
   250→      saveSeenScenes(seenRef.current)
   251→      setSeenScenes(new Set(seenRef.current))
   252→      afterSceneRef.current = after
   253→      setCutscene(STORY[id])
   254→    },
   255→    [],
   256→  )
   257→
   258→  const finishCutscene = useCallback(() => {
   259→    const after = afterSceneRef.current
   260→    afterSceneRef.current = null
   261→    setCutscene(null)
   262→    after?.()
   263→  }, [])
   264→
   265→  useEffect(() => {
   266→    const canvas = canvasRef.current
   267→    if (!canvas) return
   268→    const dpr = Math.min(2, window.devicePixelRatio || 1)
   269→    canvas.width = VIEW_W * dpr
   270→    canvas.height = VIEW_H * dpr
   271→    const ctx = canvas.getContext('2d')
   272→    if (!ctx) return
   273→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   274→
   275→    const music = new Music()
   276→    musicRef.current = music
   277→
   278→    let mOn = true
   279→    let sOn = true
   280→    try {
   281→      mOn = localStorage.getItem(PREF_MUSIC) !== 'off'
   282→      sOn = localStorage.getItem(PREF_SFX) !== 'off'
   283→    } catch {
   284→      /* ignore */
   285→    }
   286→    music.enabled = mOn
   287→
   288→    const game = new EnergoGame(canvas, {
   289→      onPhase: (s) => {
   290→        setPhase(s.phase)
   291→        setSnap(s)
   292→        if (s.phase === 'menu' || s.phase === 'intro') {
   293→          if (music.enabled) music.start()
   294→          music.setDuck(0.55)
   295→        } else if (s.phase === 'playing') {
   296→          music.setDuck(1)
   297→        } else if (s.phase === 'paused' || s.phase === 'complete') {
   298→          music.setDuck(0.3)
   299→        } else if (s.phase === 'gameover' || s.phase === 'victory') {
   300→          music.stop()
   301→        }
   302→        if (s.phase === 'complete' && s.result) {
   303→          saveLevelProgress(s.levelIndex, s.result)
   304→          setProgress(loadProgress())
   305→        }
   306→        if (s.phase === 'victory' && s.summary) {
   307→          saveLevelProgress(s.levelIndex, null, s.summary)
   308→          setProgress(loadProgress())
   309→          try {
   310→            setBest(
   311→              Number(localStorage.getItem('energohero-best') ?? '0') || 0,
   312→            )
   313→          } catch {
   314→            /* ignore */
   315→          }
   316→        }
   317→      },
   318→    })
   319→    game.sfx.muted = !sOn
   320→    gameRef.current = game
   321→    // Отладочный хук (используется автотестами: __energo.loadLevel(n))
   322→    ;(window as unknown as { __energo?: EnergoGame }).__energo = game
   323→
   324→    // Отложенная инициализация некритичного UI-состояния
   325→    const raf = requestAnimationFrame(() => {
   326→      setIsTouch(
   327→        window.matchMedia('(pointer: coarse)').matches ||
   328→          'ontouchstart' in window ||
   329→          window.innerWidth < 768,
   330→      )
   331→      setProgress(loadProgress())
   332→      setMusicOn(mOn)
   333→      setSfxOn(sOn)
   334→      const seen = loadSeenScenes()
   335→      seenRef.current = seen
   336→      setSeenScenes(new Set(seen))
   337→      try {
   338→        setBest(Number(localStorage.getItem('energohero-best') ?? '0') || 0)
   339→      } catch {
   340→        /* ignore */
   341→      }
   342→    })
   343→
   344→    // Музыка стартует после первого жеста пользователя (политика автоплея)
   345→    const kick = () => {
   346→      if (music.enabled) music.start()
   347→    }
   348→    window.addEventListener('pointerdown', kick, { once: true })
   349→    window.addEventListener('keydown', kick, { once: true })
   350→
   351→    return () => {
   352→      cancelAnimationFrame(raf)
   353→      window.removeEventListener('pointerdown', kick)
   354→      window.removeEventListener('keydown', kick)
   355→      music.stop()
   356→      game.destroy()
   357→      gameRef.current = null
   358→    }
   359→  }, [])
   360→
   361→  const act = useCallback(
   362→    (
   363→      action:
   364→        | 'start'
   365→        | 'begin'
   366→        | 'retry'
   367→        | 'next'
   368→        | 'menu'
   369→        | 'pause'
   370→        | { select: number },
   371→    ) => {
   372→      const g = gameRef.current
   373→      if (!g) return
   374→      if (typeof action === 'object') {
   375→        g.selectLevel(action.select)
   376→        return
   377→      }
   378→      switch (action) {
   379→        case 'start':
   380→          g.startRun()
   381→          break
   382→        case 'begin':
   383→          g.beginLevel()
   384→          break
   385→        case 'retry':
   386→          g.retryLevel()
   387→          break
   388→        case 'next':
   389→          g.nextLevel()
   390→          break
   391→        case 'menu':
   392→          g.toMenu()
   393→          break
   394→        case 'pause':
   395→          g.togglePause()
   396→          break
   397→      }
   398→    },
   399→    [],
   400→  )
   401→
   402→  /** «Следующий наряд» с проверкой: не начинается ли новая глава. */
   403→  const nextWithStory = useCallback(() => {
   404→    const g = gameRef.current
   405→    if (!g) return
   406→    const nextIdx = (snap?.levelIndex ?? 0) + 1
   407→    const chapter = nextIdx >= LEVELS.length ? 'epilogue' : CHAPTER_AT[nextIdx]
   408→    if (chapter) playScene(chapter, () => act('next'))
   409→    else act('next')
   410→  }, [snap, act, playScene])
   411→
   412→  /** Загрузка уровня с карты с проверкой: не открывает ли он главу. */
   413→  const pickWithStory = useCallback(
   414→    (i: number) => {
   415→      const g = gameRef.current
   416→      if (!g) return
   417→      const chapter = CHAPTER_AT[i]
   418→      if (chapter) playScene(chapter, () => act({ select: i }))
   419→      else act({ select: i })
   420→    },
   421→    [act, playScene],
   422→  )
   423→
   424→  /** Новая смена: всегда с пролога (кнопка пропуска в кадре есть). */
   425→  const startFresh = useCallback(() => {
   426→    playScene('prologue', () => act('start'), true)
   427→  }, [playScene, act])
   428→
   429→  // Enter — основное действие на оверлеях (когда не открыта катсцена)
   430→  useEffect(() => {
   431→    const onKey = (e: KeyboardEvent) => {
   432→      if (e.code !== 'Enter' && e.code !== 'NumpadEnter' && e.key !== 'Enter')
   433→        return
   434→      if (cutscene) return // катсцена сама слушает Enter
   435→      if (phase === 'menu') startFresh()
   436→      else if (phase === 'intro') act('begin')
   437→      else if (phase === 'complete') nextWithStory()
   438→      else if (phase === 'gameover') act('retry')
   439→      else if (phase === 'victory') startFresh()
   440→      else if (phase === 'paused') act('pause')
   441→    }
   442→    window.addEventListener('keydown', onKey)
   443→    return () => window.removeEventListener('keydown', onKey)
   444→  }, [phase, act, cutscene, startFresh, nextWithStory])
   445→
   446→  const hold = useCallback(
   447→    (k: 'left' | 'right' | 'jump') => ({
   448→      onPointerDown: (e: React.PointerEvent) => {
   449→        e.preventDefault()
   450→        gameRef.current?.setTouch(k, true)
   451→      },
   452→      onPointerUp: () => gameRef.current?.setTouch(k, false),
   453→      onPointerLeave: () => gameRef.current?.setTouch(k, false),
   454→      onPointerCancel: () => gameRef.current?.setTouch(k, false),
   455→    }),
   456→    [],
   457→  )
   458→
   459→  const toggleMusic = useCallback(() => {
   460→    setMusicOn((v) => {
   461→      const nv = !v
   462→      musicRef.current?.setEnabled(nv)
   463→      try {
   464→        localStorage.setItem(PREF_MUSIC, nv ? 'on' : 'off')
   465→      } catch {
   466→        /* ignore */
   467→      }
   468→      return nv
   469→    })
   470→  }, [])
   471→
   472→  const toggleSfx = useCallback(() => {
   473→    setSfxOn((v) => {
   474→      const nv = !v
   475→      if (gameRef.current) gameRef.current.sfx.muted = !nv
   476→      try {
   477→        localStorage.setItem(PREF_SFX, nv ? 'on' : 'off')
   478→      } catch {
   479→        /* ignore */
   480→      }
   481→      return nv
   482→    })
   483→  }, [])
   484→
   485→  const playing = phase === 'playing' || phase === 'paused'
   486→
   487→  return (
   488→    <div className="ah-grid-bg flex min-h-screen w-full flex-col items-center justify-center gap-3 p-2 text-[#efe7d2] sm:p-4">
   489→      <main className="w-full max-w-[1080px]">
   490→        <div className="relative aspect-[16/9] max-md:aspect-[4/5] w-full overflow-hidden rounded-md border-2 border-[#2a2f28] bg-[#0b0d0c] shadow-[0_0_0_6px_rgba(255,255,255,0.03),0_24px_70px_rgba(0,0,0,0.6)]">
   491→          <canvas
   492→            ref={canvasRef}
   493→            className="absolute inset-0 h-full w-full object-contain"
   494→            aria-label="Игровое поле ЭнергоГероя"
   495→          />
   496→
   497→          {/* Кнопки музыки, звука и паузы */}
   498→          <div className="absolute right-2 top-2 z-20 flex gap-1.5">
   499→            <CanvasIconBtn
   500→              label={musicOn ? 'Выключить музыку' : 'Включить музыку'}
   501→              on={musicOn}
   502→              onClick={toggleMusic}
   503→            >
   504→              <Music2 className="h-4 w-4" />
   505→            </CanvasIconBtn>
   506→            <CanvasIconBtn
   507→              label={sfxOn ? 'Выключить звук' : 'Включить звук'}
   508→              on={sfxOn}
   509→              onClick={toggleSfx}
   510→            >
   511→              {sfxOn ? (
   512→                <Volume2 className="h-4 w-4" />
   513→              ) : (
   514→                <VolumeX className="h-4 w-4" />
   515→              )}
   516→            </CanvasIconBtn>
   517→            {playing && (
   518→              <CanvasIconBtn
   519→                label="Пауза"
   520→                on
   521→                onClick={() => act('pause')}
   522→              >
   523→                <Pause className="h-4 w-4" />
   524→              </CanvasIconBtn>
   525→            )}
   526→          </div>
   527→
   528→          {/* ---------- Оверлеи ---------- */}
   529→          {phase === 'menu' && (
   530→            <Overlay>
   531→              {menuView === 'main' && (
   532→                <MenuMain
   533→                  best={best}
   534→                  unlocked={progress.unlocked}
   535→                  onContinue={() => act({ select: progress.unlocked - 1 })}
   536→                  onFresh={startFresh}
   537→                  onLevels={() => setMenuView('levels')}
   538→                  onHowto={() => setMenuView('howto')}
   539→                  onStory={() => setMenuView('story')}
   540→                />
   541→              )}
   542→              {menuView === 'levels' && (

   971→function CompleteDoc({
   972→  snap,
   973→  onNext,
   974→  onRetry,
   975→  onMenu,
   976→}: {
   977→  snap: Snapshot
   978→  onNext: () => void
   979→  onRetry: () => void
   980→  onMenu: () => void
   981→}) {
   982→  const r = snap.result!
   983→  const rank =
   984→    r.bulbs / Math.max(1, r.bulbsTotal) >= 0.95 &&
   985→    r.switches / Math.max(1, r.switchesTotal) >= 0.95 &&
   986→    (r.fusesTotal === 0 || r.fuses / r.fusesTotal >= 0.95)
   987→      ? 'S'
   988→      : starsFromResult(r) === 3
   989→        ? 'A'
   990→        : starsFromResult(r) === 2
   991→          ? 'B'
   992→          : 'C'
   993→  return (
   994→    <Doc
   995→      docNo={`ЭГ-${String(snap.levelIndex + 1).padStart(2, '0')}/АКТ`}
   996→      title="АКТ ВЫПОЛНЕННЫХ РАБОТ"
   997→      tone="tape"
   998→    >
   999→      <span className="ah-stamp ah-stamp-ok absolute right-4 top-14 rotate-6 px-3 py-1 text-sm font-black uppercase tracking-[0.3em] sm:right-7">
  1000→        Зачтено
  1001→      </span>
  1002→      <span className="absolute right-6 top-[74px] flex h-9 w-9 rotate-6 items-center justify-center rounded-full border-[3px] border-[#2e7d43] font-mono text-lg font-black text-[#2e7d43] mix-blend-multiply sm:right-9">
  1003→        {rank}
  1004→      </span>
  1005→
  1006→      <h2 className="text-2xl font-black tracking-wide text-[#221e17] sm:text-3xl">
  1007→        «{r.levelName}» — свет восстановлен
  1008→      </h2>
  1009→      <p className="mt-0.5 font-mono text-[10px] uppercase tracking-[0.28em] text-[#8a8171]">
  1010→        {snap.levelSubtitle}
  1011→      </p>
  1012→
  1013→      <div className="mx-auto mt-4 max-w-sm text-left">
  1014→        <LedgerRow k="LED-лампочки" v={`${r.bulbs}/${r.bulbsTotal}`} />
  1015→        <LedgerRow k="Рубильники" v={`${r.switches}/${r.switchesTotal}`} />
  1016→        {r.fusesTotal > 0 && (
  1017→          <LedgerRow k="Предохранители" v={`${r.fuses}/${r.fusesTotal}`} />

  1365→function ScoreCount({ value }: { value: number }) {
  1366→  const [v, setV] = useState(0)
  1367→  useEffect(() => {
  1368→    let raf = 0
  1369→    const t0 = performance.now()
  1370→    const dur = 700
  1371→    const tick = (t: number) => {
  1372→      const k = Math.min(1, (t - t0) / dur)
  1373→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
  1374→      if (k < 1) raf = requestAnimationFrame(tick)
  1375→    }
  1376→    raf = requestAnimationFrame(tick)
  1377→    return () => cancelAnimationFrame(raf)
  1378→  }, [value])
  1379→  return <>{v}</>
  1380→}
