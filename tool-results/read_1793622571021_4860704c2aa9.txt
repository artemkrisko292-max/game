   231→  private finishRun() {
   232→    const total = this.runBulbsTotal + this.runSwitchesTotal

   236→    try {
   237→      best = Number(localStorage.getItem('energohero-best') ?? '0') || 0
   238→      if (this.score > best) {
   239→        best = this.score
   240→        localStorage.setItem('energohero-best', String(best))
   241→      }
   242→    } catch {
   243→      /* localStorage недоступен */
   244→    }
   245→    this.summary = {
   246→      score: this.score,
   247→      best,
   248→      bulbs: this.runBulbs,
   249→      bulbsTotal: this.runBulbsTotal,
   250→      switches: this.runSwitches,
   251→      switchesTotal: this.runSwitchesTotal,
   252→      stars,
   253→    }
   254→    this.setPhase('victory')

   430→  private onKeyDown = (e: KeyboardEvent) => {
   431→    if (this.GAME_KEYS.has(e.code)) e.preventDefault()
   432→    switch (e.code) {
   433→      case 'ArrowLeft':
   434→      case 'KeyA':
   435→        this.kb.left = true
   436→        break
   437→      case 'ArrowRight':
   438→      case 'KeyD':
   439→        this.kb.right = true
   440→        break
   441→      case 'Space':
   442→      case 'ArrowUp':
   443→      case 'KeyW':
   444→      case 'KeyZ':
   445→        this.kb.jump = true
   446→        this.sfx.ensure()
   447→        break
   448→      case 'Escape':
   449→      case 'KeyP':
   450→        if (this.phase === 'playing' || this.phase === 'paused')
   451→          this.togglePause()
   452→        break
   453→    }
   454→  }
   455→
   456→  private onKeyUp = (e: KeyboardEvent) => {
   457→    switch (e.code) {
   458→      case 'ArrowLeft':
   459→      case 'KeyA':
   460→        this.kb.left = false
   461→        break
   462→      case 'ArrowRight':
   463→      case 'KeyD':
   464→        this.kb.right = false
   465→        break
   466→      case 'Space':
   467→      case 'ArrowUp':
   468→      case 'KeyW':
   469→      case 'KeyZ':
   470→        this.kb.jump = false
   471→        break
   472→    }
   473→  }

   476→    this.kb.left = this.kb.right = this.kb.jump = false

   481→  private get input() {
   482→    return {
   483→      left: this.kb.left || this.touch.left,
   484→      right: this.kb.right || this.touch.right,
   485→      jump: this.kb.jump || this.touch.jump,

   489→  // ---------- Главный цикл ----------
   490→
   491→  private loop = (t: number) => {
   492→    if (this.destroyed) return
   493→    this.raf = requestAnimationFrame(this.loop)
   494→    let dt = (t - this.lastT) / 1000
   495→    this.lastT = t
   496→    if (dt > 0.1) dt = 0.1
   497→    if (this.phase === 'playing') {
   498→      this.acc += dt
   499→      let steps = 0
   500→      while (this.acc >= STEP && steps < 4) {
   501→        this.step()
   502→        this.acc -= STEP
   503→        steps++
   504→      }
   505→      if (steps === 4) this.acc = 0
   506→    }
   507→    render(this.ctx, this)
   508→  }

   510→  // ---------- Шаг симуляции ----------
   511→
   512→  private step() {
   513→    const p = this.player
   514→    const inp = this.input
   515→    this.frame++
   516→    if (this.shake > 0) this.shake *= 0.85
   517→
   518→    // Прыжок: буфер + койот-время
   519→    const jumpPressed = inp.jump && !this.prevJump
   520→    this.prevJump = inp.jump
   521→    if (jumpPressed) p.buffer = BUFFER
   522→    if (p.buffer > 0) p.buffer--
   523→    if (p.coyote > 0) p.coyote--
   524→
   525→    // Горизонталь
   526→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   527→    if (inp.left && !inp.right) {
   528→      p.vx -= acc
   529→      p.facing = -1
   530→    } else if (inp.right && !inp.left) {
   531→      p.vx += acc
   532→      p.facing = 1
   533→    } else {
   534→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   535→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   536→    }
   537→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   538→
   539→    // Прыжок
   540→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   541→      p.vy = JUMP_V
   542→      p.onGround = false
   543→      p.coyote = 0
   544→      p.buffer = 0
   545→      p.squash = 1.18
   546→      this.sfx.jump()
   547→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   548→    }
   549→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   550→
   551→    // Гравитация
   552→    p.vy = Math.min(p.vy + GRAV, 14)
   553→
   554→    // Движущиеся платформы
   555→    for (const pl of this.plats) {
   556→      pl.phase += pl.speed
   557→      const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   558→      pl.dx = nx - pl.x
   559→      pl.x = nx
   560→    }
   561→
   562→    // Движение и коллизии
   563→    this.moveX(p, p.vx)
   564→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   565→
   566→    const wasOnGround = p.onGround
   567→    const prevFeet = p.y + p.h
   568→    this.moveY(p, p.vy, prevFeet)
   569→
   570→    if (p.onGround) {
   571→      if (!wasOnGround) {
   572→        p.squash = 0.78
   573→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   574→        this.sfx.land()
   575→      }
   576→      p.coyote = COYOTE
   577→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   578→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   579→      if (underL && underR) {
   580→        p.safeX = p.x
   581→        p.safeY = p.y
   582→      }
   583→    }
   584→    p.squash += (1 - p.squash) * 0.18
   585→    p.anim += Math.abs(p.vx) * 0.09
   586→    if (p.invuln > 0) p.invuln--
   587→    if (p.blink > 0) p.blink--
   588→    else if (Math.random() < 0.012) p.blink = 8
   589→
   590→    // Провал за нижнюю границу мира (люки)
   591→    if (p.y > this.worldH + 60) {
   592→      this.damage(DMG_FALL, 0)
   593→      p.x = p.safeX
   594→      p.y = p.safeY
   595→      p.vx = 0
   596→      p.vy = 0
   597→    }
   598→
   599→    this.updateVampires()
   600→    this.updatePhantoms()
   601→    this.updatePickups()
   602→    this.updateSwitches()
   603→    this.updateHazards()
   604→    this.updateExit()
   605→    this.updateEnergy()
   606→    this.updateParticles()
   607→    this.updateCamera()
   608→  }

   610→  // ---------- Коллизии с тайлами ----------
   611→
   612→  solidAt(px: number, py: number): boolean {
   613→    const c = Math.floor(px / TILE)
   614→    const r = Math.floor(py / TILE)
   615→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   616→    return this.grid[r * this.cols + c] === 1
   617→  }
   618→
   619→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   620→    const c0 = Math.floor(x / TILE)
   621→    const c1 = Math.floor((x + w - EPS) / TILE)
   622→    const r0 = Math.floor(y / TILE)
   623→    const r1 = Math.floor((y + h - EPS) / TILE)
   624→    for (let r = r0; r <= r1; r++) {
   625→      for (let c = c0; c <= c1; c++) {
   626→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   627→        if (this.grid[r * this.cols + c] === 1) return true
   628→      }
   629→    }
   630→    return false
   631→  }

   633→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   634→    e.x += dx
   635→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   636→    if (dx > 0) {
   637→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   638→    } else if (dx < 0) {
   639→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   640→    }
   641→    e.vx = 0
   642→    return true
   643→  }

   645→  private moveY(
   646→    e: Box & { vy: number; onGround: boolean },
   647→    dy: number,
   648→    prevFeet: number,
   649→  ): void {
   650→    e.y += dy
   651→    if (dy >= 0) {
   652→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   653→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   654→        e.vy = 0
   655→        e.onGround = true
   656→      }
   657→      for (const pl of this.plats) {
   658→        const feet = e.y + e.h
   659→        if (
   660→          e.vy >= 0 &&
   661→          feet >= pl.y &&
   662→          prevFeet <= pl.y + 6 &&
   663→          e.x + e.w > pl.x &&
   664→          e.x < pl.x + pl.w
   665→        ) {
   666→          e.y = pl.y - e.h
   667→          e.vy = 0
   668→          e.onGround = true
   669→          e.x += pl.dx
   670→        }
   671→      }
   672→    } else {
   673→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   674→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   675→        e.vy = 0
   676→      }
   677→    }
   678→  }

   682→  private updateVampires() {
   683→    const p = this.player
   684→    for (const v of this.vampires) {
   685→      if (v.dead) continue
   686→      v.anim += 0.08
   687→      v.vy = Math.min(v.vy + GRAV, 12)
   688→      const oldVx = v.vx
   689→      this.moveX(v, v.vx)
   690→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   691→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   692→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   693→
   694→      const prevFeet = v.y + v.h
   695→      const wasGround = v.onGround
   696→      v.onGround = false
   697→      this.moveY(v, v.vy, prevFeet)
   698→      if (v.onGround && !wasGround) v.vy = 0
   699→      // разворот у обрыва
   700→      if (v.onGround) {
   701→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   702→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   703→      }
   704→
   705→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   706→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   707→          v.dead = true
   708→          p.vy = -9.5
   709→          this.score += 100
   710→          this.shake = 6
   711→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   712→          this.sfx.stomp()
   713→        } else {
   714→          this.damage(DMG_VAMPIRE, p.x < v.x + v.w / 2 ? -1 : 1)
   715→        }
   716→      }
   717→    }
   718→  }

   720→  private updatePhantoms() {
   721→    const p = this.player
   722→    for (const f of this.phantoms) {
   723→      if (f.dead) continue
   724→      f.phase += f.speed
   725→      f.x = f.cx + Math.sin(f.phase) * f.amp
   726→      f.y = f.cy + Math.sin(f.phase * 2.3) * 26
   727→      if (p.invuln <= 0 && circleRect(f.x, f.y, 16, p)) {
   728→        this.damage(DMG_PHANTOM, p.x < f.x ? -1 : 1)
   729→        f.phase += Math.PI
   730→      }
   731→    }

   751→  private updateSwitches() {

   767→          this.runSwitches++
   768→          for (const l of this.lamps) {
   769→            if (l.group === s.group)
   770→              this.spawnParticles(l.x, l.y + 10, 7, 'spark')
   771→          }
   772→          this.spawnParticles(s.x, s.y - 20, 8, 'star')
   773→          this.sfx.flip()
   774→        }
   775→      }
   776→    }

   779→  private lampDrain(): number {
   780→    let drain = 0
   781→    const p = this.player
   782→    const pcx = p.x + p.w / 2
   783→    const pcy = p.y + p.h / 2
   784→    for (const l of this.lamps) {
   785→      if (!l.on) continue
   786→      const dx = Math.abs(pcx - l.x)
   787→      const dy = pcy - l.y
   788→      if (dy > 0 && dy < LAMP_CONE_LEN) {
   789→        const halfW = 16 + dy * 0.34
   790→        if (dx < halfW) {
   791→          drain += DRAIN_LAMP * (1 - (dy / LAMP_CONE_LEN) * 0.45)
   792→        }
   793→      }
   794→    }
   795→    return Math.min(drain, DRAIN_LAMP * 2)
   796→  }

   863→  private updateParticles() {
   864→    for (let i = this.particles.length - 1; i >= 0; i--) {
   865→      const q = this.particles[i]
   866→      q.life--
   867→      q.x += q.vx
   868→      q.y += q.vy
   869→      q.vy += q.grav
   870→      q.vx *= 0.98
   871→      if (q.life <= 0) this.particles.splice(i, 1)
   872→    }
   873→  }

   875→  spawnParticles(x: number, y: number, n: number, kind: Particle['kind']) {
   876→    const colors: Record<Particle['kind'], string[]> = {
   877→      spark: ['#ffd94d', '#ffb63d', '#fff3b0'],
   878→      poof: ['#c9b6d8', '#9c7fae', '#e6d9f0'],
   879→      star: ['#ffe873', '#a4f04a', '#7ee081'],
   880→      dust: ['#c9b08a', '#a8916e'],
   881→    }
   882→    for (let i = 0; i < n; i++) {
   883→      const a = Math.random() * Math.PI * 2
   884→      const sp = 1 + Math.random() * 2.6
   885→      this.particles.push({
   886→        x,
   887→        y,
   888→        vx: Math.cos(a) * sp,
   889→        vy: Math.sin(a) * sp - 1.2,
   890→        life: 22 + Math.random() * 18,
   891→        maxLife: 40,
   892→        size: 2 + Math.random() * 3,
   893→        color: colors[kind][Math.floor(Math.random() * colors[kind].length)],
   894→        kind,
   895→        grav: kind === 'poof' ? 0 : kind === 'dust' ? 0.08 : 0.14,
   896→      })
   897→    }
   898→    if (this.particles.length > 300) this.particles.splice(0, 100)
   899→  }

   920→function rectsOverlap(a: Box, b: Box): boolean {
   921→  return (
   922→    a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y
   923→  )
   924→}
   925→
   926→function circleRect(
   927→  cx: number,
   928→  cy: number,
   929→  r: number,
   930→  rect: Box,
   931→): boolean {
   932→  const nx = Math.max(rect.x, Math.min(cx, rect.x + rect.w))
   933→  const ny = Math.max(rect.y, Math.min(cy, rect.y + rect.h))
   934→  const dx = cx - nx
   935→  const dy = cy - ny
   936→  return dx * dx + dy * dy <= r * r
   937→}
   938→
   939→function clamp(v: number, min: number, max: number): number {
   940→  return v < min ? min : v > max ? max : v
   941→}

   943→function countChar(map: string[], ch: string): number {
   944→  let n = 0
   945→  for (const row of map) for (const c of row) if (c === ch) n++
   946→  return n
   947→}
