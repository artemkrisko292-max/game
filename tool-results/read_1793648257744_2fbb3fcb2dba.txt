    68→export function render(ctx: CanvasRenderingContext2D, g: EnergoGame) {
    69→  const theme = THEMES[LEVELS[g.levelIndex].theme]
    70→  ctx.clearRect(0, 0, VIEW_W, VIEW_H)
    71→  ctx.save()
    72→
    73→  // Тряска камеры
    74→  if (g.shake > 0.3) {
    75→    ctx.translate(
    76→      (Math.random() - 0.5) * g.shake,
    77→      (Math.random() - 0.5) * g.shake,
    78→    )
    79→  }
    80→
    81→  drawBackground(ctx, g, theme)
    82→
    83→  ctx.save()
    84→  ctx.translate(-Math.round(g.camX), -Math.round(g.camY))
    85→
    86→  drawTiles(ctx, g, theme)
    87→  drawHazards(ctx, g)
    88→  drawSwitches(ctx, g)
    89→  drawExit(ctx, g)
    90→  drawLampCones(ctx, g)
    91→  drawBulbs(ctx, g)
    92→  drawPlats(ctx, g)
    93→  drawVampires(ctx, g)
    94→  drawPhantoms(ctx, g)
    95→  drawPlayer(ctx, g)
    96→  drawParticles(ctx, g)
    97→  drawLampGlows(ctx, g)
    98→
    99→  ctx.restore()
   100→  ctx.restore()
   101→
   102→  drawHUD(ctx, g)
   103→}

   107→function drawBackground(
   108→  ctx: CanvasRenderingContext2D,
   109→  g: EnergoGame,
   110→  th: Theme,
   111→) {
   112→  const grad = ctx.createLinearGradient(0, 0, 0, VIEW_H)
   113→  grad.addColorStop(0, th.skyTop)
   114→  grad.addColorStop(1, th.skyBot)
   115→  ctx.fillStyle = grad
   116→  ctx.fillRect(0, 0, VIEW_W, VIEW_H)
   117→
   118→  // Одна и та же «комната» рисуется одним из трёх художников фона,
   119→  // палитру задаёт тема уровня.
   120→  const name = LEVELS[g.levelIndex].theme
   121→  const painter =
   122→    name === 'bedroom' || name === 'attic'
   123→      ? drawBedroomBg
   124→      : name === 'kitchen' || name === 'living'
   125→        ? drawKitchenBg
   126→        : drawBasementBg
   127→  painter(ctx, g)
   128→}

   130→function wrapX(i: number, camX: number, k: number, span: number): number {
   131→  const m = (i * 137.13 - camX * k) % span
   132→  return (m < 0 ? m + span : m) - 60
   133→}

   135→function drawBedroomBg(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   136→  // Звёзды
   137→  ctx.fillStyle = 'rgba(255,240,210,0.8)'
   138→  for (let i = 0; i < 46; i++) {
   139→    const x = wrapX(i * 89, g.camX, 0.22, VIEW_W + 120)
   140→    const y = (i * 53) % 260
   141→    const s = 1 + ((i * 7) % 3) * 0.5
   142→    const tw = 0.5 + 0.5 * Math.sin(g.frame * 0.04 + i * 1.7)
   143→    ctx.globalAlpha = 0.35 + tw * 0.5
   144→    ctx.fillRect(x, y, s, s)
   145→  }
   146→  ctx.globalAlpha = 1
   147→
   148→  // Луна
   149→  const mx = VIEW_W * 0.8 - g.camX * 0.08
   150→  const mg = ctx.createRadialGradient(mx, 92, 8, mx, 92, 70)
   151→  mg.addColorStop(0, 'rgba(255,236,180,0.5)')
   152→  mg.addColorStop(1, 'rgba(255,236,180,0)')
   153→  ctx.fillStyle = mg
   154→  ctx.fillRect(mx - 70, 22, 140, 140)
   155→  ctx.fillStyle = '#f7e6ae'
   156→  ctx.beginPath()
   157→  ctx.arc(mx, 92, 26, 0, Math.PI * 2)
   158→  ctx.fill()
   159→  ctx.fillStyle = 'rgba(214,190,130,0.7)'
   160→  ctx.beginPath()
   161→  ctx.arc(mx - 8, 86, 5, 0, Math.PI * 2)
   162→  ctx.arc(mx + 9, 99, 4, 0, Math.PI * 2)
   163→  ctx.fill()
   164→
   165→  // Полосы обоев
   166→  ctx.strokeStyle = 'rgba(255,255,255,0.045)'
   167→  ctx.lineWidth = 26
   168→  for (let i = 0; i < 14; i++) {
   169→    const x = wrapX(i * 150, g.camX, 0.45, VIEW_W + 300)
   170→    ctx.beginPath()

   246→function hash2(c: number, r: number): number {
   247→  return ((c * 73856093) ^ (r * 19349663)) % 1000 / 1000
   248→}

   250→function drawTiles(
   251→  ctx: CanvasRenderingContext2D,
   252→  g: EnergoGame,
   253→  th: Theme,
   254→) {
   255→  const c0 = Math.max(0, Math.floor(g.camX / TILE) - 1)
   256→  const c1 = Math.min(g.cols - 1, Math.ceil((g.camX + VIEW_W) / TILE) + 1)
   257→  const r0 = Math.max(0, Math.floor(g.camY / TILE) - 1)
   258→  const r1 = Math.min(g.rows - 1, Math.ceil((g.camY + VIEW_H) / TILE) + 1)
   259→
   260→  for (let r = r0; r <= r1; r++) {
   261→    for (let c = c0; c <= c1; c++) {
   262→      if (g.grid[r * g.cols + c] !== 1) continue
   263→      const x = c * TILE
   264→      const y = r * TILE
   265→      ctx.fillStyle = th.wall
   266→      ctx.fillRect(x, y, TILE, TILE)
   267→
   268→      // Тёмный низ
   269→      if (r + 1 >= g.rows || g.grid[(r + 1) * g.cols + c] !== 1) {
   270→        ctx.fillStyle = th.wallDark
   271→        ctx.fillRect(x, y + TILE - 6, TILE, 6)
   272→      }
   273→      // Светлый верх
   274→      if (r === 0 || g.grid[(r - 1) * g.cols + c] !== 1) {
   275→        ctx.fillStyle = th.wallTop
   276→        ctx.fillRect(x, y, TILE, 7)
   277→        ctx.fillStyle = th.accent
   278→        ctx.fillRect(x, y + 7, TILE, 2)
   279→      }
   280→      // Пятнышки/шов
   281→      const h = hash2(c, r)
   282→      ctx.fillStyle = th.wallDark
   283→      ctx.globalAlpha = 0.35
   284→      if (h < 0.3) {
   285→        ctx.fillRect(x + 6 + h * 60, y + 12 + h * 40, 8, 4)
   286→      } else if (h < 0.55) {
   287→        ctx.fillRect(x + 24, y + 8 + h * 30, 4, 10)
   288→      } else if (h < 0.75) {
   289→        ctx.fillRect(x + 4, y + 22, 12, 3)
   290→      }
   291→      ctx.globalAlpha = 1
   292→      // Тонкая рамка
   293→      ctx.strokeStyle = 'rgba(0,0,0,0.18)'
   294→      ctx.lineWidth = 1
   295→      ctx.strokeRect(x + 0.5, y + 0.5, TILE - 1, TILE - 1)
   296→    }
   297→  }
   298→}

   302→function drawBulbs(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   303→  for (const b of g.bulbs) {
   304→    if (b.taken) continue
   305→    const y = b.y + Math.sin(g.frame * 0.05 + b.phase) * 4
   306→    const gl = ctx.createRadialGradient(b.x, y, 2, b.x, y, 22)
   307→    gl.addColorStop(0, 'rgba(255,232,130,0.55)')
   308→    gl.addColorStop(1, 'rgba(255,232,130,0)')
   309→    ctx.fillStyle = gl
   310→    ctx.fillRect(b.x - 22, y - 22, 44, 44)
   311→
   312→    ctx.fillStyle = '#fff6c9'
   313→    ctx.strokeStyle = '#d9a441'
   314→    ctx.lineWidth = 2
   315→    ctx.beginPath()
   316→    ctx.arc(b.x, y - 2, 8, 0, Math.PI * 2)
   317→    ctx.fill()
   318→    ctx.stroke()
   319→    ctx.fillStyle = '#c8cdd2'
   320→    rr(ctx, b.x - 4, y + 5, 8, 5, 1.5)
   321→    ctx.fill()
   322→    // Нить накала
   323→    ctx.strokeStyle = '#e8b23d'
   324→    ctx.lineWidth = 1.4
   325→    ctx.beginPath()
   326→    ctx.moveTo(b.x - 3, y + 1)
   327→    ctx.quadraticCurveTo(b.x, y - 5, b.x + 3, y + 1)
   328→    ctx.stroke()
   329→    // Блик-звёздочка
   330→    const a = 0.5 + 0.5 * Math.sin(g.frame * 0.1 + b.phase * 2)
   331→    ctx.strokeStyle = `rgba(255,255,255,${0.35 + a * 0.5})`
   332→    ctx.lineWidth = 1.6
   333→    ctx.beginPath()
   334→    ctx.moveTo(b.x + 6, y - 10)
   335→    ctx.lineTo(b.x + 6, y - 16)
   336→    ctx.moveTo(b.x + 3, y - 13)
   337→    ctx.lineTo(b.x + 9, y - 13)
   338→    ctx.stroke()
   339→  }
   340→}

   348→function drawLampGlows(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   349→  for (const l of g.lamps) {
   350→    if (!l.on) continue
   351→    const flicker = 1 + Math.sin(g.frame * 0.21 + l.x) * 0.06
   352→    const gl = ctx.createRadialGradient(l.x, l.y + 8, 3, l.x, l.y + 8, 34)
   353→    gl.addColorStop(0, `rgba(255,224,130,${0.4 * flicker})`)
   354→    gl.addColorStop(1, 'rgba(255,224,130,0)')
   355→    ctx.fillStyle = gl
   356→    ctx.fillRect(l.x - 34, l.y - 26, 68, 68)
   357→  }
   358→}

   360→function drawLamp(
   361→  ctx: CanvasRenderingContext2D,
   362→  g: EnergoGame,
   363→  x: number,
   364→  y: number,
   365→  on: boolean,
   366→) {
   367→  // Провод
   368→  ctx.strokeStyle = 'rgba(30,28,26,0.85)'
   369→  ctx.lineWidth = 2.5
   370→  ctx.beginPath()
   371→  ctx.moveTo(x, 0)
   372→  ctx.lineTo(x, y - 14)
   373→  ctx.stroke()
   374→
   375→  if (on) {
   376→    const len = 420
   377→    const flick = 0.16 + Math.sin(g.frame * 0.18 + x * 0.7) * 0.02
   378→    const grad = ctx.createLinearGradient(0, y, 0, y + len)
   379→    grad.addColorStop(0, `rgba(255,216,110,${flick + 0.1})`)
   380→    grad.addColorStop(1, 'rgba(255,216,110,0)')
   381→    ctx.fillStyle = grad
   382→    ctx.beginPath()
   383→    ctx.moveTo(x - 14, y + 6)
   384→    ctx.lineTo(x + 14, y + 6)
   385→    ctx.lineTo(x + 16 + len * 0.34, y + len)
   386→    ctx.lineTo(x - 16 - len * 0.34, y + len)
   387→    ctx.closePath()
   388→    ctx.fill()
   389→  }
   390→
   391→  // Плафон
   392→  ctx.fillStyle = on ? '#4a4150' : '#4c4c52'
   393→  ctx.beginPath()
   394→  ctx.moveTo(x - 10, y - 14)
   395→  ctx.lineTo(x + 10, y - 14)
   396→  ctx.lineTo(x + 17, y + 2)
   397→  ctx.lineTo(x - 17, y + 2)
   398→  ctx.closePath()
   399→  ctx.fill()
   400→  ctx.strokeStyle = 'rgba(0,0,0,0.35)'
   401→  ctx.lineWidth = 1.5
   402→  ctx.stroke()
   403→
   404→  // Лампочка
   405→  ctx.fillStyle = on ? '#ffdf8a' : '#83838a'
   406→  ctx.beginPath()
   407→  ctx.arc(x, y + 7, 6, 0, Math.PI * 2)
   408→  ctx.fill()
   409→  if (on) {
   410→    ctx.strokeStyle = 'rgba(255,255,255,0.8)'
   411→    ctx.lineWidth = 1.4
   412→    ctx.beginPath()
   413→    ctx.arc(x - 1.5, y + 5.5, 2.6, Math.PI * 0.9, Math.PI * 1.6)
   414→    ctx.stroke()
   415→  }
   416→}

   418→function drawSwitches(ctx: CanvasRenderingContext2D, g: EnergoGame) {

   420→    const x = s.x
   421→    const y = s.y

   423→    ctx.fillStyle = '#3c3c44'
   424→    rr(ctx, x - 14, y - 36, 28, 36, 4)
   425→    ctx.fill()
   426→    ctx.strokeStyle = '#23232a'
   427→    ctx.lineWidth = 2
   428→    ctx.stroke()
   429→    // Индикатор
   430→    ctx.fillStyle = s.flipped ? '#63d471' : '#ff5a4e'
   431→    ctx.beginPath()
   432→    ctx.arc(x, y - 29, 3, 0, Math.PI * 2)
   433→    ctx.fill()
   434→    // Рычаг
   435→    ctx.strokeStyle = '#c8cdd2'
   436→    ctx.lineWidth = 4.5
   437→    ctx.lineCap = 'round'
   438→    ctx.beginPath()
   439→    ctx.moveTo(x, y - 16)
   440→    ctx.lineTo(x, s.flipped ? y - 7 : y - 25)
   441→    ctx.stroke()
   442→    ctx.fillStyle = s.flipped ? '#63d471' : '#ff5a4e'
   443→    ctx.beginPath()
   444→    ctx.arc(x, s.flipped ? y - 7 : y - 25, 4.2, 0, Math.PI * 2)
   445→    ctx.fill()
   446→    ctx.lineCap = 'butt'
   447→
   448→    if (!s.flipped) {
   449→      // Подсказка — прыгающая стрелка
   450→      const bob = Math.sin(g.frame * 0.12) * 3.5
   451→      ctx.fillStyle = '#ffe873'
   452→      ctx.beginPath()
   453→      ctx.moveTo(x - 9, y - 50 + bob)
   454→      ctx.lineTo(x + 9, y - 50 + bob)
   455→      ctx.lineTo(x, y - 40 + bob)
   456→      ctx.closePath()
   457→      ctx.fill()
   458→      ctx.strokeStyle = 'rgba(0,0,0,0.4)'
   459→      ctx.lineWidth = 1.5
   460→      ctx.stroke()
   461→    }
   462→  }
   463→}

   816→function drawParticles(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   817→  for (const q of g.particles) {
   818→    const a = q.life / q.maxLife
   819→    ctx.globalAlpha = Math.max(0, Math.min(1, a))
   820→    if (q.kind === 'star') {
   821→      ctx.save()
   822→      ctx.translate(q.x, q.y)
   823→      ctx.rotate(q.life * 0.2)
   824→      ctx.fillStyle = q.color
   825→      const s = q.size * 1.6
   826→      ctx.beginPath()
   827→      for (let i = 0; i < 8; i++) {
   828→        const r = i % 2 === 0 ? s : s * 0.42
   829→        const ang = (i / 8) * Math.PI * 2
   830→        const px = Math.cos(ang) * r
   831→        const py = Math.sin(ang) * r
   832→        if (i === 0) ctx.moveTo(px, py)
   833→        else ctx.lineTo(px, py)
   834→      }
   835→      ctx.closePath()
   836→      ctx.fill()
   837→      ctx.restore()
   838→    } else if (q.kind === 'spark') {
   839→      ctx.strokeStyle = q.color
   840→      ctx.lineWidth = q.size * 0.6
   841→      ctx.beginPath()
   842→      ctx.moveTo(q.x, q.y)
   843→      ctx.lineTo(q.x - q.vx * 2.4, q.y - q.vy * 2.4)
   844→      ctx.stroke()
   845→    } else {
   846→      ctx.fillStyle = q.color
   847→      ctx.beginPath()
   848→      ctx.arc(q.x, q.y, q.size * (q.kind === 'poof' ? 1 + (1 - a) * 1.6 : 1), 0, Math.PI * 2)
   849→      ctx.fill()
   850→    }
   851→  }
   852→  ctx.globalAlpha = 1
   853→}
