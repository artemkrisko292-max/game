    44→export function render(ctx: CanvasRenderingContext2D, g: EnergoGame) {
    45→  const theme = THEMES[LEVELS[g.levelIndex].theme]
    46→  ctx.clearRect(0, 0, VIEW_W, VIEW_H)
    47→  ctx.save()
    48→
    49→  // Тряска камеры
    50→  if (g.shake > 0.3) {
    51→    ctx.translate(
    52→      (Math.random() - 0.5) * g.shake,
    53→      (Math.random() - 0.5) * g.shake,
    54→    )
    55→  }
    56→
    57→  drawBackground(ctx, g, theme)
    58→
    59→  ctx.save()
    60→  ctx.translate(-Math.round(g.camX), -Math.round(g.camY))
    61→
    62→  drawTiles(ctx, g, theme)
    63→  drawHazards(ctx, g)
    64→  drawSwitches(ctx, g)
    65→  drawExit(ctx, g)
    66→  drawLampCones(ctx, g)
    67→  drawBulbs(ctx, g)
    68→  drawPlats(ctx, g)
    69→  drawVampires(ctx, g)
    70→  drawPhantoms(ctx, g)
    71→  drawPlayer(ctx, g)
    72→  drawParticles(ctx, g)
    73→  drawLampGlows(ctx, g)
    74→
    75→  ctx.restore()
    76→  ctx.restore()
    77→
    78→  drawHUD(ctx, g)
    79→}

    83→function drawBackground(
    84→  ctx: CanvasRenderingContext2D,
    85→  g: EnergoGame,
    86→  th: Theme,
    87→) {
    88→  const grad = ctx.createLinearGradient(0, 0, 0, VIEW_H)
    89→  grad.addColorStop(0, th.skyTop)
    90→  grad.addColorStop(1, th.skyBot)
    91→  ctx.fillStyle = grad
    92→  ctx.fillRect(0, 0, VIEW_W, VIEW_H)
    93→
    94→  const name = LEVELS[g.levelIndex].theme
    95→  if (name === 'bedroom') drawBedroomBg(ctx, g)
    96→  else if (name === 'kitchen') drawKitchenBg(ctx, g)
    97→  else drawBasementBg(ctx, g)
    98→}

   100→function wrapX(i: number, camX: number, k: number, span: number): number {
   101→  const m = (i * 137.13 - camX * k) % span
   102→  return (m < 0 ? m + span : m) - 60
   103→}

   106→  // Звёзды
   107→  ctx.fillStyle = 'rgba(255,240,210,0.8)'
   108→  for (let i = 0; i < 46; i++) {
   109→    const x = wrapX(i * 89, g.camX, 0.22, VIEW_W + 120)
   110→    const y = (i * 53) % 260
   111→    const s = 1 + ((i * 7) % 3) * 0.5
   112→    const tw = 0.5 + 0.5 * Math.sin(g.frame * 0.04 + i * 1.7)
   113→    ctx.globalAlpha = 0.35 + tw * 0.5
   114→    ctx.fillRect(x, y, s, s)
   115→  }
   116→  ctx.globalAlpha = 1

   122→  mg.addColorStop(1, 'rgba(255,236,180,0)')
   123→  ctx.fillStyle = mg
   124→  ctx.fillRect(mx - 70, 22, 140, 140)
   125→  ctx.fillStyle = '#f7e6ae'
   126→  ctx.beginPath()
   127→  ctx.arc(mx, 92, 26, 0, Math.PI * 2)
   128→  ctx.fill()
   129→  ctx.fillStyle = 'rgba(214,190,130,0.7)'
   130→  ctx.beginPath()
   131→  ctx.arc(mx - 8, 86, 5, 0, Math.PI * 2)
   132→  ctx.arc(mx + 9, 99, 4, 0, Math.PI * 2)
   133→  ctx.fill()
   134→
   135→  // Полосы обоев
   136→  ctx.strokeStyle = 'rgba(255,255,255,0.045)'

   216→function hash2(c: number, r: number): number {
   217→  return ((c * 73856093) ^ (r * 19349663)) % 1000 / 1000
   218→}

   220→function drawTiles(
   221→  ctx: CanvasRenderingContext2D,
   222→  g: EnergoGame,
   223→  th: Theme,
   224→) {
   225→  const c0 = Math.max(0, Math.floor(g.camX / TILE) - 1)
   226→  const c1 = Math.min(g.cols - 1, Math.ceil((g.camX + VIEW_W) / TILE) + 1)
   227→  const r0 = Math.max(0, Math.floor(g.camY / TILE) - 1)
   228→  const r1 = Math.min(g.rows - 1, Math.ceil((g.camY + VIEW_H) / TILE) + 1)
   229→
   230→  for (let r = r0; r <= r1; r++) {
   231→    for (let c = c0; c <= c1; c++) {
   232→      if (g.grid[r * g.cols + c] !== 1) continue
   233→      const x = c * TILE
   234→      const y = r * TILE
   235→      ctx.fillStyle = th.wall
   236→      ctx.fillRect(x, y, TILE, TILE)
   237→
   238→      // Тёмный низ
   239→      if (r + 1 >= g.rows || g.grid[(r + 1) * g.cols + c] !== 1) {
   240→        ctx.fillStyle = th.wallDark
   241→        ctx.fillRect(x, y + TILE - 6, TILE, 6)
   242→      }
   243→      // Светлый верх
   244→      if (r === 0 || g.grid[(r - 1) * g.cols + c] !== 1) {
   245→        ctx.fillStyle = th.wallTop
   246→        ctx.fillRect(x, y, TILE, 7)
   247→        ctx.fillStyle = th.accent
   248→        ctx.fillRect(x, y + 7, TILE, 2)
   249→      }
   250→      // Пятнышки/шов
   251→      const h = hash2(c, r)
   252→      ctx.fillStyle = th.wallDark
   253→      ctx.globalAlpha = 0.35
   254→      if (h < 0.3) {
   255→        ctx.fillRect(x + 6 + h * 60, y + 12 + h * 40, 8, 4)
   256→      } else if (h < 0.55) {
   257→        ctx.fillRect(x + 24, y + 8 + h * 30, 4, 10)
   258→      } else if (h < 0.75) {
   259→        ctx.fillRect(x + 4, y + 22, 12, 3)
   260→      }
   261→      ctx.globalAlpha = 1
   262→      // Тонкая рамка
   263→      ctx.strokeStyle = 'rgba(0,0,0,0.18)'
   264→      ctx.lineWidth = 1
   265→      ctx.strokeRect(x + 0.5, y + 0.5, TILE - 1, TILE - 1)
   266→    }
   267→  }
   268→}

   272→function drawBulbs(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   273→  for (const b of g.bulbs) {
   274→    if (b.taken) continue
   275→    const y = b.y + Math.sin(g.frame * 0.05 + b.phase) * 4
   276→    const gl = ctx.createRadialGradient(b.x, y, 2, b.x, y, 22)
   277→    gl.addColorStop(0, 'rgba(255,232,130,0.55)')
   278→    gl.addColorStop(1, 'rgba(255,232,130,0)')
   279→    ctx.fillStyle = gl
   280→    ctx.fillRect(b.x - 22, y - 22, 44, 44)
   281→
   282→    ctx.fillStyle = '#fff6c9'
   283→    ctx.strokeStyle = '#d9a441'
   284→    ctx.lineWidth = 2
   285→    ctx.beginPath()
   286→    ctx.arc(b.x, y - 2, 8, 0, Math.PI * 2)
   287→    ctx.fill()
   288→    ctx.stroke()
   289→    ctx.fillStyle = '#c8cdd2'
   290→    rr(ctx, b.x - 4, y + 5, 8, 5, 1.5)
   291→    ctx.fill()
   292→    // Нить накала
   293→    ctx.strokeStyle = '#e8b23d'
   294→    ctx.lineWidth = 1.4
   295→    ctx.beginPath()
   296→    ctx.moveTo(b.x - 3, y + 1)
   297→    ctx.quadraticCurveTo(b.x, y - 5, b.x + 3, y + 1)
   298→    ctx.stroke()
   299→    // Блик-звёздочка
   300→    const a = 0.5 + 0.5 * Math.sin(g.frame * 0.1 + b.phase * 2)
   301→    ctx.strokeStyle = `rgba(255,255,255,${0.35 + a * 0.5})`
   302→    ctx.lineWidth = 1.6
   303→    ctx.beginPath()
   304→    ctx.moveTo(b.x + 6, y - 10)
   305→    ctx.lineTo(b.x + 6, y - 16)
   306→    ctx.moveTo(b.x + 3, y - 13)
   307→    ctx.lineTo(b.x + 9, y - 13)
   308→    ctx.stroke()
   309→  }
   310→}

   318→function drawLampGlows(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   319→  for (const l of g.lamps) {
   320→    if (!l.on) continue
   321→    const flicker = 1 + Math.sin(g.frame * 0.21 + l.x) * 0.06
   322→    const gl = ctx.createRadialGradient(l.x, l.y + 8, 3, l.x, l.y + 8, 34)
   323→    gl.addColorStop(0, `rgba(255,224,130,${0.4 * flicker})`)
   324→    gl.addColorStop(1, 'rgba(255,224,130,0)')
   325→    ctx.fillStyle = gl
   326→    ctx.fillRect(l.x - 34, l.y - 26, 68, 68)
   327→  }
   328→}

   330→function drawLamp(
   331→  ctx: CanvasRenderingContext2D,
   332→  g: EnergoGame,
   333→  x: number,
   334→  y: number,
   335→  on: boolean,
   336→) {
   337→  // Провод
   338→  ctx.strokeStyle = 'rgba(30,28,26,0.85)'
   339→  ctx.lineWidth = 2.5
   340→  ctx.beginPath()
   341→  ctx.moveTo(x, 0)
   342→  ctx.lineTo(x, y - 14)
   343→  ctx.stroke()
   344→
   345→  if (on) {
   346→    const len = 420
   347→    const flick = 0.16 + Math.sin(g.frame * 0.18 + x * 0.7) * 0.02
   348→    const grad = ctx.createLinearGradient(0, y, 0, y + len)
   349→    grad.addColorStop(0, `rgba(255,216,110,${flick + 0.1})`)
   350→    grad.addColorStop(1, 'rgba(255,216,110,0)')
   351→    ctx.fillStyle = grad
   352→    ctx.beginPath()
   353→    ctx.moveTo(x - 14, y + 6)
   354→    ctx.lineTo(x + 14, y + 6)
   355→    ctx.lineTo(x + 16 + len * 0.34, y + len)
   356→    ctx.lineTo(x - 16 - len * 0.34, y + len)
   357→    ctx.closePath()
   358→    ctx.fill()
   359→  }
   360→
   361→  // Плафон
   362→  ctx.fillStyle = on ? '#4a4150' : '#4c4c52'
   363→  ctx.beginPath()
   364→  ctx.moveTo(x - 10, y - 14)
   365→  ctx.lineTo(x + 10, y - 14)
   366→  ctx.lineTo(x + 17, y + 2)
   367→  ctx.lineTo(x - 17, y + 2)
   368→  ctx.closePath()
   369→  ctx.fill()
   370→  ctx.strokeStyle = 'rgba(0,0,0,0.35)'
   371→  ctx.lineWidth = 1.5
   372→  ctx.stroke()
   373→
   374→  // Лампочка
   375→  ctx.fillStyle = on ? '#ffdf8a' : '#83838a'
   376→  ctx.beginPath()
   377→  ctx.arc(x, y + 7, 6, 0, Math.PI * 2)
   378→  ctx.fill()
   379→  if (on) {
   380→    ctx.strokeStyle = 'rgba(255,255,255,0.8)'
   381→    ctx.lineWidth = 1.4
   382→    ctx.beginPath()
   383→    ctx.arc(x - 1.5, y + 5.5, 2.6, Math.PI * 0.9, Math.PI * 1.6)
   384→    ctx.stroke()
   385→  }
   386→}

   397→    ctx.lineWidth = 2
   398→    ctx.stroke()

   401→    ctx.beginPath()

   407→    ctx.lineCap = 'round'
   408→    ctx.beginPath()
   409→    ctx.moveTo(x, y - 16)
   410→    ctx.lineTo(x, s.flipped ? y - 7 : y - 25)
   411→    ctx.stroke()
   412→    ctx.fillStyle = s.flipped ? '#63d471' : '#ff5a4e'

   414→    ctx.arc(x, s.flipped ? y - 7 : y - 25, 4.2, 0, Math.PI * 2)

   416→    ctx.lineCap = 'butt'

   428→      ctx.strokeStyle = 'rgba(0,0,0,0.4)'
   429→      ctx.lineWidth = 1.5
   430→      ctx.stroke()
   431→    }
   432→  }

   482→    const y = h.y

   486→    ctx.strokeStyle = '#202024'
   487→    ctx.lineWidth = 2
   488→    ctx.stroke()
   489→    ctx.fillStyle = '#17171a'

   785→function drawParticles(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   786→  for (const q of g.particles) {
   787→    const a = q.life / q.maxLife
   788→    ctx.globalAlpha = Math.max(0, Math.min(1, a))
   789→    if (q.kind === 'star') {

   792→      ctx.rotate(q.life * 0.2)
   793→      ctx.fillStyle = q.color
   794→      const s = q.size * 1.6
   795→      ctx.beginPath()
   796→      for (let i = 0; i < 8; i++) {
   797→        const r = i % 2 === 0 ? s : s * 0.42
   798→        const ang = (i / 8) * Math.PI * 2
   799→        const px = Math.cos(ang) * r
   800→        const py = Math.sin(ang) * r
   801→        if (i === 0) ctx.moveTo(px, py)
   802→        else ctx.lineTo(px, py)
   803→      }
   804→      ctx.closePath()
   805→      ctx.fill()
   806→      ctx.restore()
   807→    } else if (q.kind === 'spark') {
   808→      ctx.strokeStyle = q.color
   809→      ctx.lineWidth = q.size * 0.6
   810→      ctx.beginPath()
   811→      ctx.moveTo(q.x, q.y)
   812→      ctx.lineTo(q.x - q.vx * 2.4, q.y - q.vy * 2.4)
   813→      ctx.stroke()
   814→    } else {
   815→      ctx.fillStyle = q.color
   816→      ctx.beginPath()
   817→      ctx.arc(q.x, q.y, q.size * (q.kind === 'poof' ? 1 + (1 - a) * 1.6 : 1), 0, Math.PI * 2)
   818→      ctx.fill()
   819→    }
   820→  }
   821→  ctx.globalAlpha = 1
   822→}

   958→  y: number,

   970→  ctx.closePath()
