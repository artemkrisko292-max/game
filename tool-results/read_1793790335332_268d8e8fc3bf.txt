   512→   512→    )
   513→   513→
   514→   514→    this.player = {
   515→   515→      x: px,
   516→   516→      y: py,
   517→   517→      w: 26,

   521→   521→      onGround: false,

   523→   523→      buffer: 0,
   524→   524→      facing: 1,
   525→   525→      invuln: 0,
   526→   526→      anim: 0,
   527→   527→      blink: 120,
   528→   528→      squash: 1,
   529→   529→      safeX: px,
   530→   530→      safeY: py,
   531→   531→    }
   532→   532→    this.energy = ENERGY_MAX
   533→   533→    this.camX = 0
   534→   534→    this.camY = 0
   535→   535→    this.shake = 0

   557→   557→  private onKeyDown = (e: KeyboardEvent) => {
   558→   558→    const c = this.keyId(e)
   559→   559→    if (this.GAME_KEYS.has(c)) e.preventDefault()
   560→   560→    switch (c) {
   561→   561→      case 'ArrowLeft':
   562→   562→      case 'KeyA':
   563→   563→        this.kb.left = true
   564→   564→        break
   565→   565→      case 'ArrowRight':
   566→   566→      case 'KeyD':
   567→   567→        this.kb.right = true
   568→   568→        break
   569→   569→      case 'Space':
   570→   570→      case 'ArrowUp':
   571→   571→      case 'KeyW':
   572→   572→      case 'KeyZ':
   573→   573→        this.kb.jump = true
   574→   574→        this.sfx.ensure()
   575→   575→        break
   576→   576→      case 'Escape':
   577→   577→      case 'KeyP':
   578→   578→        if (this.phase === 'playing' || this.phase === 'paused')
   579→   579→          this.togglePause()
   580→   580→        break
   581→   581→    }
   582→   582→  }

   586→   586→      case 'ArrowLeft':
   587→   587→      case 'KeyA':
   588→   588→        this.kb.left = false
   589→   589→        break
   590→   590→      case 'ArrowRight':
   591→   591→      case 'KeyD':
   592→   592→        this.kb.right = false
   593→   593→        break
   594→   594→      case 'Space':
   595→   595→      case 'ArrowUp':
   596→   596→      case 'KeyW':
   597→   597→      case 'KeyZ':
   598→   598→        this.kb.jump = false
   599→   599→        break

   604→   604→  // тогда откатываемся к e.key ('ArrowLeft', 'a', ' ', 'Enter').
   605→   605→  private keyId(e: KeyboardEvent): string {
   606→   606→    if (e.code) return e.code
   607→   607→    const k = e.key
   608→   608→    if (k === ' ') return 'Space'
   609→   609→    return k.length === 1 ? `Key${k.toUpperCase()}` : k
   610→   610→  }

   628→   628→  private loop = (t: number) => {
   629→   629→    if (this.destroyed) return
   630→   630→    this.raf = requestAnimationFrame(this.loop)
   631→   631→    let dt = (t - this.lastT) / 1000
   632→   632→    this.lastT = t
   633→   633→    if (dt > 0.1) dt = 0.1
   634→   634→    if (this.phase === 'playing') {
   635→   635→      this.acc += dt
   636→   636→      let steps = 0
   637→   637→      while (this.acc >= STEP && steps < 4) {
   638→   638→        this.step()
   639→   639→        this.acc -= STEP
   640→   640→        steps++
   641→   641→      }
   642→   642→      if (steps === 4) this.acc = 0
   643→   643→    } else if (this.phase === 'victory') {
   644→   644→      // Финальный салют: сцена живёт под отчётом
   645→   645→      this.celebrate()
   646→   646→    }
   647→   647→    render(this.ctx, this)
   648→   648→  }

   650→   650→  /** Праздничный такт: заряды салюта, полёт частиц, лёгкая дрожь камеры. */
   651→   651→  private celebrate() {
   652→   652→    this.frame++
   653→   653→    if (this.shake > 0) this.shake *= 0.85
   654→   654→    if (this.frame % 30 === 0) {
   655→   655→      const fx =
   656→   656→        this.camX + 70 + Math.random() * (VIEW_W - 140)
   657→   657→      const fy =
   658→   658→        this.camY + 50 + Math.random() * (VIEW_H * 0.5)
   659→   659→      this.firework(fx, fy)
   660→   660→    }
   661→   661→    this.updateParticles()
   662→   662→  }

   664→   664→  /** Один залп: венок цветных искр с мягким хлопком. */
   665→   665→  private firework(x: number, y: number) {
   666→   666→    const palette = [
   667→   667→      '#ffd94d',
   668→   668→      '#ff8a3d',
   669→   669→      '#a4f04a',
   670→   670→      '#7ec8ff',
   671→   671→      '#ff6b9d',
   672→   672→      '#fff3b0',
   673→   673→    ]
   674→   674→    const col = palette[Math.floor(Math.random() * palette.length)]
   675→   675→    this.shake = 1.6
   676→   676→    this.sfx.pop()
   677→   677→    for (let i = 0; i < 26; i++) {
   678→   678→      const a = (i / 26) * Math.PI * 2 + Math.random() * 0.22
   679→   679→      const sp = 2.1 + Math.random() * 2.3
   680→   680→      this.particles.push({
   681→   681→        x,
   682→   682→        y,
   683→   683→        vx: Math.cos(a) * sp,
   684→   684→        vy: Math.sin(a) * sp,
   685→   685→        life: 34 + Math.random() * 22,
   686→   686→        maxLife: 56,
   687→   687→        size: 2 + Math.random() * 2.6,
   688→   688→        color: Math.random() < 0.75 ? col : '#fff8dc',
   689→   689→        kind: 'star',
   690→   690→        grav: 0.055,
   691→   691→      })
   692→   692→    }
   693→   693→    if (this.particles.length > 320) this.particles.splice(0, 110)
   694→   694→  }

   696→   696→  // ---------- Шаг симуляции ----------

   698→   698→  private step() {
   699→   699→    const p = this.player
   700→   700→    const inp = this.input
   701→   701→    this.frame++
   702→   702→    this.levelFrames++
   703→   703→    if (this.shake > 0) this.shake *= 0.85
   704→   704→
   705→   705→    // Прыжок: буфер + койот-время
   706→   706→    const jumpPressed = inp.jump && !this.prevJump
   707→   707→    this.prevJump = inp.jump
   708→   708→    if (jumpPressed) p.buffer = BUFFER
   709→   709→    if (p.buffer > 0) p.buffer--
   710→   710→    if (p.coyote > 0) p.coyote--
   711→   711→
   712→   712→    // Горизонталь
   713→   713→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   714→   714→    if (inp.left && !inp.right) {
   715→   715→      p.vx -= acc
   716→   716→      p.facing = -1
   717→   717→    } else if (inp.right && !inp.left) {
   718→   718→      p.vx += acc
   719→   719→      p.facing = 1
   720→   720→    } else {
   721→   721→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   722→   722→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   723→   723→    }
   724→   724→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   725→   725→
   726→   726→    // Прыжок
   727→   727→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   728→   728→      p.vy = JUMP_V
   729→   729→      p.onGround = false
   730→   730→      p.coyote = 0
   731→   731→      p.buffer = 0
   732→   732→      p.squash = 1.18
   733→   733→      this.sfx.jump()
   734→   734→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   735→   735→    }
   736→   736→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   737→   737→
   738→   738→    // Гравитация
   739→   739→    p.vy = Math.min(p.vy + GRAV, 14)
   740→   740→
   741→   741→    // Движущиеся платформы (горизонтальные и вертикальные)
   742→   742→    for (const pl of this.plats) {
   743→   743→      pl.phase += pl.speed
   744→   744→      if (pl.axis === 'x') {
   745→   745→        const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   746→   746→        pl.dx = nx - pl.x
   747→   747→        pl.dy = 0
   748→   748→        pl.x = nx
   749→   749→      } else {
   750→   750→        const ny = pl.by + Math.sin(pl.phase) * pl.amp
   751→   751→        pl.dy = ny - pl.y
   752→   752→        pl.dx = 0
   753→   753→        pl.y = ny
   754→   754→      }
   755→   755→    }
   756→   756→
   757→   757→    // Движение и коллизии
   758→   758→    this.moveX(p, p.vx)
   759→   759→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   760→   760→
   761→   761→    const wasOnGround = p.onGround
   762→   762→    const prevFeet = p.y + p.h
   763→   763→    this.moveY(p, p.vy, prevFeet)
   764→   764→
   765→   765→    if (p.onGround) {
   766→   766→      if (!wasOnGround) {
   767→   767→        p.squash = 0.78
   768→   768→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   769→   769→        this.sfx.land()
   770→   770→      }
   771→   771→      p.coyote = COYOTE
   772→   772→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   773→   773→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   774→   774→      if (underL && underR) {
   775→   775→        p.safeX = p.x
   776→   776→        p.safeY = p.y
   777→   777→      }
   778→   778→    }
   779→   779→    p.squash += (1 - p.squash) * 0.18
   780→   780→    p.anim += Math.abs(p.vx) * 0.09
   781→   781→    if (p.invuln > 0) p.invuln--
   782→   782→    if (p.blink > 0) p.blink--
   783→   783→    else if (Math.random() < 0.012) p.blink = 8
   784→   784→
   785→   785→    // Провал за нижнюю границу мира (люки)
   786→   786→    if (p.y > this.worldH + 60) {
   787→   787→      this.damage(DMG_FALL, 0, 'fall')
   788→   788→      p.x = p.safeX
   789→   789→      p.y = p.safeY
   790→   790→      p.vx = 0
   791→   791→      p.vy = 0
   792→   792→    }
   793→   793→
   794→   794→    this.updateWalkers(this.vampires, DMG_VAMPIRE, 100, 'vampire')
   795→   795→    this.updateWalkers(this.runners, DMG_RUNNER, SCORE_RUNNER, 'runner')
   796→   796→    this.updateDrones()
   797→   797→    this.updatePhantoms()
   798→   798→    this.updateCrumbs()
   799→   799→    this.updateFuses()
   800→   800→    this.updatePickups()
   801→   801→    this.updateSwitches()
   802→   802→    this.updateHazards()
   803→   803→    this.updateExit()
   804→   804→    this.updateEnergy()
   805→   805→    this.updateParticles()
   806→   806→    this.updateCamera()
   807→   807→  }

   809→   809→  // ---------- Коллизии с тайлами ----------
   810→   810→
   811→   811→  solidAt(px: number, py: number): boolean {
   812→   812→    const c = Math.floor(px / TILE)
   813→   813→    const r = Math.floor(py / TILE)
   814→   814→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   815→   815→    return this.grid[r * this.cols + c] === 1
   816→   816→  }
   817→   817→
   818→   818→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   819→   819→    const c0 = Math.floor(x / TILE)
   820→   820→    const c1 = Math.floor((x + w - EPS) / TILE)
   821→   821→    const r0 = Math.floor(y / TILE)
   822→   822→    const r1 = Math.floor((y + h - EPS) / TILE)
   823→   823→    for (let r = r0; r <= r1; r++) {
   824→   824→      for (let c = c0; c <= c1; c++) {
   825→   825→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   826→   826→        if (this.grid[r * this.cols + c] === 1) return true
   827→   827→      }
   828→   828→    }
   829→   829→    return false
   830→   830→  }
   831→   831→
   832→   832→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   833→   833→    e.x += dx
   834→   834→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   835→   835→    if (dx > 0) {
   836→   836→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   837→   837→    } else if (dx < 0) {
   838→   838→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   839→   839→    }
   840→   840→    e.vx = 0
   841→   841→    return true
   842→   842→  }
   843→   843→
   844→   844→  private moveY(
   845→   845→    e: Box & { vy: number; onGround: boolean },
   846→   846→    dy: number,
   847→   847→    prevFeet: number,
   848→   848→  ): void {
   849→   849→    e.y += dy
   850→   850→    if (dy >= 0) {
   851→   851→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   852→   852→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   853→   853→        e.vy = 0
   854→   854→        e.onGround = true
   855→   855→      }
   856→   856→      for (const pl of this.plats) {
   857→   857→        const feet = e.y + e.h
   858→   858→        if (
   859→   859→          e.vy >= 0 &&
   860→   860→          feet >= pl.y &&
   861→   861→          prevFeet <= pl.y + 6 &&
   862→   862→          e.x + e.w > pl.x &&
   863→   863→          e.x < pl.x + pl.w
   864→   864→        ) {
   865→   865→          e.y = pl.y - e.h
   866→   866→          e.vy = 0
   867→   867→          e.onGround = true
   868→   868→          e.x += pl.dx
   869→   869→        }
   870→   870→      }
   871→   871→      // Осыпающиеся плиты: та же «односторонняя» коллизия сверху
   872→   872→      for (const c of this.crumbs) {
   873→   873→        if (c.state === 2) continue
   874→   874→        const feet = e.y + e.h
   875→   875→        if (
   876→   876→          feet >= c.y &&
   877→   877→          prevFeet <= c.y + 6 &&
   878→   878→          e.x + e.w > c.x + 2 &&
   879→   879→          e.x < c.x + c.w - 2
   880→   880→        ) {
   881→   881→          e.y = c.y - e.h
   882→   882→          e.vy = 0
   883→   883→          e.onGround = true
   884→   884→        }
   885→   885→      }
   886→   886→    } else {
   887→   887→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   888→   888→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   889→   889→        e.vy = 0
   890→   890→      }
   891→   891→    }
   892→   892→  }

   896→   896→  /** Наземные ходячие враги (вампиры и гонцы). */
   897→   897→  private updateWalkers(
   898→   898→    list: Vampire[],
   899→   899→    dmg: number,
   900→   900→    stompScore: number,
   901→   901→    cause: DeathCause,
   902→   902→  ) {
   903→   903→    const p = this.player
   904→   904→    for (const v of list) {
   905→   905→      if (v.dead) continue
   906→   906→      v.anim += Math.abs(v.vx) * 0.06
   907→   907→      v.vy = Math.min(v.vy + GRAV, 12)
   908→   908→      const oldVx = v.vx
   909→   909→      this.moveX(v, v.vx)
   910→   910→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   911→   911→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   912→   912→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   913→   913→
   914→   914→      const prevFeet = v.y + v.h
   915→   915→      const wasGround = v.onGround
   916→   916→      v.onGround = false
   917→   917→      this.moveY(v, v.vy, prevFeet)
   918→   918→      if (v.onGround && !wasGround) v.vy = 0
   919→   919→      // разворот у обрыва
   920→   920→      if (v.onGround) {
   921→   921→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   922→   922→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   923→   923→      }
   924→   924→
   925→   925→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   926→   926→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   927→   927→          v.dead = true
   928→   928→          p.vy = -9.5
   929→   929→          this.score += stompScore
   930→   930→          this.shake = 6
   931→   931→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   932→   932→          this.sfx.stomp()
   933→   933→        } else {
   934→   934→          this.damage(dmg, p.x < v.x + v.w / 2 ? -1 : 1, cause)
   935→   935→        }
   936→   936→      }
   937→   937→    }
   938→   938→  }

   940→   940→  /** Дроны-пылесосы: летают по диагонали, отскакивая от стен. */
   941→   941→  private updateDrones() {
   942→   942→    const p = this.player
   943→   943→    for (const d of this.drones) {
   944→   944→      if (d.dead) continue
   945→   945→      d.anim += 0.12
   946→   946→      const oldVx = d.vx
   947→   947→      this.moveX(d, d.vx)
   948→   948→      if (d.vx === 0 && oldVx !== 0) d.vx = -oldVx
   949→   949→
   950→   950→      const oldVy = d.vy
   951→   951→      this.moveY(d, d.vy, d.y + d.h)
   952→   952→      if (d.vy === 0 && oldVy !== 0) d.vy = -oldVy // пол или потолок
   953→   953→
   954→   954→      // Границы мира: не даём дрону улететь в пустоту
   955→   955→      if (d.x <= 0) {
   956→   956→        d.x = 0
