   656→   656→/** Плавный счётчик очков. */
   657→   657→function ScoreCount({ value }: { value: number }) {
   658→   658→  const [v, setV] = useState(0)
   659→   659→  useEffect(() => {
   660→   660→    let raf = 0
   661→   661→    const t0 = performance.now()
   662→   662→    const dur = 700
   663→   663→    const tick = (t: number) => {
   664→   664→      const k = Math.min(1, (t - t0) / dur)
   665→   665→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
   666→   666→      if (k < 1) raf = requestAnimationFrame(tick)
   667→   667→    }
   668→   668→    raf = requestAnimationFrame(tick)
   669→   669→    return () => cancelAnimationFrame(raf)
   670→   670→  }, [value])
   671→   671→  return <>{v}</>
   672→   672→}
