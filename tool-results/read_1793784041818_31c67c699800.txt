    47→    47→interface Progress {
    48→    48→  unlocked: number // сколько уровней открыто (1…20)
    49→    49→  stars: number[]
    50→    50→  best: number[]
    51→    51→}

    53→    53→const EMPTY_PROGRESS: Progress = {
    54→    54→  unlocked: 1,
    55→    55→  stars: Array(LEVELS.length).fill(0),
    56→    56→  best: Array(LEVELS.length).fill(0),
    57→    57→}

    59→    59→function loadProgress(): Progress {
    60→    60→  try {
    61→    61→    const raw = localStorage.getItem(PROGRESS_KEY)
    62→    62→    if (!raw) return EMPTY_PROGRESS
    63→    63→    const p = JSON.parse(raw) as Partial<Progress>
    64→    64→    const stars = Array(LEVELS.length).fill(0)
    65→    65→    const best = Array(LEVELS.length).fill(0)
    66→    66→    ;(p.stars ?? []).forEach((v, i) => (stars[i] = Number(v) || 0))
    67→    67→    ;(p.best ?? []).forEach((v, i) => (best[i] = Number(v) || 0))
    68→    68→    return {
    69→    69→      unlocked: Math.min(LEVELS.length, Math.max(1, Number(p.unlocked) || 1)),
    70→    70→      stars,
    71→    71→      best,
    72→    72→    }
    73→    73→  } catch {
    74→    74→    return EMPTY_PROGRESS
    75→    75→  }
    76→    76→}

    78→    78→function starsFromResult(r: LevelResult): number {
    79→    79→  const got = r.bulbs + r.switches + r.fuses
    80→    80→  const tot = r.bulbsTotal + r.switchesTotal + r.fusesTotal
    81→    81→  const pct = tot > 0 ? got / tot : 1
    82→    82→  return pct >= 0.9 ? 3 : pct >= 0.55 ? 2 : 1
    83→    83→}

    85→    85→function saveLevelProgress(
    86→    86→  levelIndex: number,
    87→    87→  result: LevelResult | null,
    88→    88→  summary?: RunSummary | null,
    89→    89→) {
    90→    90→  try {
    91→    91→    const p = loadProgress()
    92→    92→    const stars = result ? starsFromResult(result) : (summary?.stars ?? 1)
    93→    93→    const score = result ? result.score : (summary?.score ?? 0)
    94→    94→    p.stars[levelIndex] = Math.max(p.stars[levelIndex] ?? 0, stars)
    95→    95→    p.best[levelIndex] = Math.max(p.best[levelIndex] ?? 0, score)
    96→    96→    p.unlocked = Math.min(
    97→    97→      LEVELS.length,
    98→    98→      Math.max(p.unlocked, levelIndex + 2),
    99→    99→    )
   100→   100→    localStorage.setItem(PROGRESS_KEY, JSON.stringify(p))
   101→   101→  } catch {
   102→   102→    /* localStorage недоступен */
   103→   103→  }
   104→   104→}

   421→   421→    [act, playScene],

   428→   428→

   437→   437→      else if (phase === 'complete') nextWithStory()
   438→   438→      else if (phase === 'gameover') act('retry')

   440→   440→      else if (phase === 'paused') act('pause')

  1365→  1365→function ScoreCount({ value }: { value: number }) {
  1366→  1366→  const [v, setV] = useState(0)
  1367→  1367→  useEffect(() => {
  1368→  1368→    let raf = 0
  1369→  1369→    const t0 = performance.now()
  1370→  1370→    const dur = 700
  1371→  1371→    const tick = (t: number) => {
  1372→  1372→      const k = Math.min(1, (t - t0) / dur)
  1373→  1373→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
  1374→  1374→      if (k < 1) raf = requestAnimationFrame(tick)
  1375→  1375→    }
  1376→  1376→    raf = requestAnimationFrame(tick)
  1377→  1377→    return () => cancelAnimationFrame(raf)
  1378→  1378→  }, [value])
  1379→  1379→  return <>{v}</>
  1380→  1380→}
