    68→export function render(ctx: CanvasRenderingContext2D, g: EnergoGame) {
    69→  const theme = THEMES[LEVELS[g.levelIndex].theme]
    70→  ctx.clearRect(0, 0, VIEW_W, VIEW_H)
    71→  ctx.save()
    72→
    73→  // Тряска камеры
    74→  if (g.shake > 0.3) {
    75→    ctx.translate(
    76→      (Math.random() - 0.5) * g.shake,
    77→      (Math.random() - 0.5) * g.shake,
    78→    )
    79→  }
    80→
    81→  drawBackground(ctx, g, theme)
    82→
    83→  ctx.save()
    84→  ctx.translate(-Math.round(g.camX), -Math.round(g.camY))
    85→
    86→  // Мебель — ЗА геометрией уровня, приглушённая: читается задним планом
    87→  drawDecor(ctx, g)
    88→  // Воздушная дымка: весь тыл сцены (фон + интерьер) тонет в цвете комнаты,
    89→  // а геометрия, герой и объекты поверх остаются яркими — иерархия читается сразу
    90→  drawDepthWash(ctx, g, theme)
    91→  drawTiles(ctx, g, theme)
    92→  drawHazards(ctx, g)
    93→  drawSwitches(ctx, g)
    94→  drawExit(ctx, g)
    95→  drawLampCones(ctx, g)
    96→  drawBulbs(ctx, g)
    97→  drawFuses(ctx, g)
    98→  drawPlats(ctx, g)
    99→  drawCrumbs(ctx, g)
   100→  drawVampires(ctx, g)
   101→  drawRunners(ctx, g)
   102→  drawPhantoms(ctx, g)
   103→  drawDrones(ctx, g)
   104→  drawPlayer(ctx, g)
   105→  drawParticles(ctx, g)
   106→  drawLampGlows(ctx, g)
   107→
   108→  ctx.restore()
   109→  ctx.restore()
   110→
   111→  // Пылинки в воздухе — комната живая
   112→  drawAmbient(ctx, g)
   113→  drawHUD(ctx, g)
   114→}
   115→
   116→/**
   117→ * Дымка глубины: заливка видимой области полупрозрачным цветом комнаты.
   118→ * Идёт ПОСЛЕ фона и интерьера, но ДО геометрии — тыл сцены становится
   119→ * тусклее и «дальше», а игровые объекты выделяются контрастом.
   120→ */
   121→function drawDepthWash(
   122→  ctx: CanvasRenderingContext2D,
   123→  g: EnergoGame,
   124→  th: Theme,
   125→) {
   126→  ctx.fillStyle = th.skyTop
   127→  ctx.globalAlpha = 0.32
   128→  ctx.fillRect(g.camX - 4, g.camY - 4, VIEW_W + 8, VIEW_H + 8)
   129→  ctx.fillStyle = 'rgba(8,6,14,0.17)'
   130→  ctx.fillRect(g.camX - 4, g.camY - 4, VIEW_W + 8, VIEW_H + 8)
   131→  ctx.globalAlpha = 1
   132→}

   134→/** Медленно дрейфующие пылинки (экранное пространство). */
   135→function drawAmbient(ctx: CanvasRenderingContext2D, g: EnergoGame) {
   136→  for (let i = 0; i < 24; i++) {
   137→    const sp = 0.1 + (i % 5) * 0.045
   138→    const span = VIEW_W + 60
   139→    const x =
   140→      ((((i * 163 + g.frame * sp - g.camX * 0.25) % span) + span) % span) - 30
   141→    const vspan = VIEW_H + 40
   142→    const y =
   143→      ((((i * 97 - g.frame * (0.14 + (i % 3) * 0.05)) % vspan) + vspan) %
   144→        vspan) -
   145→      20 +
   146→      Math.sin(g.frame * 0.01 + i * 1.9) * 12
   147→    const a = 0.035 + 0.045 * (0.5 + 0.5 * Math.sin(g.frame * 0.025 + i * 2.4))
   148→    ctx.fillStyle = `rgba(255,240,200,${a.toFixed(3)})`
   149→    ctx.fillRect(x, y, 1.7, 1.7)
   150→  }
   151→}

   155→function drawBackground(
   156→  ctx: CanvasRenderingContext2D,
   157→  g: EnergoGame,
   158→  th: Theme,
   159→) {
   160→  const grad = ctx.createLinearGradient(0, 0, 0, VIEW_H)
   161→  grad.addColorStop(0, th.skyTop)
   162→  grad.addColorStop(1, th.skyBot)
   163→  ctx.fillStyle = grad
   164→  ctx.fillRect(0, 0, VIEW_W, VIEW_H)
   165→
   166→  // Каждой теме — свой художник фона: комната читается сразу.
   167→  const name = LEVELS[g.levelIndex].theme
   168→  switch (name) {
   169→    case 'bedroom':
   170→      drawBedroomBg(ctx, g)
   171→      break
   172→    case 'kitchen':
   173→      drawKitchenBg(ctx, g)
   174→      break
   175→    case 'basement':
   176→      drawBasementBg(ctx, g)
   177→      break
   178→    case 'living':
   179→      drawLivingBg(ctx, g)
   180→      break
   181→    case 'attic':
   182→      drawAtticBg(ctx, g)
   183→      break
   184→    case 'office':
   185→      drawOfficeBg(ctx, g)
   186→      break
   187→  }
   188→}

   190→function wrapX(i: number, camX: number, k: number, span: number): number {
   191→  const m = (i * 137.13 - camX * k) % span
   192→  return (m < 0 ? m + span : m) - 60
   193→}

   252→  for (let i = 0; i < 12; i++) {
   253→    const t = (i + 0.5) / 12
   254→    const bx = -40 + (VIEW_W + 80) * t

   256→      gy - 22 + Math.sin(Math.PI * t) * 48 + Math.sin(g.frame * 0.03 + i) * 1.5

   387→  for (const py of [110, 250]) {
   388→    ctx.fillStyle = 'rgba(90,110,95,0.5)'
   389→    ctx.fillRect(0, py, VIEW_W, 16)
   390→    ctx.fillStyle = 'rgba(60,75,64,0.6)'
   391→    ctx.fillRect(0, py + 12, VIEW_W, 4)
   392→    for (let i = 0; i < 8; i++) {
   393→      const x = wrapX(i * 260, g.camX, 0.4, VIEW_W + 500)
   394→      ctx.fillStyle = 'rgba(110,130,115,0.6)'
   395→      ctx.fillRect(x, py - 4, 18, 24)
   396→      // вентиль — медленно крутится
   397→      const vx = x + 130
   398→      if (vx < VIEW_W + 60) {
   399→        ctx.save()
   400→        ctx.translate(vx, py + 8)
   401→        ctx.rotate(g.frame * 0.006 + i)
   402→        ctx.strokeStyle = 'rgba(150,90,60,0.75)'
   403→        ctx.lineWidth = 3.4
   404→        ctx.beginPath()
   405→        ctx.arc(0, 0, 8, 0, Math.PI * 2)
   406→        ctx.moveTo(-8, 0)
   407→        ctx.lineTo(8, 0)
   408→        ctx.moveTo(0, -8)
   409→        ctx.lineTo(0, 8)
   410→        ctx.stroke()
   411→        ctx.restore()
   412→      }
   413→    }
   414→  }
   415→  // Лампы в клетках-защитах на потолке
   416→  for (let i = 0; i < 4; i++) {
   417→    const x = wrapX(i * 340 + 170, g.camX, 0.5, VIEW_W + 700)

   432→      ctx.moveTo(x - 12 - k * 3.4, 34)
   433→    }
   434→    ctx.strokeStyle = 'rgba(24,30,26,0.75)'
   435→    ctx.lineWidth = 1.4
   436→    ctx.stroke()
   437→    if (on) {
   438→      const gg = ctx.createRadialGradient(x, 34, 2, x, 34, 60)
   439→      gg.addColorStop(0, 'rgba(200,255,190,0.16)')
   440→      gg.addColorStop(1, 'rgba(200,255,190,0)')
   441→      ctx.fillStyle = gg
   442→      ctx.fillRect(x - 60, 0, 120, 110)
   443→    }

   451→  ctx.lineWidth = 2
   452→  ctx.stroke()
   453→  ctx.fillStyle = 'rgba(30,26,16,0.85)'
   454→  ctx.font = fnt(11, 800)

   462→    gl.addColorStop(1, 'rgba(140,230,140,0)')

   465→  }

   468→function drawLivingBg(ctx: CanvasRenderingContext2D, g: EnergoGame) {

   470→  ctx.strokeStyle = 'rgba(255,255,255,0.035)'

   789→function hash2(c: number, r: number): number {
   790→  const h = Math.imul(c | 0, 73856093) ^ Math.imul(r | 0, 19349663)
   791→  return ((h >>> 0) % 1000) / 1000
   792→}

   794→function drawTiles(
   795→  ctx: CanvasRenderingContext2D,
   796→  g: EnergoGame,
   797→  th: Theme,
   798→) {
   799→  const c0 = Math.max(0, Math.floor(g.camX / TILE) - 1)
   800→  const c1 = Math.min(g.cols - 1, Math.ceil((g.camX + VIEW_W) / TILE) + 1)
   801→  const r0 = Math.max(0, Math.floor(g.camY / TILE) - 1)
   802→  const r1 = Math.min(g.rows - 1, Math.ceil((g.camY + VIEW_H) / TILE) + 1)
   803→
   804→  for (let r = r0; r <= r1; r++) {
   805→    for (let c = c0; c <= c1; c++) {
   806→      if (g.grid[r * g.cols + c] !== 1) continue
   807→      const x = c * TILE
   808→      const y = r * TILE
   809→      ctx.fillStyle = th.wall
   810→      ctx.fillRect(x, y, TILE, TILE)
   811→
   812→      // Тёмный низ
   813→      if (r + 1 >= g.rows || g.grid[(r + 1) * g.cols + c] !== 1) {
   814→        ctx.fillStyle = th.wallDark
   815→        ctx.fillRect(x, y + TILE - 6, TILE, 6)
   816→      }
   817→      // Светлый верх
   818→      if (r === 0 || g.grid[(r - 1) * g.cols + c] !== 1) {
   819→        ctx.fillStyle = th.wallTop
   820→        ctx.fillRect(x, y, TILE, 7)
   821→        ctx.fillStyle = th.accent
   822→        ctx.fillRect(x, y + 7, TILE, 2)
   823→      }
   824→      // Пятнышки/шов
   825→      const h = hash2(c, r)
   826→      ctx.fillStyle = th.wallDark
   827→      ctx.globalAlpha = 0.35
   828→      if (h < 0.3) {
   829→        ctx.fillRect(x + 6 + h * 60, y + 12 + h * 40, 8, 4)
   830→      } else if (h < 0.55) {
   831→        ctx.fillRect(x + 24, y + 8 + h * 30, 4, 10)
   832→      } else if (h < 0.75) {
   833→        ctx.fillRect(x + 4, y + 22, 12, 3)
   834→      }
   835→      ctx.globalAlpha = 1
   836→      // Тонкая рамка
   837→      ctx.strokeStyle = 'rgba(0,0,0,0.18)'
   838→      ctx.lineWidth = 1
   839→      ctx.strokeRect(x + 0.5, y + 0.5, TILE - 1, TILE - 1)
   840→    }
   841→  }
