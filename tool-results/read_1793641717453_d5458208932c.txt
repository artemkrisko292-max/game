   432→   432→  private onKeyDown = (e: KeyboardEvent) => {
   433→   433→    const c = this.keyId(e)
   434→   434→    if (this.GAME_KEYS.has(c)) e.preventDefault()
   435→   435→    switch (c) {
   436→   436→      case 'ArrowLeft':
   437→   437→      case 'KeyA':
   438→   438→        this.kb.left = true
   439→   439→        break
   440→   440→      case 'ArrowRight':
   441→   441→      case 'KeyD':
   442→   442→        this.kb.right = true
   443→   443→        break
   444→   444→      case 'Space':
   445→   445→      case 'ArrowUp':
   446→   446→      case 'KeyW':
   447→   447→      case 'KeyZ':
   448→   448→        this.kb.jump = true
   449→   449→        this.sfx.ensure()
   450→   450→        break
   451→   451→      case 'Escape':
   452→   452→      case 'KeyP':
   453→   453→        if (this.phase === 'playing' || this.phase === 'paused')
   454→   454→          this.togglePause()
   455→   455→        break
   456→   456→    }
   457→   457→  }
   458→   458→
   459→   459→  private onKeyUp = (e: KeyboardEvent) => {
   460→   460→    switch (this.keyId(e)) {
   461→   461→      case 'ArrowLeft':
   462→   462→      case 'KeyA':
   463→   463→        this.kb.left = false
   464→   464→        break
   465→   465→      case 'ArrowRight':
   466→   466→      case 'KeyD':
   467→   467→        this.kb.right = false
   468→   468→        break
   469→   469→      case 'Space':
   470→   470→      case 'ArrowUp':
   471→   471→      case 'KeyW':
   472→   472→      case 'KeyZ':
   473→   473→        this.kb.jump = false
   474→   474→        break
   475→   475→    }
   476→   476→  }
   477→   477→
   478→   478→  // Нормализация: некоторые окружения присылают события без e.code —
   479→   479→  // тогда откатываемся к e.key ('ArrowLeft', 'a', ' ', 'Enter').
   480→   480→  private keyId(e: KeyboardEvent): string {
   481→   481→    if (e.code) return e.code
   482→   482→    const k = e.key
   483→   483→    if (k === ' ') return 'Space'
   484→   484→    return k.length === 1 ? `Key${k.toUpperCase()}` : k
   485→   485→  }
   486→   486→
   487→   487→  private onBlur = () => {
   488→   488→    this.kb.left = this.kb.right = this.kb.jump = false
   489→   489→    this.touch.left = this.touch.right = this.touch.jump = false
   490→   490→    if (this.phase === 'playing') this.setPhase('paused')
   491→   491→  }
   492→   492→
   493→   493→  private get input() {
   494→   494→    return {
   495→   495→      left: this.kb.left || this.touch.left,
   496→   496→      right: this.kb.right || this.touch.right,
   497→   497→      jump: this.kb.jump || this.touch.jump,
   498→   498→    }
   499→   499→  }

   501→   501→  // ---------- Главный цикл ----------
   502→   502→
   503→   503→  private loop = (t: number) => {
   504→   504→    if (this.destroyed) return
   505→   505→    this.raf = requestAnimationFrame(this.loop)
   506→   506→    let dt = (t - this.lastT) / 1000
   507→   507→    this.lastT = t
   508→   508→    if (dt > 0.1) dt = 0.1
   509→   509→    if (this.phase === 'playing') {
   510→   510→      this.acc += dt
   511→   511→      let steps = 0
   512→   512→      while (this.acc >= STEP && steps < 4) {
   513→   513→        this.step()
   514→   514→        this.acc -= STEP
   515→   515→        steps++
   516→   516→      }
   517→   517→      if (steps === 4) this.acc = 0
   518→   518→    }
   519→   519→    render(this.ctx, this)
   520→   520→  }
   521→   521→
   522→   522→  // ---------- Шаг симуляции ----------
   523→   523→
   524→   524→  private step() {
   525→   525→    const p = this.player
   526→   526→    const inp = this.input
   527→   527→    this.frame++
   528→   528→    if (this.shake > 0) this.shake *= 0.85
   529→   529→
   530→   530→    // Прыжок: буфер + койот-время
   531→   531→    const jumpPressed = inp.jump && !this.prevJump
   532→   532→    this.prevJump = inp.jump
   533→   533→    if (jumpPressed) p.buffer = BUFFER
   534→   534→    if (p.buffer > 0) p.buffer--
   535→   535→    if (p.coyote > 0) p.coyote--
   536→   536→
   537→   537→    // Горизонталь
   538→   538→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   539→   539→    if (inp.left && !inp.right) {
   540→   540→      p.vx -= acc
   541→   541→      p.facing = -1
   542→   542→    } else if (inp.right && !inp.left) {
   543→   543→      p.vx += acc
   544→   544→      p.facing = 1
   545→   545→    } else {
   546→   546→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   547→   547→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   548→   548→    }
   549→   549→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   550→   550→
   551→   551→    // Прыжок
   552→   552→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   553→   553→      p.vy = JUMP_V
   554→   554→      p.onGround = false
   555→   555→      p.coyote = 0
   556→   556→      p.buffer = 0
   557→   557→      p.squash = 1.18
   558→   558→      this.sfx.jump()
   559→   559→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   560→   560→    }
   561→   561→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   562→   562→
   563→   563→    // Гравитация
   564→   564→    p.vy = Math.min(p.vy + GRAV, 14)
   565→   565→
   566→   566→    // Движущиеся платформы
   567→   567→    for (const pl of this.plats) {
   568→   568→      pl.phase += pl.speed
   569→   569→      const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   570→   570→      pl.dx = nx - pl.x
   571→   571→      pl.x = nx
   572→   572→    }
   573→   573→
   574→   574→    // Движение и коллизии
   575→   575→    this.moveX(p, p.vx)
   576→   576→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   577→   577→
   578→   578→    const wasOnGround = p.onGround
   579→   579→    const prevFeet = p.y + p.h
   580→   580→    this.moveY(p, p.vy, prevFeet)
   581→   581→
   582→   582→    if (p.onGround) {
   583→   583→      if (!wasOnGround) {
   584→   584→        p.squash = 0.78
   585→   585→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   586→   586→        this.sfx.land()
   587→   587→      }
   588→   588→      p.coyote = COYOTE
   589→   589→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   590→   590→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   591→   591→      if (underL && underR) {
   592→   592→        p.safeX = p.x
   593→   593→        p.safeY = p.y
   594→   594→      }
   595→   595→    }
   596→   596→    p.squash += (1 - p.squash) * 0.18
   597→   597→    p.anim += Math.abs(p.vx) * 0.09
   598→   598→    if (p.invuln > 0) p.invuln--
   599→   599→    if (p.blink > 0) p.blink--
   600→   600→    else if (Math.random() < 0.012) p.blink = 8
   601→   601→
   602→   602→    // Провал за нижнюю границу мира (люки)
   603→   603→    if (p.y > this.worldH + 60) {
   604→   604→      this.damage(DMG_FALL, 0)
   605→   605→      p.x = p.safeX
   606→   606→      p.y = p.safeY
   607→   607→      p.vx = 0
   608→   608→      p.vy = 0
   609→   609→    }
   610→   610→
   611→   611→    this.updateVampires()
   612→   612→    this.updatePhantoms()
   613→   613→    this.updatePickups()
   614→   614→    this.updateSwitches()
   615→   615→    this.updateHazards()
   616→   616→    this.updateExit()

   624→   624→  solidAt(px: number, py: number): boolean {
   625→   625→    const c = Math.floor(px / TILE)
   626→   626→    const r = Math.floor(py / TILE)
   627→   627→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false

   631→   631→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   632→   632→    const c0 = Math.floor(x / TILE)
   633→   633→    const c1 = Math.floor((x + w - EPS) / TILE)
   634→   634→    const r0 = Math.floor(y / TILE)
   635→   635→    const r1 = Math.floor((y + h - EPS) / TILE)
   636→   636→    for (let r = r0; r <= r1; r++) {
   637→   637→      for (let c = c0; c <= c1; c++) {
   638→   638→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   639→   639→        if (this.grid[r * this.cols + c] === 1) return true
   640→   640→      }
   641→   641→    }
   642→   642→    return false
   643→   643→  }

   645→   645→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   646→   646→    e.x += dx
   647→   647→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   648→   648→    if (dx > 0) {
   649→   649→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   650→   650→    } else if (dx < 0) {
   651→   651→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   652→   652→    }
   653→   653→    e.vx = 0
   654→   654→    return true
   655→   655→  }

   657→   657→  private moveY(
   658→   658→    e: Box & { vy: number; onGround: boolean },
   659→   659→    dy: number,
   660→   660→    prevFeet: number,
   661→   661→  ): void {
   662→   662→    e.y += dy
   663→   663→    if (dy >= 0) {
   664→   664→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   665→   665→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   666→   666→        e.vy = 0
   667→   667→        e.onGround = true
   668→   668→      }
   669→   669→      for (const pl of this.plats) {
   670→   670→        const feet = e.y + e.h
   671→   671→        if (
   672→   672→          e.vy >= 0 &&
   673→   673→          feet >= pl.y &&
   674→   674→          prevFeet <= pl.y + 6 &&
   675→   675→          e.x + e.w > pl.x &&
   676→   676→          e.x < pl.x + pl.w
   677→   677→        ) {
   678→   678→          e.y = pl.y - e.h
   679→   679→          e.vy = 0
   680→   680→          e.onGround = true
   681→   681→          e.x += pl.dx
   682→   682→        }
   683→   683→      }
   684→   684→    } else {
   685→   685→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   686→   686→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   687→   687→        e.vy = 0
   688→   688→      }
   689→   689→    }
   690→   690→  }

   938→   938→function circleRect(
   939→   939→  cx: number,
   940→   940→  cy: number,
   941→   941→  r: number,
   942→   942→  rect: Box,
   943→   943→): boolean {
   944→   944→  const nx = Math.max(rect.x, Math.min(cx, rect.x + rect.w))
   945→   945→  const ny = Math.max(rect.y, Math.min(cy, rect.y + rect.h))
   946→   946→  const dx = cx - nx
   947→   947→  const dy = cy - ny
   948→   948→  return dx * dx + dy * dy <= r * r
   949→   949→}
