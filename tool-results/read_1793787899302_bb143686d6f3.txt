   539→  private readonly GAME_KEYS = new Set([
   540→    'ArrowLeft',
   541→    'ArrowRight',
   542→    'ArrowUp',
   543→    'ArrowDown',
   544→    'Space',
   545→    'KeyA',

   550→
   551→  private onKeyDown = (e: KeyboardEvent) => {
   552→    const c = this.keyId(e)
   553→    if (this.GAME_KEYS.has(c)) e.preventDefault()
   554→    switch (c) {
   555→      case 'ArrowLeft':
   556→      case 'KeyA':
   557→        this.kb.left = true
   558→        break
   559→      case 'ArrowRight':
   560→      case 'KeyD':
   561→        this.kb.right = true
   562→        break
   563→      case 'Space':
   564→      case 'ArrowUp':
   565→      case 'KeyW':
   566→      case 'KeyZ':
   567→        this.kb.jump = true
   568→        this.sfx.ensure()
   569→        break
   570→      case 'Escape':
   571→      case 'KeyP':
   572→        if (this.phase === 'playing' || this.phase === 'paused')
   573→          this.togglePause()
   574→        break
   575→    }
   576→  }
   577→
   578→  private onKeyUp = (e: KeyboardEvent) => {
   579→    switch (this.keyId(e)) {
   580→      case 'ArrowLeft':
   581→      case 'KeyA':
   582→        this.kb.left = false
   583→        break
   584→      case 'ArrowRight':
   585→      case 'KeyD':
   586→        this.kb.right = false
   587→        break
   588→      case 'Space':
   589→      case 'ArrowUp':
   590→      case 'KeyW':
   591→      case 'KeyZ':
   592→        this.kb.jump = false
   593→        break
   594→    }
   595→  }
   596→
   597→  // Нормализация: некоторые окружения присылают события без e.code —
   598→  // тогда откатываемся к e.key ('ArrowLeft', 'a', ' ', 'Enter').
   599→  private keyId(e: KeyboardEvent): string {
   600→    if (e.code) return e.code
   601→    const k = e.key
   602→    if (k === ' ') return 'Space'
   603→    return k.length === 1 ? `Key${k.toUpperCase()}` : k
   604→  }
   605→
   606→  private onBlur = () => {
   607→    this.kb.left = this.kb.right = this.kb.jump = false
   608→    this.touch.left = this.touch.right = this.touch.jump = false
   609→    if (this.phase === 'playing') this.setPhase('paused')
   610→  }
   611→
   612→  private get input() {
   613→    return {
   614→      left: this.kb.left || this.touch.left,
   615→      right: this.kb.right || this.touch.right,
   616→      jump: this.kb.jump || this.touch.jump,
   617→    }
   618→  }
   619→
   620→  // ---------- Главный цикл ----------
   621→
   622→  private loop = (t: number) => {
   623→    if (this.destroyed) return
   624→    this.raf = requestAnimationFrame(this.loop)
   625→    let dt = (t - this.lastT) / 1000
   626→    this.lastT = t
   627→    if (dt > 0.1) dt = 0.1
   628→    if (this.phase === 'playing') {
   629→      this.acc += dt
   630→      let steps = 0
   631→      while (this.acc >= STEP && steps < 4) {
   632→        this.step()
   633→        this.acc -= STEP
   634→        steps++
   635→      }
   636→      if (steps === 4) this.acc = 0
   637→    }
   638→    render(this.ctx, this)
   639→  }
   640→
   641→  // ---------- Шаг симуляции ----------
   642→
   643→  private step() {
   644→    const p = this.player
   645→    const inp = this.input
   646→    this.frame++
   647→    this.levelFrames++
   648→    if (this.shake > 0) this.shake *= 0.85
   649→
   650→    // Прыжок: буфер + койот-время
   651→    const jumpPressed = inp.jump && !this.prevJump
   652→    this.prevJump = inp.jump
   653→    if (jumpPressed) p.buffer = BUFFER
   654→    if (p.buffer > 0) p.buffer--
   655→    if (p.coyote > 0) p.coyote--
   656→
   657→    // Горизонталь
   658→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   659→    if (inp.left && !inp.right) {
   660→      p.vx -= acc
   661→      p.facing = -1
   662→    } else if (inp.right && !inp.left) {
   663→      p.vx += acc
   664→      p.facing = 1
   665→    } else {
   666→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   667→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   668→    }
   669→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   670→
   671→    // Прыжок
   672→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   673→      p.vy = JUMP_V
   674→      p.onGround = false
   675→      p.coyote = 0
   676→      p.buffer = 0
   677→      p.squash = 1.18
   678→      this.sfx.jump()
   679→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   680→    }
   681→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   682→
   683→    // Гравитация
   684→    p.vy = Math.min(p.vy + GRAV, 14)
   685→
   686→    // Движущиеся платформы (горизонтальные и вертикальные)
   687→    for (const pl of this.plats) {
   688→      pl.phase += pl.speed
   689→      if (pl.axis === 'x') {
   690→        const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   691→        pl.dx = nx - pl.x
   692→        pl.dy = 0
   693→        pl.x = nx
   694→      } else {
   695→        const ny = pl.by + Math.sin(pl.phase) * pl.amp
   696→        pl.dy = ny - pl.y
   697→        pl.dx = 0
   698→        pl.y = ny
   699→      }
   700→    }
   701→
   702→    // Движение и коллизии
   703→    this.moveX(p, p.vx)
   704→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   705→
   706→    const wasOnGround = p.onGround
   707→    const prevFeet = p.y + p.h
   708→    this.moveY(p, p.vy, prevFeet)
   709→
   710→    if (p.onGround) {
   711→      if (!wasOnGround) {
   712→        p.squash = 0.78
   713→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   714→        this.sfx.land()
   715→      }
   716→      p.coyote = COYOTE
   717→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   718→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   719→      if (underL && underR) {
   720→        p.safeX = p.x
   721→        p.safeY = p.y
   722→      }
   723→    }
   724→    p.squash += (1 - p.squash) * 0.18
   725→    p.anim += Math.abs(p.vx) * 0.09
   726→    if (p.invuln > 0) p.invuln--
   727→    if (p.blink > 0) p.blink--
   728→    else if (Math.random() < 0.012) p.blink = 8
   729→
   730→    // Провал за нижнюю границу мира (люки)
   731→    if (p.y > this.worldH + 60) {
   732→      this.damage(DMG_FALL, 0, 'fall')
   733→      p.x = p.safeX
   734→      p.y = p.safeY
   735→      p.vx = 0
   736→      p.vy = 0
   737→    }
   738→
   739→    this.updateWalkers(this.vampires, DMG_VAMPIRE, 100, 'vampire')
   740→    this.updateWalkers(this.runners, DMG_RUNNER, SCORE_RUNNER, 'runner')
   741→    this.updateDrones()
   742→    this.updatePhantoms()
   743→    this.updateCrumbs()
   744→    this.updateFuses()
   745→    this.updatePickups()
   746→    this.updateSwitches()
   747→    this.updateHazards()
   748→    this.updateExit()
   749→    this.updateEnergy()
   750→    this.updateParticles()
   751→    this.updateCamera()
   752→  }

   754→  // ---------- Коллизии с тайлами ----------
   755→
   756→  solidAt(px: number, py: number): boolean {
   757→    const c = Math.floor(px / TILE)
   758→    const r = Math.floor(py / TILE)
   759→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   760→    return this.grid[r * this.cols + c] === 1
   761→  }
   762→
   763→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   764→    const c0 = Math.floor(x / TILE)
   765→    const c1 = Math.floor((x + w - EPS) / TILE)
   766→    const r0 = Math.floor(y / TILE)
   767→    const r1 = Math.floor((y + h - EPS) / TILE)
   768→    for (let r = r0; r <= r1; r++) {
   769→      for (let c = c0; c <= c1; c++) {
   770→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   771→        if (this.grid[r * this.cols + c] === 1) return true
   772→      }
   773→    }
   774→    return false
   775→  }
   776→
   777→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   778→    e.x += dx
   779→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   780→    if (dx > 0) {
   781→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   782→    } else if (dx < 0) {
   783→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   784→    }
   785→    e.vx = 0
   786→    return true
   787→  }
   788→
   789→  private moveY(
   790→    e: Box & { vy: number; onGround: boolean },
   791→    dy: number,
   792→    prevFeet: number,
   793→  ): void {
   794→    e.y += dy
   795→    if (dy >= 0) {
   796→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   797→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   798→        e.vy = 0
   799→        e.onGround = true
   800→      }
   801→      for (const pl of this.plats) {
   802→        const feet = e.y + e.h
   803→        if (
   804→          e.vy >= 0 &&
   805→          feet >= pl.y &&
   806→          prevFeet <= pl.y + 6 &&
   807→          e.x + e.w > pl.x &&
   808→          e.x < pl.x + pl.w
   809→        ) {
   810→          e.y = pl.y - e.h
   811→          e.vy = 0
   812→          e.onGround = true
   813→          e.x += pl.dx
   814→        }
   815→      }
   816→      // Осыпающиеся плиты: та же «односторонняя» коллизия сверху
   817→      for (const c of this.crumbs) {
   818→        if (c.state === 2) continue
   819→        const feet = e.y + e.h
   820→        if (
   821→          feet >= c.y &&
   822→          prevFeet <= c.y + 6 &&
   823→          e.x + e.w > c.x + 2 &&
   824→          e.x < c.x + c.w - 2
   825→        ) {
   826→          e.y = c.y - e.h
   827→          e.vy = 0
   828→          e.onGround = true
   829→        }
   830→      }
   831→    } else {
   832→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   833→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   834→        e.vy = 0
   835→      }
   836→    }
   837→  }

   841→  /** Наземные ходячие враги (вампиры и гонцы). */
   842→  private updateWalkers(
   843→    list: Vampire[],
   844→    dmg: number,
   845→    stompScore: number,
   846→    cause: DeathCause,
   847→  ) {
   848→    const p = this.player
   849→    for (const v of list) {
   850→      if (v.dead) continue
   851→      v.anim += Math.abs(v.vx) * 0.06
   852→      v.vy = Math.min(v.vy + GRAV, 12)
   853→      const oldVx = v.vx
   854→      this.moveX(v, v.vx)
   855→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   856→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   857→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   858→
   859→      const prevFeet = v.y + v.h
   860→      const wasGround = v.onGround
   861→      v.onGround = false
   862→      this.moveY(v, v.vy, prevFeet)
   863→      if (v.onGround && !wasGround) v.vy = 0
   864→      // разворот у обрыва
   865→      if (v.onGround) {
   866→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   867→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   868→      }
   869→
   870→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   871→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   872→          v.dead = true
   873→          p.vy = -9.5
   874→          this.score += stompScore
   875→          this.shake = 6
   876→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   877→          this.sfx.stomp()
   878→        } else {
   879→          this.damage(dmg, p.x < v.x + v.w / 2 ? -1 : 1, cause)
   880→        }
   881→      }
   882→    }
   883→  }

   886→  private updateDrones() {
   887→    const p = this.player
   888→    for (const d of this.drones) {
   889→      if (d.dead) continue
   890→      d.anim += 0.12
   891→      const oldVx = d.vx
   892→      this.moveX(d, d.vx)
   893→      if (d.vx === 0 && oldVx !== 0) d.vx = -oldVx
   894→
   895→      const oldVy = d.vy
   896→      this.moveY(d, d.vy, d.y + d.h)
   897→      if (d.vy === 0 && oldVy !== 0) d.vy = -oldVy // пол или потолок
   898→
   899→      // Границы мира: не даём дрону улететь в пустоту
   900→      if (d.x <= 0) {
   901→        d.x = 0
   902→        d.vx = Math.abs(d.vx)
   903→      } else if (d.x + d.w >= this.worldW) {
   904→        d.x = this.worldW - d.w
   905→        d.vx = -Math.abs(d.vx)
   906→      }
   907→      if (d.y <= 0) {
   908→        d.y = 0
   909→        d.vy = Math.abs(d.vy)
   910→      } else if (d.y + d.h >= this.worldH) {
   911→        d.y = this.worldH - d.h
   912→        d.vy = -Math.abs(d.vy)
   913→      }
   914→
   915→      if (p.invuln <= 0 && rectsOverlap(p, d)) {
   916→        if (p.vy > 2 && p.y + p.h - d.y < 20) {
   917→          d.dead = true
   918→          p.vy = -9.5
   919→          this.score += SCORE_DRONE
   920→          this.shake = 6
   921→          this.spawnParticles(d.x + d.w / 2, d.y + d.h / 2, 12, 'poof')
   922→          this.sfx.stomp()
   923→        } else {
   924→          this.damage(DMG_DRONE, p.x < d.x + d.w / 2 ? -1 : 1, 'drone')
   925→        }
   926→      }
   927→    }

  1054→  private lampDrain(): number {
  1055→    let drain = 0
  1056→    const p = this.player

  1058→    const pcy = p.y + p.h / 2
  1059→    for (const l of this.lamps) {
  1060→      if (!l.on) continue
  1061→      const dx = Math.abs(pcx - l.x)
  1062→      const dy = pcy - l.y
  1063→      if (dy > 0 && dy < LAMP_CONE_LEN) {
  1064→        const halfW = 16 + dy * 0.34
  1065→        if (dx < halfW) {
  1066→          drain += DRAIN_LAMP * (1 - (dy / LAMP_CONE_LEN) * 0.45)
  1067→        }
  1068→      }
  1069→    }
  1070→    return Math.min(drain, DRAIN_LAMP * 2)

  1133→      (DRAIN_BASE * mul + this.lampDrain() * (0.8 + mul * 0.2)) * STEP

  1135→      this.energy = 0

  1137→      this.sfx.lose()

  1142→  private damage(amount: number, knockDir: number, cause: DeathCause) {
  1143→    const p = this.player
  1144→    if (p.invuln > 0) return
  1145→    this.energy = Math.max(0, this.energy - amount)
  1146→    p.invuln = 75
  1147→    p.vx = knockDir * 5
