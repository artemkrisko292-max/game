    56→
    57→export default function EnergoHeroGame() {
    58→  const canvasRef = useRef<HTMLCanvasElement>(null)
    59→  const gameRef = useRef<EnergoGame | null>(null)
    60→  const [phase, setPhase] = useState<Phase>('menu')
    61→  const [snap, setSnap] = useState<Snapshot | null>(null)
    62→  const [muted, setMuted] = useState(false)
    63→  const [isTouch, setIsTouch] = useState(false)
    64→  const [best, setBest] = useState(0)
    65→
    66→  useEffect(() => {
    67→    const canvas = canvasRef.current
    68→    if (!canvas) return
    69→    const dpr = Math.min(2, window.devicePixelRatio || 1)
    70→    canvas.width = VIEW_W * dpr
    71→    canvas.height = VIEW_H * dpr
    72→    const ctx = canvas.getContext('2d')
    73→    if (!ctx) return
    74→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
    75→
    76→    const game = new EnergoGame(canvas, {
    77→      onPhase: (s) => {
    78→        setPhase(s.phase)
    79→        setSnap(s)
    80→        if (s.phase === 'victory') {
    81→          try {
    82→            setBest(Number(localStorage.getItem('energohero-best') ?? '0') || 0)
    83→          } catch {
    84→            /* ignore */
    85→          }
    86→        }
    87→      },
    88→    })
    89→    gameRef.current = game
    90→    // Отладочный хук (используется автотестами: __energo.loadLevel(n))

    96→        window.matchMedia('(pointer: coarse)').matches ||
    97→          'ontouchstart' in window ||
    98→          window.innerWidth < 768,
    99→      )
   100→      try {
   101→        setBest(Number(localStorage.getItem('energohero-best') ?? '0') || 0)
   102→      } catch {
   103→        /* ignore */
   104→      }
   105→    })
   106→    return () => {
   107→      cancelAnimationFrame(raf)
   108→      game.destroy()
   109→      gameRef.current = null
   110→    }
   111→  }, [])
   112→

   114→    const g = gameRef.current
   115→    if (!g) return

   117→      case 'start':
   118→        g.startRun()
   119→        break
   120→      case 'begin':
   121→        g.beginLevel()
   122→        break
   123→      case 'retry':
   124→        g.retryLevel()
   125→        break
   126→      case 'next':
   127→        g.nextLevel()
   128→        break
   129→      case 'menu':
   130→        g.toMenu()
   131→        break
   132→      case 'pause':
   133→        g.togglePause()
   134→        break
   135→    }
   136→  }, [])
   137→
   138→  // Enter — основное действие на оверлеях
   139→  useEffect(() => {
   140→    const onKey = (e: KeyboardEvent) => {
   141→      if (e.code !== 'Enter' && e.code !== 'NumpadEnter' && e.key !== 'Enter')
   142→        return

   144→      else if (phase === 'intro') act('begin')
   145→      else if (phase === 'complete') act('next')

   161→      onPointerLeave: () => gameRef.current?.setTouch(k, false),
   162→      onPointerCancel: () => gameRef.current?.setTouch(k, false),
   163→    }),
   164→    [],
   165→  )
   166→

   168→    const g = gameRef.current
   169→    if (!g) return
   170→    g.sfx.muted = !g.sfx.muted
   171→    setMuted(g.sfx.muted)
   172→  }, [])
   173→

   176→      <header className="flex items-center gap-2 select-none sm:hidden">
   177→        <Zap className="h-5 w-5 text-amber-300" />
   178→        <h1 className="text-lg font-black tracking-widest text-emerald-300">
   179→          ЭНЕРГОГЕРОЙ
   180→        </h1>
   181→      </header>
   182→
   183→      <main className="w-full max-w-[1080px]">
   184→        <div className="relative aspect-[16/9] max-md:aspect-[4/5] w-full overflow-hidden rounded-2xl border-2 border-emerald-950 bg-[#0b120c] shadow-[0_0_60px_rgba(52,168,83,0.15)] sm:border-4">
   185→          <canvas
   186→            ref={canvasRef}
   187→            className="absolute inset-0 h-full w-full object-contain"
   188→            aria-label="Игровое поле ЭнергоГероя"
   189→          />
   190→
   191→          {/* Кнопки звука и паузы */}
   192→          <div className="absolute right-2 top-2 z-20 flex gap-1.5">
   193→            <button
   194→              onClick={toggleMute}
   195→              aria-label={muted ? 'Включить звук' : 'Выключить звук'}
   196→              className="flex h-9 w-9 items-center justify-center rounded-full bg-black/45 text-white/90 backdrop-blur-sm transition hover:bg-black/65"
   197→            >
   198→              {muted ? (
   199→                <VolumeX className="h-4 w-4" />
   200→              ) : (
   201→                <Volume2 className="h-4 w-4" />
   202→              )}
   203→            </button>
   204→            {(phase === 'playing' || phase === 'paused') && (
   205→              <button
   206→                onClick={() => act('pause')}
   207→                aria-label="Пауза"

   261→                      {best}
   262→                    </p>
   263→                  )}
   264→                </div>
   265→              </Panel>
   266→            </Overlay>
   267→          )}
   268→
   269→          {phase === 'intro' && snap && (
   270→            <Overlay>
   271→              <Panel>
   272→                {snap.badge && (
   273→                  <div className="mb-3 flex justify-center">
   274→                    <span className="ah-stamp inline-block -rotate-2 rounded-md border-[3px] border-amber-400 bg-amber-400/10 px-3 py-1 text-xs font-black uppercase tracking-[0.25em] text-amber-300">
   275→                      {snap.badge}

   278→                )}
   279→                <p className="text-xs font-black uppercase tracking-[0.35em] text-emerald-500">
   280→                  {snap.levelSubtitle}
   281→                </p>
   282→                <h2 className="mt-1 text-4xl font-black text-white sm:text-5xl">
   283→                  {snap.levelName}
   284→                </h2>
   285→                <div className="mx-auto mt-4 max-w-xl rounded-lg border-2 border-amber-500/40 bg-amber-500/10 p-4 text-left">

   287→                    <Lightbulb className="mt-0.5 h-5 w-5 shrink-0 text-amber-300" />
   288→                    <span>{snap.hint}</span>
   289→                  </p>
   290→                </div>
   291→                <div className="mt-5">
   292→                  <ArcadeButton onClick={() => act('begin')} size="lg">
   293→                    В БОЙ! <ChevronRight className="h-5 w-5" />
   294→                  </ArcadeButton>
   295→                </div>
   296→              </Panel>
   297→            </Overlay>
   298→          )}
   299→
   300→          {phase === 'paused' && (
   301→            <Overlay>
   302→              <Panel>
   303→                <h2 className="text-3xl font-black tracking-[0.3em] text-white">
   304→                  ПАУЗА
   305→                </h2>
   306→                <p className="mt-2 text-sm text-emerald-200/70">
   307→                  Свет подождёт. Но недолго
   308→                  <span className="ah-blink ml-1 inline-block h-4 w-2 translate-y-0.5 bg-emerald-300" />
   309→                </p>
   310→                <div className="mt-6 flex flex-wrap justify-center gap-3">
   311→                  <ArcadeButton onClick={() => act('pause')}>
   312→                    <Play className="h-4 w-4" /> Продолжить
   313→                  </ArcadeButton>
   314→                  <ArcadeButton tone="ghost" onClick={() => act('menu')}>
   315→                    <Home className="h-4 w-4" /> В меню
   316→                  </ArcadeButton>
   317→                </div>
   318→              </Panel>
   319→            </Overlay>
   320→          )}
   321→
   322→          {phase === 'complete' && snap?.result && (
   323→            <Overlay>
   324→              <Panel>
   325→                <div className="flex justify-center">
   326→                  <span className="ah-stamp inline-block -rotate-2 rounded-md border-[3px] border-emerald-400 px-4 py-1 text-base font-black uppercase tracking-[0.25em] text-emerald-300 sm:text-lg">
   327→                    Уровень зачищен
   328→                  </span>
   329→                </div>
   330→                <h2 className="mt-3 text-3xl font-black text-white sm:text-4xl">
   331→                  {snap.result.levelName} спасена!
   332→                </h2>
   333→
   334→                {/* Прогресс забега: 20 точек */}
   335→                <LevelDots current={snap.levelIndex} />

   337→                <div className="mx-auto mt-4 grid max-w-md grid-cols-2 gap-2 sm:gap-3">

   371→                    delay={300}
   372→                  />
   373→                  {snap.result.fusesTotal > 0 && (
   374→                    <div className="col-span-2">
   375→                      <Tally
   376→                        icon={Plug}
   377→                        label="Предохранители"

   379→                        pct={
   380→                          snap.result.fusesTotal > 0
   381→                            ? snap.result.fuses / snap.result.fusesTotal
   382→                            : 1
   383→                        }
   384→                        delay={340}
   385→                      />
   386→                    </div>
   387→                  )}
   388→                </div>
   389→                <p className="mt-4 text-2xl font-black text-amber-300">
   390→                  <ScoreCount value={snap.result.score} />{' '}
   391→                  <span className="text-sm tracking-widest text-amber-200/60">
   392→                    ОЧКОВ
   393→                  </span>
   394→                </p>
   395→                <div className="mt-4 flex flex-wrap justify-center gap-3">
   396→                  <ArcadeButton onClick={() => act('next')} size="lg">
   397→                    ДАЛЬШЕ <ChevronRight className="h-5 w-5" />
   398→                  </ArcadeButton>
   399→                  <ArcadeButton tone="ghost" onClick={() => act('menu')}>

   656→/** Плавный счётчик очков. */
   657→function ScoreCount({ value }: { value: number }) {
   658→  const [v, setV] = useState(0)
   659→  useEffect(() => {
   660→    let raf = 0
   661→    const t0 = performance.now()
   662→    const dur = 700
   663→    const tick = (t: number) => {
   664→      const k = Math.min(1, (t - t0) / dur)
   665→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
   666→      if (k < 1) raf = requestAnimationFrame(tick)
   667→    }
   668→    raf = requestAnimationFrame(tick)
   669→    return () => cancelAnimationFrame(raf)
   670→  }, [value])
   671→  return <>{v}</>
   672→}
