    47→interface Progress {
    48→  unlocked: number // сколько уровней открыто (1…20)
    49→  stars: number[]
    50→  best: number[]
    51→}
    52→
    53→const EMPTY_PROGRESS: Progress = {
    54→  unlocked: 1,
    55→  stars: Array(LEVELS.length).fill(0),
    56→  best: Array(LEVELS.length).fill(0),
    57→}
    58→
    59→function loadProgress(): Progress {
    60→  try {
    61→    const raw = localStorage.getItem(PROGRESS_KEY)
    62→    if (!raw) return EMPTY_PROGRESS
    63→    const p = JSON.parse(raw) as Partial<Progress>
    64→    const stars = Array(LEVELS.length).fill(0)
    65→    const best = Array(LEVELS.length).fill(0)
    66→    ;(p.stars ?? []).forEach((v, i) => (stars[i] = Number(v) || 0))
    67→    ;(p.best ?? []).forEach((v, i) => (best[i] = Number(v) || 0))
    68→    return {
    69→      unlocked: Math.min(LEVELS.length, Math.max(1, Number(p.unlocked) || 1)),
    70→      stars,
    71→      best,
    72→    }
    73→  } catch {
    74→    return EMPTY_PROGRESS
    75→  }
    76→}
    77→
    78→function starsFromResult(r: LevelResult): number {
    79→  const got = r.bulbs + r.switches + r.fuses
    80→  const tot = r.bulbsTotal + r.switchesTotal + r.fusesTotal
    81→  const pct = tot > 0 ? got / tot : 1
    82→  return pct >= 0.9 ? 3 : pct >= 0.55 ? 2 : 1
    83→}
    84→
    85→function saveLevelProgress(
    86→  levelIndex: number,
    87→  result: LevelResult | null,
    88→  summary?: RunSummary | null,
    89→) {
    90→  try {
    91→    const p = loadProgress()
    92→    const stars = result ? starsFromResult(result) : (summary?.stars ?? 1)
    93→    const score = result ? result.score : (summary?.score ?? 0)
    94→    p.stars[levelIndex] = Math.max(p.stars[levelIndex] ?? 0, stars)
    95→    p.best[levelIndex] = Math.max(p.best[levelIndex] ?? 0, score)
    96→    p.unlocked = Math.min(
    97→      LEVELS.length,
    98→      Math.max(p.unlocked, levelIndex + 2),
    99→    )
   100→    localStorage.setItem(PROGRESS_KEY, JSON.stringify(p))
   101→  } catch {
   102→    /* localStorage недоступен */
   103→  }
   104→}

   218→export default function EnergoHeroGame() {
   219→  const canvasRef = useRef<HTMLCanvasElement>(null)
   220→  const gameRef = useRef<EnergoGame | null>(null)
   221→  const musicRef = useRef<Music | null>(null)
   222→  const [phase, setPhase] = useState<Phase>('menu')
   223→  const [snap, setSnap] = useState<Snapshot | null>(null)
   224→  const [musicOn, setMusicOn] = useState(true)
   225→  const [sfxOn, setSfxOn] = useState(true)
   226→  // Зеркала состояния для обработчиков — клики всегда применяют
   227→  // актуальное значение, даже если React-состояние ещё не подъехало.
   228→  const musicOnRef = useRef(true)
   229→  const sfxOnRef = useRef(true)
   230→  const [isTouch, setIsTouch] = useState(false)
   231→  const [best, setBest] = useState(0)
   232→  const [progress, setProgress] = useState<Progress>(EMPTY_PROGRESS)
   233→  const [menuView, setMenuView] = useState<
   234→    'main' | 'levels' | 'howto' | 'story'
   235→  >('main')
   236→  // ---------- Сюжет: катсцены ----------
   237→  const [cutscene, setCutscene] = useState<(typeof STORY)[string] | null>(null)
   238→  const [seenScenes, setSeenScenes] = useState<Set<string>>(new Set())
   239→  const seenRef = useRef<Set<string>>(new Set())
   240→  const afterSceneRef = useRef<(() => void) | null>(null)
   241→
   242→  /** Показать главу хроники; повторно — только по запросу (force). */
   243→  const playScene = useCallback(
   244→    (id: string, after: () => void, force = false) => {
   245→      if (!STORY[id]) {
   246→        after()
   247→        return
   248→      }
   249→      if (!force && seenRef.current.has(id)) {
   250→        after()
   251→        return
   252→      }
   253→      seenRef.current.add(id)
   254→      saveSeenScenes(seenRef.current)
   255→      setSeenScenes(new Set(seenRef.current))
   256→      afterSceneRef.current = after
   257→      setCutscene(STORY[id])
   258→    },
   259→    [],
   260→  )
   261→
   262→  const finishCutscene = useCallback(() => {
   263→    const after = afterSceneRef.current
   264→    afterSceneRef.current = null
   265→    setCutscene(null)
   266→    after?.()
   267→  }, [])
   268→
   269→  useEffect(() => {
   270→    const canvas = canvasRef.current
   271→    if (!canvas) return
   272→    const dpr = Math.min(2, window.devicePixelRatio || 1)
   273→    canvas.width = VIEW_W * dpr
   274→    canvas.height = VIEW_H * dpr
   275→    const ctx = canvas.getContext('2d')
   276→    if (!ctx) return
   277→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   278→
   279→    const music = new Music()
   280→    musicRef.current = music
   281→
   282→    // Настройки звука читаем синхронно — иконки и реальное состояние
   283→    // совпадают с первого кадра, без окна рассинхрона.
   284→    let mOn = true
   285→    let sOn = true
   286→    try {
   287→      mOn = localStorage.getItem(PREF_MUSIC) !== 'off'
   288→      sOn = localStorage.getItem(PREF_SFX) !== 'off'
   289→    } catch {
   290→      /* ignore */
   291→    }
   292→    music.enabled = mOn
   293→    // Рефы синхронно: логика toggles верна с первого же клика.
   294→    // Состояние иконок подтянем в rAF-коллбэке (см. ниже).
   295→    musicOnRef.current = mOn
   296→    sfxOnRef.current = sOn
   297→
   298→    const game = new EnergoGame(canvas, {
   299→      onPhase: (s) => {
   300→        setPhase(s.phase)
   301→        setSnap(s)
   302→        if (s.phase === 'menu' || s.phase === 'intro') {
   303→          if (music.enabled) music.start()
   304→          music.setDuck(0.55)
   305→        } else if (s.phase === 'playing') {
   306→          music.setDuck(1)
   307→        } else if (s.phase === 'paused' || s.phase === 'complete') {
   308→          music.setDuck(0.3)
   309→        } else if (s.phase === 'gameover') {
   310→          music.stop()
   311→        } else if (s.phase === 'victory') {
   312→          // Финал: тема возвращается победным маршем под салют
   313→          if (music.enabled) music.start()
   314→          music.setDuck(0.5)
   315→        }
   316→        if (s.phase === 'complete' && s.result) {
   317→          saveLevelProgress(s.levelIndex, s.result)
   318→          setProgress(loadProgress())
   319→        }
   320→        if (s.phase === 'victory' && s.summary) {
   321→          saveLevelProgress(s.levelIndex, null, s.summary)
   322→          setProgress(loadProgress())
   323→          try {
   324→            setBest(
   325→              Number(localStorage.getItem('energohero-best') ?? '0') || 0,
   326→            )
   327→          } catch {
   328→            /* ignore */
   329→          }
   330→          // Финал: эпилог хроники поверх отчёта — всегда, это конец игры
   331→          playScene('epilogue', () => {}, true)
   332→        }
   333→      },
   334→    })
   335→    game.sfx.muted = !sOn
   336→    gameRef.current = game
   337→    // Отладочные хуки (используются автотестами)
   338→    ;(window as unknown as { __energo?: EnergoGame }).__energo = game
   339→    ;(window as unknown as { __energoMusic?: Music }).__energoMusic = music
   340→
   341→    // Отложенная инициализация некритичного UI-состояния
   342→    const raf = requestAnimationFrame(() => {
   343→      setIsTouch(
   344→        window.matchMedia('(pointer: coarse)').matches ||
   345→          'ontouchstart' in window ||
   346→          window.innerWidth < 768,
   347→      )
   348→      setProgress(loadProgress())
   349→      setMusicOn(musicOnRef.current)
   350→      setSfxOn(sfxOnRef.current)
   351→      const seen = loadSeenScenes()
   352→      seenRef.current = seen
   353→      setSeenScenes(new Set(seen))
   354→      try {
   355→        setBest(Number(localStorage.getItem('energohero-best') ?? '0') || 0)
   356→      } catch {
   357→        /* ignore */
   358→      }
   359→    })
   360→
   361→    // Музыка стартует после первого жеста пользователя (политика автоплея).
   362→    // wake() продублирован внутри Music — там же самолечение suspended.
   363→    const kick = () => {
   364→      music.wake()
   365→      if (music.enabled) music.start()
   366→    }
   367→    window.addEventListener('pointerdown', kick, { once: true })
   368→    window.addEventListener('keydown', kick, { once: true })
   369→
   370→    return () => {
   371→      cancelAnimationFrame(raf)
   372→      window.removeEventListener('pointerdown', kick)
   373→      window.removeEventListener('keydown', kick)
   374→      music.destroy()
   375→      game.destroy()
   376→      gameRef.current = null
   377→      musicRef.current = null
   378→    }
   379→  }, [playScene])
   380→
   381→  const act = useCallback(
   382→    (
   383→      action:
   384→        | 'start'
   385→        | 'begin'
   386→        | 'retry'
   387→        | 'next'
   388→        | 'menu'
   389→        | 'pause'
   390→        | { select: number },
   391→    ) => {
   392→      const g = gameRef.current
   393→      if (!g) return
   394→      if (typeof action === 'object') {
   395→        g.selectLevel(action.select)
   396→        return
   397→      }
   398→      switch (action) {
   399→        case 'start':
   400→          g.startRun()
   401→          break
   402→        case 'begin':
   403→          g.beginLevel()
   404→          break
   405→        case 'retry':
   406→          g.retryLevel()
   407→          break
   408→        case 'next':
   409→          g.nextLevel()
   410→          break
   411→        case 'menu':
   412→          g.toMenu()
   413→          break
   414→        case 'pause':
   415→          g.togglePause()
   416→          break
   417→      }
   418→    },
   419→    [],
   420→  )
   421→
   422→  /** «Следующий наряд» с проверкой: не начинается ли новая глава. */
   423→  const nextWithStory = useCallback(() => {
   424→    const g = gameRef.current
   425→    if (!g) return
   426→    const nextIdx = (snap?.levelIndex ?? 0) + 1
   427→    const chapter = nextIdx >= LEVELS.length ? 'epilogue' : CHAPTER_AT[nextIdx]
   428→    if (chapter) playScene(chapter, () => act('next'))
   429→    else act('next')
   430→  }, [snap, act, playScene])
   431→
   432→  /** Загрузка уровня с карты с проверкой: не открывает ли он главу. */
   433→  const pickWithStory = useCallback(
   434→    (i: number) => {
   435→      const g = gameRef.current
   436→      if (!g) return
   437→      const chapter = CHAPTER_AT[i]
   438→      if (chapter) playScene(chapter, () => act({ select: i }))
   439→      else act({ select: i })
   440→    },
   441→    [act, playScene],
   442→  )
   443→
   444→  /** Новая смена: всегда с пролога (кнопка пропуска в кадре есть). */
   445→  const startFresh = useCallback(() => {
   446→    playScene('prologue', () => act('start'), true)
   447→  }, [playScene, act])
   448→
   449→  // Enter — основное действие на оверлеях (когда не открыта катсцена)
   450→  useEffect(() => {
   451→    const onKey = (e: KeyboardEvent) => {
   452→      if (e.code !== 'Enter' && e.code !== 'NumpadEnter' && e.key !== 'Enter')
   453→        return
   454→      if (cutscene) return // катсцена сама слушает Enter
   455→      if (phase === 'menu') startFresh()
   456→      else if (phase === 'intro') act('begin')
   457→      else if (phase === 'complete') nextWithStory()
   458→      else if (phase === 'gameover') act('retry')
   459→      else if (phase === 'victory') startFresh()
   460→      else if (phase === 'paused') act('pause')
   461→    }
   462→    window.addEventListener('keydown', onKey)
   463→    return () => window.removeEventListener('keydown', onKey)
   464→  }, [phase, act, cutscene, startFresh, nextWithStory])
   465→
   466→  const hold = useCallback(
   467→    (k: 'left' | 'right' | 'jump') => ({
   468→      onPointerDown: (e: React.PointerEvent) => {
   469→        e.preventDefault()
   470→        gameRef.current?.setTouch(k, true)
   471→      },
   472→      onPointerUp: () => gameRef.current?.setTouch(k, false),
   473→      onPointerLeave: () => gameRef.current?.setTouch(k, false),
   474→      onPointerCancel: () => gameRef.current?.setTouch(k, false),
   475→    }),
   476→    [],
   477→  )
   478→
   479→  /** Музыка: применяем значение сразу, побочные эффекты — вне setState. */
   480→  const toggleMusic = useCallback(() => {
   481→    const nv = !musicOnRef.current
   482→    musicOnRef.current = nv
   483→    setMusicOn(nv)
   484→    musicRef.current?.setEnabled(nv)
   485→    try {
   486→      localStorage.setItem(PREF_MUSIC, nv ? 'on' : 'off')
   487→    } catch {
   488→      /* ignore */
   489→    }
   490→  }, [])
   491→
   492→  /** Эффекты: глушим мгновенно, без хвостов. */
   493→  const toggleSfx = useCallback(() => {
   494→    const nv = !sfxOnRef.current
   495→    sfxOnRef.current = nv
   496→    setSfxOn(nv)
   497→    if (gameRef.current) gameRef.current.sfx.muted = !nv
   498→    try {
   499→      localStorage.setItem(PREF_SFX, nv ? 'on' : 'off')
   500→    } catch {
   501→      /* ignore */
   502→    }
   503→  }, [])
   504→
   505→  const playing = phase === 'playing' || phase === 'paused'
   506→
   507→  return (
   508→    <div className="ah-grid-bg flex min-h-screen w-full flex-col items-center justify-center gap-3 p-2 text-[#efe7d2] sm:p-4">
   509→      <main className="w-full max-w-[1080px]">
   510→        <div className="relative aspect-[16/9] max-md:aspect-[4/5] w-full overflow-hidden rounded-md border-2 border-[#2a2f28] bg-[#0b0d0c] shadow-[0_0_0_6px_rgba(255,255,255,0.03),0_24px_70px_rgba(0,0,0,0.6)]">
   511→          <canvas

  1431→function ScoreCount({ value }: { value: number }) {
  1432→  const [v, setV] = useState(0)
  1433→  useEffect(() => {
  1434→    let raf = 0
  1435→    const t0 = performance.now()
  1436→    const dur = 700
  1437→    const tick = (t: number) => {
  1438→      const k = Math.min(1, (t - t0) / dur)
  1439→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
  1440→      if (k < 1) raf = requestAnimationFrame(tick)
  1441→    }
  1442→    raf = requestAnimationFrame(tick)
  1443→    return () => cancelAnimationFrame(raf)
  1444→  }, [value])
  1445→  return <>{v}</>
  1446→}
