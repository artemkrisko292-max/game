   250→   250→  private finishRun() {
   251→   251→    const total = this.runBulbsTotal + this.runSwitchesTotal
   252→   252→    const collectPct = total > 0 ? (this.runBulbs + this.runSwitches) / total : 1
   253→   253→    const stars = collectPct >= 0.9 ? 3 : collectPct >= 0.55 ? 2 : 1
   254→   254→    let best = 0
   255→   255→    try {
   256→   256→      best = Number(localStorage.getItem('energohero-best') ?? '0') || 0
   257→   257→      if (this.score > best) {
   258→   258→        best = this.score
   259→   259→        localStorage.setItem('energohero-best', String(best))
   260→   260→      }
   261→   261→    } catch {
   262→   262→      /* localStorage недоступен */
   263→   263→    }
   264→   264→    this.summary = {
   265→   265→      score: this.score,
   266→   266→      best,
   267→   267→      bulbs: this.runBulbs,
   268→   268→      bulbsTotal: this.runBulbsTotal,
   269→   269→      switches: this.runSwitches,
   270→   270→      switchesTotal: this.runSwitchesTotal,
   271→   271→      stars,
   272→   272→    }
   273→   273→    this.setPhase('victory')
   274→   274→  }

   604→   604→  private loop = (t: number) => {
   605→   605→    if (this.destroyed) return
   606→   606→    this.raf = requestAnimationFrame(this.loop)
   607→   607→    let dt = (t - this.lastT) / 1000
   608→   608→    this.lastT = t
   609→   609→    if (dt > 0.1) dt = 0.1
   610→   610→    if (this.phase === 'playing') {
   611→   611→      this.acc += dt
   612→   612→      let steps = 0
   613→   613→      while (this.acc >= STEP && steps < 4) {
   614→   614→        this.step()
   615→   615→        this.acc -= STEP
   616→   616→        steps++
   617→   617→      }
   618→   618→      if (steps === 4) this.acc = 0
   619→   619→    }
   620→   620→    render(this.ctx, this)
   621→   621→  }

   623→   623→  // ---------- Шаг симуляции ----------

   625→   625→  private step() {
   626→   626→    const p = this.player
   627→   627→    const inp = this.input
   628→   628→    this.frame++
   629→   629→    if (this.shake > 0) this.shake *= 0.85
   630→   630→
   631→   631→    // Прыжок: буфер + койот-время
   632→   632→    const jumpPressed = inp.jump && !this.prevJump
   633→   633→    this.prevJump = inp.jump
   634→   634→    if (jumpPressed) p.buffer = BUFFER
   635→   635→    if (p.buffer > 0) p.buffer--
   636→   636→    if (p.coyote > 0) p.coyote--
   637→   637→
   638→   638→    // Горизонталь
   639→   639→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   640→   640→    if (inp.left && !inp.right) {
   641→   641→      p.vx -= acc
   642→   642→      p.facing = -1
   643→   643→    } else if (inp.right && !inp.left) {
   644→   644→      p.vx += acc
   645→   645→      p.facing = 1
   646→   646→    } else {
   647→   647→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   648→   648→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   649→   649→    }
   650→   650→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   651→   651→
   652→   652→    // Прыжок
   653→   653→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   654→   654→      p.vy = JUMP_V
   655→   655→      p.onGround = false
   656→   656→      p.coyote = 0
   657→   657→      p.buffer = 0
   658→   658→      p.squash = 1.18
   659→   659→      this.sfx.jump()
   660→   660→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   661→   661→    }
   662→   662→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   663→   663→
   664→   664→    // Гравитация
   665→   665→    p.vy = Math.min(p.vy + GRAV, 14)
   666→   666→
   667→   667→    // Движущиеся платформы (горизонтальные и вертикальные)
   668→   668→    for (const pl of this.plats) {
   669→   669→      pl.phase += pl.speed
   670→   670→      if (pl.axis === 'x') {
   671→   671→        const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   672→   672→        pl.dx = nx - pl.x
   673→   673→        pl.dy = 0
   674→   674→        pl.x = nx
   675→   675→      } else {
   676→   676→        const ny = pl.by + Math.sin(pl.phase) * pl.amp
   677→   677→        pl.dy = ny - pl.y
   678→   678→        pl.dx = 0
   679→   679→        pl.y = ny
   680→   680→      }
   681→   681→    }
   682→   682→
   683→   683→    // Движение и коллизии
   684→   684→    this.moveX(p, p.vx)
   685→   685→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   686→   686→
   687→   687→    const wasOnGround = p.onGround
   688→   688→    const prevFeet = p.y + p.h
   689→   689→    this.moveY(p, p.vy, prevFeet)
   690→   690→
   691→   691→    if (p.onGround) {
   692→   692→      if (!wasOnGround) {
   693→   693→        p.squash = 0.78
   694→   694→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   695→   695→        this.sfx.land()
   696→   696→      }
   697→   697→      p.coyote = COYOTE
   698→   698→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   699→   699→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   700→   700→      if (underL && underR) {
   701→   701→        p.safeX = p.x
   702→   702→        p.safeY = p.y
   703→   703→      }
   704→   704→    }
   705→   705→    p.squash += (1 - p.squash) * 0.18
   706→   706→    p.anim += Math.abs(p.vx) * 0.09
   707→   707→    if (p.invuln > 0) p.invuln--
   708→   708→    if (p.blink > 0) p.blink--
   709→   709→    else if (Math.random() < 0.012) p.blink = 8
   710→   710→
   711→   711→    // Провал за нижнюю границу мира (люки)
   712→   712→    if (p.y > this.worldH + 60) {
   713→   713→      this.damage(DMG_FALL, 0)
   714→   714→      p.x = p.safeX
   715→   715→      p.y = p.safeY
   716→   716→      p.vx = 0
   717→   717→      p.vy = 0
   718→   718→    }
   719→   719→
   720→   720→    this.updateWalkers(this.vampires, DMG_VAMPIRE, 100)
   721→   721→    this.updateWalkers(this.runners, DMG_RUNNER, SCORE_RUNNER)
   722→   722→    this.updateDrones()
   723→   723→    this.updateCrumbs()
   724→   724→    this.updateFuses()
   725→   725→    this.updatePickups()
   726→   726→    this.updateSwitches()
   727→   727→    this.updateHazards()
   728→   728→    this.updateExit()
   729→   729→    this.updateEnergy()
   730→   730→    this.updateParticles()
   731→   731→    this.updateCamera()
   732→   732→  }

   736→   736→  solidAt(px: number, py: number): boolean {
   737→   737→    const c = Math.floor(px / TILE)
   738→   738→    const r = Math.floor(py / TILE)
   739→   739→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   740→   740→    return this.grid[r * this.cols + c] === 1
   741→   741→  }
   742→   742→
   743→   743→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   744→   744→    const c0 = Math.floor(x / TILE)
   745→   745→    const c1 = Math.floor((x + w - EPS) / TILE)
   746→   746→    const r0 = Math.floor(y / TILE)
   747→   747→    const r1 = Math.floor((y + h - EPS) / TILE)
   748→   748→    for (let r = r0; r <= r1; r++) {
   749→   749→      for (let c = c0; c <= c1; c++) {
   750→   750→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   751→   751→        if (this.grid[r * this.cols + c] === 1) return true
   752→   752→      }
   753→   753→    }
   754→   754→    return false
   755→   755→  }
   756→   756→
   757→   757→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   758→   758→    e.x += dx
   759→   759→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   760→   760→    if (dx > 0) {
   761→   761→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   762→   762→    } else if (dx < 0) {
   763→   763→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   764→   764→    }
   765→   765→    e.vx = 0
   766→   766→    return true
   767→   767→  }
   768→   768→
   769→   769→  private moveY(
   770→   770→    e: Box & { vy: number; onGround: boolean },
   771→   771→    dy: number,
   772→   772→    prevFeet: number,
   773→   773→  ): void {
   774→   774→    e.y += dy
   775→   775→    if (dy >= 0) {
   776→   776→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   777→   777→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   778→   778→        e.vy = 0
   779→   779→        e.onGround = true
   780→   780→      }
   781→   781→      for (const pl of this.plats) {
   782→   782→        const feet = e.y + e.h
   783→   783→        if (
   784→   784→          e.vy >= 0 &&
   785→   785→          feet >= pl.y &&
   786→   786→          prevFeet <= pl.y + 6 &&
   787→   787→          e.x + e.w > pl.x &&
   788→   788→          e.x < pl.x + pl.w
   789→   789→        ) {
   790→   790→          e.y = pl.y - e.h
   791→   791→          e.vy = 0
   792→   792→          e.onGround = true
   793→   793→          e.x += pl.dx
   794→   794→        }
   795→   795→      }
   796→   796→      // Осыпающиеся плиты: та же «односторонняя» коллизия сверху
   797→   797→      for (const c of this.crumbs) {
   798→   798→        if (c.state === 2) continue
   799→   799→        const feet = e.y + e.h
   800→   800→        if (
   801→   801→          feet >= c.y &&
   802→   802→          prevFeet <= c.y + 6 &&
   803→   803→          e.x + e.w > c.x + 2 &&
   804→   804→          e.x < c.x + c.w - 2
   805→   805→        ) {
   806→   806→          e.y = c.y - e.h
   807→   807→          e.vy = 0
   808→   808→          e.onGround = true
   809→   809→        }
   810→   810→      }
   811→   811→    } else {
   812→   812→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   813→   813→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   814→   814→        e.vy = 0
   815→   815→      }
   816→   816→    }
   817→   817→  }

   821→   821→  /** Наземные ходячие враги (вампиры и гонцы). */
   822→   822→  private updateWalkers(list: Vampire[], dmg: number, stompScore: number) {
   823→   823→    const p = this.player
   824→   824→    for (const v of list) {
   825→   825→      if (v.dead) continue
   826→   826→      v.anim += Math.abs(v.vx) * 0.06
   827→   827→      v.vy = Math.min(v.vy + GRAV, 12)
   828→   828→      const oldVx = v.vx
   829→   829→      this.moveX(v, v.vx)
   830→   830→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   831→   831→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   832→   832→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   833→   833→
   834→   834→      const prevFeet = v.y + v.h
   835→   835→      const wasGround = v.onGround
   836→   836→      v.onGround = false
   837→   837→      this.moveY(v, v.vy, prevFeet)
   838→   838→      if (v.onGround && !wasGround) v.vy = 0
   839→   839→      // разворот у обрыва
   840→   840→      if (v.onGround) {
   841→   841→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   842→   842→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   843→   843→      }
   844→   844→
   845→   845→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   846→   846→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   847→   847→          v.dead = true
   848→   848→          p.vy = -9.5
   849→   849→          this.score += stompScore
   850→   850→          this.shake = 6
   851→   851→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   852→   852→          this.sfx.stomp()
   853→   853→        } else {
   854→   854→          this.damage(dmg, p.x < v.x + v.w / 2 ? -1 : 1)
   855→   855→        }
   856→   856→      }
   857→   857→    }
   858→   858→  }
