   522→   522→      squash: 1,

   526→   526→    this.energy = ENERGY_MAX

   540→   540→    'ArrowLeft',
   541→   541→    'ArrowRight',

   551→   551→  private onKeyDown = (e: KeyboardEvent) => {
   552→   552→    const c = this.keyId(e)
   553→   553→    if (this.GAME_KEYS.has(c)) e.preventDefault()
   554→   554→    switch (c) {
   555→   555→      case 'ArrowLeft':
   556→   556→      case 'KeyA':
   557→   557→        this.kb.left = true
   558→   558→        break
   559→   559→      case 'ArrowRight':
   560→   560→      case 'KeyD':
   561→   561→        this.kb.right = true
   562→   562→        break
   563→   563→      case 'Space':
   564→   564→      case 'ArrowUp':
   565→   565→      case 'KeyW':
   566→   566→      case 'KeyZ':
   567→   567→        this.kb.jump = true
   568→   568→        this.sfx.ensure()
   569→   569→        break
   570→   570→      case 'Escape':
   571→   571→      case 'KeyP':
   572→   572→        if (this.phase === 'playing' || this.phase === 'paused')
   573→   573→          this.togglePause()
   574→   574→        break
   575→   575→    }
   576→   576→  }

   578→   578→  private onKeyUp = (e: KeyboardEvent) => {
   579→   579→    switch (this.keyId(e)) {
   580→   580→      case 'ArrowLeft':
   581→   581→      case 'KeyA':
   582→   582→        this.kb.left = false
   583→   583→        break
   584→   584→      case 'ArrowRight':
   585→   585→      case 'KeyD':
   586→   586→        this.kb.right = false
   587→   587→        break
   588→   588→      case 'Space':
   589→   589→      case 'ArrowUp':
   590→   590→      case 'KeyW':
   591→   591→      case 'KeyZ':
   592→   592→        this.kb.jump = false
   593→   593→        break
   594→   594→    }
   595→   595→  }

   597→   597→  // Нормализация: некоторые окружения присылают события без e.code —
   598→   598→  // тогда откатываемся к e.key ('ArrowLeft', 'a', ' ', 'Enter').
   599→   599→  private keyId(e: KeyboardEvent): string {
   600→   600→    if (e.code) return e.code
   601→   601→    const k = e.key
   602→   602→    if (k === ' ') return 'Space'
   603→   603→    return k.length === 1 ? `Key${k.toUpperCase()}` : k
   604→   604→  }

   606→   606→  private onBlur = () => {
   607→   607→    this.kb.left = this.kb.right = this.kb.jump = false
   608→   608→    this.touch.left = this.touch.right = this.touch.jump = false
   609→   609→    if (this.phase === 'playing') this.setPhase('paused')
   610→   610→  }

   612→   612→  private get input() {
   613→   613→    return {
   614→   614→      left: this.kb.left || this.touch.left,
   615→   615→      right: this.kb.right || this.touch.right,
   616→   616→      jump: this.kb.jump || this.touch.jump,
   617→   617→    }
   618→   618→  }

   622→   622→  private loop = (t: number) => {
   623→   623→    if (this.destroyed) return
   624→   624→    this.raf = requestAnimationFrame(this.loop)
   625→   625→    let dt = (t - this.lastT) / 1000
   626→   626→    this.lastT = t
   627→   627→    if (dt > 0.1) dt = 0.1
   628→   628→    if (this.phase === 'playing') {
   629→   629→      this.acc += dt
   630→   630→      let steps = 0
   631→   631→      while (this.acc >= STEP && steps < 4) {
   632→   632→        this.step()
   633→   633→        this.acc -= STEP
   634→   634→        steps++
   635→   635→      }
   636→   636→      if (steps === 4) this.acc = 0
   637→   637→    }
   638→   638→    render(this.ctx, this)
   639→   639→  }

   641→   641→  // ---------- Шаг симуляции ----------

   643→   643→  private step() {
   644→   644→    const p = this.player
   645→   645→    const inp = this.input
   646→   646→    this.frame++
   647→   647→    this.levelFrames++
   648→   648→    if (this.shake > 0) this.shake *= 0.85
   649→   649→
   650→   650→    // Прыжок: буфер + койот-время
   651→   651→    const jumpPressed = inp.jump && !this.prevJump
   652→   652→    this.prevJump = inp.jump
   653→   653→    if (jumpPressed) p.buffer = BUFFER
   654→   654→    if (p.buffer > 0) p.buffer--
   655→   655→    if (p.coyote > 0) p.coyote--
   656→   656→
   657→   657→    // Горизонталь
   658→   658→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   659→   659→    if (inp.left && !inp.right) {
   660→   660→      p.vx -= acc
   661→   661→      p.facing = -1
   662→   662→    } else if (inp.right && !inp.left) {
   663→   663→      p.vx += acc
   664→   664→      p.facing = 1
   665→   665→    } else {
   666→   666→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   667→   667→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   668→   668→    }
   669→   669→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   670→   670→
   671→   671→    // Прыжок
   672→   672→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   673→   673→      p.vy = JUMP_V
   674→   674→      p.onGround = false
   675→   675→      p.coyote = 0
   676→   676→      p.buffer = 0
   677→   677→      p.squash = 1.18
   678→   678→      this.sfx.jump()
   679→   679→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   680→   680→    }
   681→   681→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   682→   682→
   683→   683→    // Гравитация
   684→   684→    p.vy = Math.min(p.vy + GRAV, 14)
   685→   685→
   686→   686→    // Движущиеся платформы (горизонтальные и вертикальные)
   687→   687→    for (const pl of this.plats) {
   688→   688→      pl.phase += pl.speed
   689→   689→      if (pl.axis === 'x') {
   690→   690→        const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   691→   691→        pl.dx = nx - pl.x
   692→   692→        pl.dy = 0
   693→   693→        pl.x = nx
   694→   694→      } else {
   695→   695→        const ny = pl.by + Math.sin(pl.phase) * pl.amp
   696→   696→        pl.dy = ny - pl.y
   697→   697→        pl.dx = 0
   698→   698→        pl.y = ny
   699→   699→      }
   700→   700→    }
   701→   701→
   702→   702→    // Движение и коллизии
   703→   703→    this.moveX(p, p.vx)
   704→   704→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   705→   705→
   706→   706→    const wasOnGround = p.onGround
   707→   707→    const prevFeet = p.y + p.h
   708→   708→    this.moveY(p, p.vy, prevFeet)
   709→   709→
   710→   710→    if (p.onGround) {
   711→   711→      if (!wasOnGround) {
   712→   712→        p.squash = 0.78
   713→   713→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   714→   714→        this.sfx.land()
   715→   715→      }
   716→   716→      p.coyote = COYOTE
   717→   717→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   718→   718→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   719→   719→      if (underL && underR) {
   720→   720→        p.safeX = p.x
   721→   721→        p.safeY = p.y
   722→   722→      }
   723→   723→    }
   724→   724→    p.squash += (1 - p.squash) * 0.18
   725→   725→    p.anim += Math.abs(p.vx) * 0.09
   726→   726→    if (p.invuln > 0) p.invuln--
   727→   727→    if (p.blink > 0) p.blink--
   728→   728→    else if (Math.random() < 0.012) p.blink = 8
   729→   729→
   730→   730→    // Провал за нижнюю границу мира (люки)
   731→   731→    if (p.y > this.worldH + 60) {
   732→   732→      this.damage(DMG_FALL, 0, 'fall')
   733→   733→      p.x = p.safeX
   734→   734→      p.y = p.safeY
   735→   735→      p.vx = 0
   736→   736→      p.vy = 0
   737→   737→    }
   738→   738→
   739→   739→    this.updateWalkers(this.vampires, DMG_VAMPIRE, 100, 'vampire')
   740→   740→    this.updateWalkers(this.runners, DMG_RUNNER, SCORE_RUNNER, 'runner')
   741→   741→    this.updateDrones()
   742→   742→    this.updatePhantoms()
   743→   743→    this.updateCrumbs()
   744→   744→    this.updateFuses()
   745→   745→    this.updatePickups()
   746→   746→    this.updateSwitches()
   747→   747→    this.updateHazards()
   748→   748→    this.updateExit()
   749→   749→    this.updateEnergy()
   750→   750→    this.updateParticles()
   751→   751→    this.updateCamera()
   752→   752→  }

   754→   754→  // ---------- Коллизии с тайлами ----------
   755→   755→
   756→   756→  solidAt(px: number, py: number): boolean {
   757→   757→    const c = Math.floor(px / TILE)
   758→   758→    const r = Math.floor(py / TILE)
   759→   759→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   760→   760→    return this.grid[r * this.cols + c] === 1
   761→   761→  }
   762→   762→
   763→   763→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   764→   764→    const c0 = Math.floor(x / TILE)
   765→   765→    const c1 = Math.floor((x + w - EPS) / TILE)
   766→   766→    const r0 = Math.floor(y / TILE)
   767→   767→    const r1 = Math.floor((y + h - EPS) / TILE)
   768→   768→    for (let r = r0; r <= r1; r++) {
   769→   769→      for (let c = c0; c <= c1; c++) {
   770→   770→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   771→   771→        if (this.grid[r * this.cols + c] === 1) return true
   772→   772→      }
   773→   773→    }
   774→   774→    return false
   775→   775→  }
   776→   776→
   777→   777→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   778→   778→    e.x += dx
   779→   779→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   780→   780→    if (dx > 0) {
   781→   781→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   782→   782→    } else if (dx < 0) {
   783→   783→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   784→   784→    }
   785→   785→    e.vx = 0
   786→   786→    return true
   787→   787→  }
   788→   788→
   789→   789→  private moveY(
   790→   790→    e: Box & { vy: number; onGround: boolean },
   791→   791→    dy: number,
   792→   792→    prevFeet: number,
   793→   793→  ): void {
   794→   794→    e.y += dy
   795→   795→    if (dy >= 0) {
   796→   796→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   797→   797→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   798→   798→        e.vy = 0
   799→   799→        e.onGround = true
   800→   800→      }
   801→   801→      for (const pl of this.plats) {
   802→   802→        const feet = e.y + e.h
   803→   803→        if (
   804→   804→          e.vy >= 0 &&
   805→   805→          feet >= pl.y &&
   806→   806→          prevFeet <= pl.y + 6 &&
   807→   807→          e.x + e.w > pl.x &&
   808→   808→          e.x < pl.x + pl.w
   809→   809→        ) {
   810→   810→          e.y = pl.y - e.h
   811→   811→          e.vy = 0
   812→   812→          e.onGround = true
   813→   813→          e.x += pl.dx
   814→   814→        }
   815→   815→      }
   816→   816→      // Осыпающиеся плиты: та же «односторонняя» коллизия сверху
   817→   817→      for (const c of this.crumbs) {
   818→   818→        if (c.state === 2) continue
   819→   819→        const feet = e.y + e.h
   820→   820→        if (
   821→   821→          feet >= c.y &&
   822→   822→          prevFeet <= c.y + 6 &&
   823→   823→          e.x + e.w > c.x + 2 &&
   824→   824→          e.x < c.x + c.w - 2
   825→   825→        ) {
   826→   826→          e.y = c.y - e.h
   827→   827→          e.vy = 0
   828→   828→          e.onGround = true
   829→   829→        }
   830→   830→      }
   831→   831→    } else {
   832→   832→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   833→   833→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   834→   834→        e.vy = 0
   835→   835→      }
   836→   836→    }
   837→   837→  }

   841→   841→  /** Наземные ходячие враги (вампиры и гонцы). */
   842→   842→  private updateWalkers(
   843→   843→    list: Vampire[],
   844→   844→    dmg: number,
   845→   845→    stompScore: number,
   846→   846→    cause: DeathCause,
   847→   847→  ) {
   848→   848→    const p = this.player
   849→   849→    for (const v of list) {
   850→   850→      if (v.dead) continue
   851→   851→      v.anim += Math.abs(v.vx) * 0.06
   852→   852→      v.vy = Math.min(v.vy + GRAV, 12)
   853→   853→      const oldVx = v.vx
   854→   854→      this.moveX(v, v.vx)
   855→   855→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   856→   856→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   857→   857→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   858→   858→
   859→   859→      const prevFeet = v.y + v.h
   860→   860→      const wasGround = v.onGround
   861→   861→      v.onGround = false
   862→   862→      this.moveY(v, v.vy, prevFeet)
   863→   863→      if (v.onGround && !wasGround) v.vy = 0
   864→   864→      // разворот у обрыва
   865→   865→      if (v.onGround) {
   866→   866→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   867→   867→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   868→   868→      }
   869→   869→
   870→   870→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   871→   871→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   872→   872→          v.dead = true
   873→   873→          p.vy = -9.5
   874→   874→          this.score += stompScore
   875→   875→          this.shake = 6
   876→   876→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   877→   877→          this.sfx.stomp()
   878→   878→        } else {
   879→   879→          this.damage(dmg, p.x < v.x + v.w / 2 ? -1 : 1, cause)
   880→   880→        }
   881→   881→      }
   882→   882→    }
   883→   883→  }

   885→   885→  /** Дроны-пылесосы: летают по диагонали, отскакивая от стен. */
   886→   886→  private updateDrones() {
   887→   887→    const p = this.player
   888→   888→    for (const d of this.drones) {
   889→   889→      if (d.dead) continue
   890→   890→      d.anim += 0.12
   891→   891→      const oldVx = d.vx
   892→   892→      this.moveX(d, d.vx)
   893→   893→      if (d.vx === 0 && oldVx !== 0) d.vx = -oldVx
   894→   894→
   895→   895→      const oldVy = d.vy
   896→   896→      this.moveY(d, d.vy, d.y + d.h)
   897→   897→      if (d.vy === 0 && oldVy !== 0) d.vy = -oldVy // пол или потолок
   898→   898→
   899→   899→      // Границы мира: не даём дрону улететь в пустоту
   900→   900→      if (d.x <= 0) {
   901→   901→        d.x = 0
   902→   902→        d.vx = Math.abs(d.vx)
   903→   903→      } else if (d.x + d.w >= this.worldW) {
   904→   904→        d.x = this.worldW - d.w
   905→   905→        d.vx = -Math.abs(d.vx)
   906→   906→      }
   907→   907→      if (d.y <= 0) {
   908→   908→        d.y = 0
   909→   909→        d.vy = Math.abs(d.vy)
   910→   910→      } else if (d.y + d.h >= this.worldH) {
   911→   911→        d.y = this.worldH - d.h
   912→   912→        d.vy = -Math.abs(d.vy)
   913→   913→      }
   914→   914→
   915→   915→      if (p.invuln <= 0 && rectsOverlap(p, d)) {
   916→   916→        if (p.vy > 2 && p.y + p.h - d.y < 20) {
   917→   917→          d.dead = true
   918→   918→          p.vy = -9.5
   919→   919→          this.score += SCORE_DRONE
   920→   920→          this.shake = 6
   921→   921→          this.spawnParticles(d.x + d.w / 2, d.y + d.h / 2, 12, 'poof')
   922→   922→          this.sfx.stomp()
   923→   923→        } else {
   924→   924→          this.damage(DMG_DRONE, p.x < d.x + d.w / 2 ? -1 : 1, 'drone')
   925→   925→        }
   926→   926→      }
   927→   927→    }
   928→   928→  }

   930→   930→  /** Осыпающиеся плиты: разрушаются под игроком и восстанавливаются. */
   931→   931→  private updateCrumbs() {
   932→   932→    const p = this.player
   933→   933→    for (const c of this.crumbs) {
   934→   934→      if (c.state === 2) {
   935→   935→        c.timer--
   936→   936→        if (c.timer <= 0) {
   937→   937→          c.state = 0
   938→   938→          c.timer = 0
   939→   939→        }
   940→   940→        continue
   941→   941→      }
   942→   942→      const standing =
   943→   943→        p.vy >= 0 &&
   944→   944→        p.x + p.w > c.x + 3 &&
   945→   945→        p.x < c.x + c.w - 3 &&
   946→   946→        Math.abs(p.y + p.h - c.y) < 4
   947→   947→      if (c.state === 0 && standing) {
   948→   948→        c.state = 1
   949→   949→        c.timer = CRUMBLE_DELAY
   950→   950→      } else if (c.state === 1) {
   951→   951→        c.timer--
   952→   952→        if (c.timer <= 0) {
   953→   953→          c.state = 2
   954→   954→          c.timer = CRUMBLE_RESPAWN
   955→   955→          this.spawnParticles(c.x + c.w / 2, c.y + 6, 11, 'dust')
   956→   956→          this.sfx.crumble()
   957→   957→        }
   958→   958→      }
   959→   959→    }
   960→   960→  }

   962→   962→  /** Предохранители: снимают замок с выхода. */
   963→   963→  private updateFuses() {
   964→   964→    const p = this.player
