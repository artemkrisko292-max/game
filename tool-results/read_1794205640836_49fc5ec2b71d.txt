   545→  private readonly GAME_KEYS = new Set([
   546→    'ArrowLeft',
   547→    'ArrowRight',
   548→    'ArrowUp',
   549→    'ArrowDown',
   550→    'Space',
   551→    'KeyA',
   552→    'KeyD',
   553→    'KeyW',
   554→    'KeyZ',

   556→
   557→  private onKeyDown = (e: KeyboardEvent) => {
   558→    const c = this.keyId(e)
   559→    if (this.GAME_KEYS.has(c)) e.preventDefault()
   560→    switch (c) {
   561→      case 'ArrowLeft':
   562→      case 'KeyA':
   563→        this.kb.left = true
   564→        break
   565→      case 'ArrowRight':
   566→      case 'KeyD':
   567→        this.kb.right = true
   568→        break
   569→      case 'Space':
   570→      case 'ArrowUp':
   571→      case 'KeyW':
   572→      case 'KeyZ':
   573→        this.kb.jump = true
   574→        this.sfx.ensure()
   575→        break
   576→      case 'Escape':
   577→      case 'KeyP':
   578→        if (this.phase === 'playing' || this.phase === 'paused')
   579→          this.togglePause()
   580→        break
   581→    }
   582→  }
   583→
   584→  private onKeyUp = (e: KeyboardEvent) => {
   585→    switch (this.keyId(e)) {
   586→      case 'ArrowLeft':
   587→      case 'KeyA':
   588→        this.kb.left = false
   589→        break
   590→      case 'ArrowRight':
   591→      case 'KeyD':
   592→        this.kb.right = false
   593→        break
   594→      case 'Space':
   595→      case 'ArrowUp':
   596→      case 'KeyW':
   597→      case 'KeyZ':
   598→        this.kb.jump = false
   599→        break
   600→    }
   601→  }
   602→
   603→  // Нормализация: некоторые окружения присылают события без e.code —
   604→  // тогда откатываемся к e.key ('ArrowLeft', 'a', ' ', 'Enter').
   605→  private keyId(e: KeyboardEvent): string {
   606→    if (e.code) return e.code
   607→    const k = e.key
   608→    if (k === ' ') return 'Space'
   609→    return k.length === 1 ? `Key${k.toUpperCase()}` : k
   610→  }
   611→
   612→  private onBlur = () => {
   613→    this.kb.left = this.kb.right = this.kb.jump = false
   614→    this.touch.left = this.touch.right = this.touch.jump = false
   615→    if (this.phase === 'playing') this.setPhase('paused')
   616→  }
   617→
   618→  private get input() {
   619→    return {
   620→      left: this.kb.left || this.touch.left,
   621→      right: this.kb.right || this.touch.right,
   622→      jump: this.kb.jump || this.touch.jump,
   623→    }
   624→  }

   626→  // ---------- Главный цикл ----------
   627→
   628→  private loop = (t: number) => {
   629→    if (this.destroyed) return
   630→    this.raf = requestAnimationFrame(this.loop)
   631→    let dt = (t - this.lastT) / 1000
   632→    this.lastT = t
   633→    if (dt > 0.1) dt = 0.1
   634→    if (this.phase === 'playing') {
   635→      this.acc += dt
   636→      let steps = 0
   637→      while (this.acc >= STEP && steps < 4) {
   638→        this.step()
   639→        this.acc -= STEP
   640→        steps++
   641→      }
   642→      if (steps === 4) this.acc = 0
   643→    } else if (this.phase === 'victory') {
   644→      // Финальный салют: сцена живёт под отчётом
   645→      this.celebrate()
   646→    }
   647→    render(this.ctx, this)
   648→  }

   650→  /** Праздничный такт: заряды салюта, полёт частиц, лёгкая дрожь камеры. */
   651→  private celebrate() {
   652→    this.frame++
   653→    if (this.shake > 0) this.shake *= 0.85
   654→    if (this.frame % 30 === 0) {
   655→      const fx =
   656→        this.camX + 70 + Math.random() * (VIEW_W - 140)
   657→      const fy =
   658→        this.camY + 50 + Math.random() * (VIEW_H * 0.5)
   659→      this.firework(fx, fy)
   660→    }
   661→    this.updateParticles()
   662→  }

   664→  /** Один залп: венок цветных искр с мягким хлопком. */
   665→  private firework(x: number, y: number) {
   666→    const palette = [
   667→      '#ffd94d',
   668→      '#ff8a3d',
   669→      '#a4f04a',
   670→      '#7ec8ff',
   671→      '#ff6b9d',
   672→      '#fff3b0',
   673→    ]
   674→    const col = palette[Math.floor(Math.random() * palette.length)]
   675→    this.shake = 1.6
   676→    this.sfx.pop()
   677→    for (let i = 0; i < 26; i++) {
   678→      const a = (i / 26) * Math.PI * 2 + Math.random() * 0.22
   679→      const sp = 2.1 + Math.random() * 2.3
   680→      this.particles.push({
   681→        x,
   682→        y,
   683→        vx: Math.cos(a) * sp,
   684→        vy: Math.sin(a) * sp,
   685→        life: 34 + Math.random() * 22,
   686→        maxLife: 56,
   687→        size: 2 + Math.random() * 2.6,
   688→        color: Math.random() < 0.75 ? col : '#fff8dc',
   689→        kind: 'star',
   690→        grav: 0.055,
   691→      })
   692→    }
   693→    if (this.particles.length > 320) this.particles.splice(0, 110)
   694→  }

   696→  // ---------- Шаг симуляции ----------

   698→  private step() {
   699→    const p = this.player
   700→    const inp = this.input
   701→    this.frame++
   702→    this.levelFrames++
   703→    if (this.shake > 0) this.shake *= 0.85
   704→
   705→    // Прыжок: буфер + койот-время
   706→    const jumpPressed = inp.jump && !this.prevJump
   707→    this.prevJump = inp.jump
   708→    if (jumpPressed) p.buffer = BUFFER
   709→    if (p.buffer > 0) p.buffer--
   710→    if (p.coyote > 0) p.coyote--
   711→
   712→    // Горизонталь
   713→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   714→    if (inp.left && !inp.right) {
   715→      p.vx -= acc
   716→      p.facing = -1
   717→    } else if (inp.right && !inp.left) {
   718→      p.vx += acc
   719→      p.facing = 1
   720→    } else {
   721→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   722→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   723→    }
   724→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   725→
   726→    // Прыжок
   727→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   728→      p.vy = JUMP_V
   729→      p.onGround = false
   730→      p.coyote = 0
   731→      p.buffer = 0
   732→      p.squash = 1.18
   733→      this.sfx.jump()
   734→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   735→    }
   736→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   737→
   738→    // Гравитация
   739→    p.vy = Math.min(p.vy + GRAV, 14)
   740→
   741→    // Движущиеся платформы (горизонтальные и вертикальные)
   742→    for (const pl of this.plats) {
   743→      pl.phase += pl.speed
   744→      if (pl.axis === 'x') {
   745→        const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   746→        pl.dx = nx - pl.x
   747→        pl.dy = 0
   748→        pl.x = nx
   749→      } else {
   750→        const ny = pl.by + Math.sin(pl.phase) * pl.amp
   751→        pl.dy = ny - pl.y
   752→        pl.dx = 0
   753→        pl.y = ny
   754→      }
   755→    }
   756→
   757→    // Движение и коллизии
   758→    this.moveX(p, p.vx)
   759→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   760→
   761→    const wasOnGround = p.onGround
   762→    const prevFeet = p.y + p.h
   763→    this.moveY(p, p.vy, prevFeet)
   764→
   765→    if (p.onGround) {
   766→      if (!wasOnGround) {
   767→        p.squash = 0.78
   768→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   769→        this.sfx.land()
   770→      }
   771→      p.coyote = COYOTE
   772→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   773→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   774→      if (underL && underR) {
   775→        p.safeX = p.x
   776→        p.safeY = p.y
   777→      }
   778→    }
   779→    p.squash += (1 - p.squash) * 0.18
   780→    p.anim += Math.abs(p.vx) * 0.09
   781→    if (p.invuln > 0) p.invuln--
   782→    if (p.blink > 0) p.blink--
   783→    else if (Math.random() < 0.012) p.blink = 8
   784→
   785→    // Провал за нижнюю границу мира (люки)
   786→    if (p.y > this.worldH + 60) {
   787→      this.damage(DMG_FALL, 0, 'fall')
   788→      p.x = p.safeX
   789→      p.y = p.safeY
   790→      p.vx = 0
   791→      p.vy = 0
   792→    }
   793→
   794→    this.updateWalkers(this.vampires, DMG_VAMPIRE, 100, 'vampire')
   795→    this.updateWalkers(this.runners, DMG_RUNNER, SCORE_RUNNER, 'runner')
   796→    this.updateDrones()
   797→    this.updatePhantoms()
   798→    this.updateCrumbs()
   799→    this.updateFuses()
   800→    this.updatePickups()
   801→    this.updateSwitches()
   802→    this.updateHazards()
   803→    this.updateExit()
   804→    this.updateEnergy()
   805→    this.updateParticles()
   806→    this.updateCamera()
   807→  }

   809→  // ---------- Коллизии с тайлами ----------

   811→  solidAt(px: number, py: number): boolean {
   812→    const c = Math.floor(px / TILE)
   813→    const r = Math.floor(py / TILE)
   814→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   815→    return this.grid[r * this.cols + c] === 1
   816→  }
   817→
   818→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   819→    const c0 = Math.floor(x / TILE)
   820→    const c1 = Math.floor((x + w - EPS) / TILE)
   821→    const r0 = Math.floor(y / TILE)
   822→    const r1 = Math.floor((y + h - EPS) / TILE)
   823→    for (let r = r0; r <= r1; r++) {
   824→      for (let c = c0; c <= c1; c++) {
   825→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   826→        if (this.grid[r * this.cols + c] === 1) return true
   827→      }
   828→    }
   829→    return false
   830→  }
   831→
   832→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   833→    e.x += dx
   834→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   835→    if (dx > 0) {
   836→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   837→    } else if (dx < 0) {
   838→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   839→    }
   840→    e.vx = 0
   841→    return true
   842→  }
   843→
   844→  private moveY(
   845→    e: Box & { vy: number; onGround: boolean },
   846→    dy: number,
   847→    prevFeet: number,
   848→  ): void {
   849→    e.y += dy
   850→    if (dy >= 0) {
   851→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   852→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   853→        e.vy = 0
   854→        e.onGround = true
   855→      }
   856→      for (const pl of this.plats) {
   857→        const feet = e.y + e.h
   858→        if (
   859→          e.vy >= 0 &&
   860→          feet >= pl.y &&
   861→          prevFeet <= pl.y + 6 &&
   862→          e.x + e.w > pl.x &&
   863→          e.x < pl.x + pl.w
   864→        ) {
   865→          e.y = pl.y - e.h
   866→          e.vy = 0
   867→          e.onGround = true
   868→          e.x += pl.dx
   869→        }
   870→      }
   871→      // Осыпающиеся плиты: та же «односторонняя» коллизия сверху
   872→      for (const c of this.crumbs) {
   873→        if (c.state === 2) continue
   874→        const feet = e.y + e.h
   875→        if (
   876→          feet >= c.y &&
   877→          prevFeet <= c.y + 6 &&
   878→          e.x + e.w > c.x + 2 &&
   879→          e.x < c.x + c.w - 2
   880→        ) {
   881→          e.y = c.y - e.h
   882→          e.vy = 0
   883→          e.onGround = true
   884→        }
   885→      }
   886→    } else {
   887→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   888→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   889→        e.vy = 0
   890→      }
   891→    }
   892→  }

   896→  /** Наземные ходячие враги (вампиры и гонцы). */
   897→  private updateWalkers(
   898→    list: Vampire[],
   899→    dmg: number,
   900→    stompScore: number,
   901→    cause: DeathCause,
   902→  ) {
   903→    const p = this.player
   904→    for (const v of list) {
   905→      if (v.dead) continue
   906→      v.anim += Math.abs(v.vx) * 0.06
   907→      v.vy = Math.min(v.vy + GRAV, 12)
   908→      const oldVx = v.vx
   909→      this.moveX(v, v.vx)
   910→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   911→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   912→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   913→
   914→      const prevFeet = v.y + v.h
   915→      const wasGround = v.onGround
   916→      v.onGround = false
   917→      this.moveY(v, v.vy, prevFeet)
   918→      if (v.onGround && !wasGround) v.vy = 0
   919→      // разворот у обрыва
   920→      if (v.onGround) {
   921→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   922→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   923→      }
   924→
   925→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   926→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   927→          v.dead = true
   928→          p.vy = -9.5
   929→          this.score += stompScore
   930→          this.shake = 6
   931→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   932→          this.sfx.stomp()
   933→        } else {
   934→          this.damage(dmg, p.x < v.x + v.w / 2 ? -1 : 1, cause)
   935→        }
   936→      }
   937→    }
   938→  }

   940→  /** Дроны-пылесосы: летают по диагонали, отскакивая от стен. */
   941→  private updateDrones() {
   942→    const p = this.player
   943→    for (const d of this.drones) {
   944→      if (d.dead) continue
   945→      d.anim += 0.12
   946→      const oldVx = d.vx
   947→      this.moveX(d, d.vx)
   948→      if (d.vx === 0 && oldVx !== 0) d.vx = -oldVx
   949→
   950→      const oldVy = d.vy
   951→      this.moveY(d, d.vy, d.y + d.h)
   952→      if (d.vy === 0 && oldVy !== 0) d.vy = -oldVy // пол или потолок
   953→
   954→      // Границы мира: не даём дрону улететь в пустоту
   955→      if (d.x <= 0) {
   956→        d.x = 0
   957→        d.vx = Math.abs(d.vx)
   958→      } else if (d.x + d.w >= this.worldW) {
   959→        d.x = this.worldW - d.w
   960→        d.vx = -Math.abs(d.vx)
   961→      }
   962→      if (d.y <= 0) {
   963→        d.y = 0
   964→        d.vy = Math.abs(d.vy)
   965→      } else if (d.y + d.h >= this.worldH) {
   966→        d.y = this.worldH - d.h
   967→        d.vy = -Math.abs(d.vy)
   968→      }
   969→
   970→      if (p.invuln <= 0 && rectsOverlap(p, d)) {
   971→        if (p.vy > 2 && p.y + p.h - d.y < 20) {
   972→          d.dead = true
   973→          p.vy = -9.5
   974→          this.score += SCORE_DRONE
   975→          this.shake = 6
   976→          this.spawnParticles(d.x + d.w / 2, d.y + d.h / 2, 12, 'poof')
   977→          this.sfx.stomp()
   978→        } else {
   979→          this.damage(DMG_DRONE, p.x < d.x + d.w / 2 ? -1 : 1, 'drone')
   980→        }
   981→      }
   982→    }
   983→  }

  1109→  private lampDrain(): number {
  1110→    let drain = 0
  1111→    const p = this.player
  1112→    const pcx = p.x + p.w / 2
  1113→    const pcy = p.y + p.h / 2
  1114→    for (const l of this.lamps) {
  1115→      if (!l.on) continue
  1116→      const dx = Math.abs(pcx - l.x)
  1117→      const dy = pcy - l.y
  1118→      if (dy > 0 && dy < LAMP_CONE_LEN) {
  1119→        const halfW = 16 + dy * 0.34
  1120→        if (dx < halfW) {
  1121→          drain += DRAIN_LAMP * (1 - (dy / LAMP_CONE_LEN) * 0.45)
  1122→        }
  1123→      }
  1124→    }
  1125→    return Math.min(drain, DRAIN_LAMP * 2)
  1126→  }

  1128→  private updateHazards() {

  1278→function circleRect(

  1280→  cy: number,
  1281→  r: number,
  1282→  rect: Box,

  1284→  const nx = Math.max(rect.x, Math.min(cx, rect.x + rect.w))
