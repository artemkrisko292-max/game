   533→  private onKeyDown = (e: KeyboardEvent) => {
   534→    const c = this.keyId(e)
   535→    if (this.GAME_KEYS.has(c)) e.preventDefault()
   536→    switch (c) {
   537→      case 'ArrowLeft':
   538→      case 'KeyA':
   539→        this.kb.left = true
   540→        break
   541→      case 'ArrowRight':
   542→      case 'KeyD':
   543→        this.kb.right = true
   544→        break
   545→      case 'Space':
   546→      case 'ArrowUp':
   547→      case 'KeyW':
   548→      case 'KeyZ':
   549→        this.kb.jump = true
   550→        this.sfx.ensure()
   551→        break
   552→      case 'Escape':
   553→      case 'KeyP':
   554→        if (this.phase === 'playing' || this.phase === 'paused')
   555→          this.togglePause()
   556→        break
   557→    }
   558→  }
   559→
   560→  private onKeyUp = (e: KeyboardEvent) => {
   561→    switch (this.keyId(e)) {
   562→      case 'ArrowLeft':
   563→      case 'KeyA':
   564→        this.kb.left = false
   565→        break
   566→      case 'ArrowRight':
   567→      case 'KeyD':
   568→        this.kb.right = false
   569→        break
   570→      case 'Space':
   571→      case 'ArrowUp':
   572→      case 'KeyW':
   573→      case 'KeyZ':
   574→        this.kb.jump = false
   575→        break
   576→    }
   577→  }
   578→
   579→  // Нормализация: некоторые окружения присылают события без e.code —
   580→  // тогда откатываемся к e.key ('ArrowLeft', 'a', ' ', 'Enter').
   581→  private keyId(e: KeyboardEvent): string {
   582→    if (e.code) return e.code
   583→    const k = e.key
   584→    if (k === ' ') return 'Space'
   585→    return k.length === 1 ? `Key${k.toUpperCase()}` : k
   586→  }
   587→
   588→  private onBlur = () => {
   589→    this.kb.left = this.kb.right = this.kb.jump = false
   590→    this.touch.left = this.touch.right = this.touch.jump = false
   591→    if (this.phase === 'playing') this.setPhase('paused')
   592→  }
   593→
   594→  private get input() {
   595→    return {
   596→      left: this.kb.left || this.touch.left,
   597→      right: this.kb.right || this.touch.right,
   598→      jump: this.kb.jump || this.touch.jump,
   599→    }
   600→  }
   601→
   602→  // ---------- Главный цикл ----------
   603→
   604→  private loop = (t: number) => {
   605→    if (this.destroyed) return
   606→    this.raf = requestAnimationFrame(this.loop)
   607→    let dt = (t - this.lastT) / 1000
   608→    this.lastT = t
   609→    if (dt > 0.1) dt = 0.1
   610→    if (this.phase === 'playing') {
   611→      this.acc += dt
   612→      let steps = 0
   613→      while (this.acc >= STEP && steps < 4) {
   614→        this.step()
   615→        this.acc -= STEP
   616→        steps++
   617→      }
   618→      if (steps === 4) this.acc = 0
   619→    }
   620→    render(this.ctx, this)
   621→  }
   622→
   623→  // ---------- Шаг симуляции ----------
   624→
   625→  private step() {
   626→    const p = this.player
   627→    const inp = this.input
   628→    this.frame++
   629→    if (this.shake > 0) this.shake *= 0.85
   630→
   631→    // Прыжок: буфер + койот-время
   632→    const jumpPressed = inp.jump && !this.prevJump
   633→    this.prevJump = inp.jump
   634→    if (jumpPressed) p.buffer = BUFFER
   635→    if (p.buffer > 0) p.buffer--
   636→    if (p.coyote > 0) p.coyote--
   637→
   638→    // Горизонталь
   639→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   640→    if (inp.left && !inp.right) {
   641→      p.vx -= acc
   642→      p.facing = -1
   643→    } else if (inp.right && !inp.left) {
   644→      p.vx += acc
   645→      p.facing = 1
   646→    } else {
   647→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   648→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   649→    }
   650→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   651→
   652→    // Прыжок
   653→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   654→      p.vy = JUMP_V
   655→      p.onGround = false
   656→      p.coyote = 0
   657→      p.buffer = 0
   658→      p.squash = 1.18
   659→      this.sfx.jump()
   660→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   661→    }
   662→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   663→
   664→    // Гравитация
   665→    p.vy = Math.min(p.vy + GRAV, 14)
   666→
   667→    // Движущиеся платформы (горизонтальные и вертикальные)
   668→    for (const pl of this.plats) {
   669→      pl.phase += pl.speed
   670→      if (pl.axis === 'x') {
   671→        const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   672→        pl.dx = nx - pl.x
   673→        pl.dy = 0
   674→        pl.x = nx
   675→      } else {
   676→        const ny = pl.by + Math.sin(pl.phase) * pl.amp
   677→        pl.dy = ny - pl.y
   678→        pl.dx = 0
   679→        pl.y = ny
   680→      }
   681→    }
   682→
   683→    // Движение и коллизии
   684→    this.moveX(p, p.vx)
   685→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   686→
   687→    const wasOnGround = p.onGround
   688→    const prevFeet = p.y + p.h
   689→    this.moveY(p, p.vy, prevFeet)
   690→
   691→    if (p.onGround) {
   692→      if (!wasOnGround) {
   693→        p.squash = 0.78
   694→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   695→        this.sfx.land()
   696→      }
   697→      p.coyote = COYOTE
   698→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   699→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   700→      if (underL && underR) {
   701→        p.safeX = p.x
   702→        p.safeY = p.y
   703→      }
   704→    }
   705→    p.squash += (1 - p.squash) * 0.18
   706→    p.anim += Math.abs(p.vx) * 0.09
   707→    if (p.invuln > 0) p.invuln--
   708→    if (p.blink > 0) p.blink--
   709→    else if (Math.random() < 0.012) p.blink = 8
   710→
   711→    // Провал за нижнюю границу мира (люки)
   712→    if (p.y > this.worldH + 60) {
   713→      this.damage(DMG_FALL, 0)
   714→      p.x = p.safeX
   715→      p.y = p.safeY
   716→      p.vx = 0
   717→      p.vy = 0
   718→    }
   719→
   720→    this.updateWalkers(this.vampires, DMG_VAMPIRE, 100)
   721→    this.updateWalkers(this.runners, DMG_RUNNER, SCORE_RUNNER)
   722→    this.updateDrones()
   723→    this.updateCrumbs()
   724→    this.updateFuses()
   725→    this.updatePickups()
   726→    this.updateSwitches()
   727→    this.updateHazards()
   728→    this.updateExit()
   729→    this.updateEnergy()
   730→    this.updateParticles()
   731→    this.updateCamera()
   732→  }

   736→  solidAt(px: number, py: number): boolean {
   737→    const c = Math.floor(px / TILE)
   738→    const r = Math.floor(py / TILE)
   739→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   740→    return this.grid[r * this.cols + c] === 1
   741→  }

   743→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   744→    const c0 = Math.floor(x / TILE)
   745→    const c1 = Math.floor((x + w - EPS) / TILE)
   746→    const r0 = Math.floor(y / TILE)
   747→    const r1 = Math.floor((y + h - EPS) / TILE)
   748→    for (let r = r0; r <= r1; r++) {
   749→      for (let c = c0; c <= c1; c++) {
   750→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   751→        if (this.grid[r * this.cols + c] === 1) return true
   752→      }
   753→    }
   754→    return false
   755→  }

   757→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   758→    e.x += dx
   759→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   760→    if (dx > 0) {
   761→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   762→    } else if (dx < 0) {
   763→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   764→    }
   765→    e.vx = 0
   766→    return true
   767→  }
   768→
   769→  private moveY(
   770→    e: Box & { vy: number; onGround: boolean },
   771→    dy: number,
   772→    prevFeet: number,
   773→  ): void {
   774→    e.y += dy
   775→    if (dy >= 0) {
   776→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   777→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   778→        e.vy = 0
   779→        e.onGround = true
   780→      }
   781→      for (const pl of this.plats) {
   782→        const feet = e.y + e.h
   783→        if (
   784→          e.vy >= 0 &&
   785→          feet >= pl.y &&
   786→          prevFeet <= pl.y + 6 &&
   787→          e.x + e.w > pl.x &&
   788→          e.x < pl.x + pl.w
   789→        ) {
   790→          e.y = pl.y - e.h
   791→          e.vy = 0
   792→          e.onGround = true
   793→          e.x += pl.dx
   794→        }
   795→      }
   796→      // Осыпающиеся плиты: та же «односторонняя» коллизия сверху
   797→      for (const c of this.crumbs) {
   798→        if (c.state === 2) continue
   799→        const feet = e.y + e.h
   800→        if (
   801→          feet >= c.y &&
   802→          prevFeet <= c.y + 6 &&
   803→          e.x + e.w > c.x + 2 &&
   804→          e.x < c.x + c.w - 2
   805→        ) {
   806→          e.y = c.y - e.h
   807→          e.vy = 0
   808→          e.onGround = true
   809→        }
   810→      }
   811→    } else {
   812→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   813→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   814→        e.vy = 0
   815→      }
   816→    }
   817→  }

   821→  /** Наземные ходячие враги (вампиры и гонцы). */
   822→  private updateWalkers(list: Vampire[], dmg: number, stompScore: number) {
   823→    const p = this.player
   824→    for (const v of list) {
   825→      if (v.dead) continue
   826→      v.anim += Math.abs(v.vx) * 0.06
   827→      v.vy = Math.min(v.vy + GRAV, 12)
   828→      const oldVx = v.vx
   829→      this.moveX(v, v.vx)
   830→      if (v.vx === 0 && oldVx !== 0) v.vx = -oldVx // упёрся в стену
   831→      if (v.x <= v.minX) v.vx = Math.abs(v.vx)
   832→      if (v.x >= v.maxX) v.vx = -Math.abs(v.vx)
   833→
   834→      const prevFeet = v.y + v.h
   835→      const wasGround = v.onGround
   836→      v.onGround = false
   837→      this.moveY(v, v.vy, prevFeet)
   838→      if (v.onGround && !wasGround) v.vy = 0
   839→      // разворот у обрыва
   840→      if (v.onGround) {
   841→        const aheadX = v.vx > 0 ? v.x + v.w + 5 : v.x - 5
   842→        if (!this.solidAt(aheadX, v.y + v.h + 8)) v.vx = -v.vx
   843→      }
   844→
   845→      if (p.invuln <= 0 && rectsOverlap(p, v)) {
   846→        if (p.vy > 2 && p.y + p.h - v.y < 22) {
   847→          v.dead = true
   848→          p.vy = -9.5
   849→          this.score += stompScore
   850→          this.shake = 6
   851→          this.spawnParticles(v.x + v.w / 2, v.y + v.h / 2, 14, 'poof')
   852→          this.sfx.stomp()
   853→        } else {
   854→          this.damage(dmg, p.x < v.x + v.w / 2 ? -1 : 1)
   855→        }
   856→      }
   857→    }
   858→  }

   861→  private updateDrones() {
   862→    const p = this.player
   863→    for (const d of this.drones) {
   864→      if (d.dead) continue
   865→      d.anim += 0.12
   866→      const oldVx = d.vx
   867→      this.moveX(d, d.vx)
   868→      if (d.vx === 0 && oldVx !== 0) d.vx = -oldVx
   869→
   870→      const oldVy = d.vy
   871→      this.moveY(d, d.vy, d.y + d.h)
   872→      if (d.vy === 0 && oldVy !== 0) d.vy = -oldVy // пол или потолок
   873→
   874→      // Границы мира: не даём дрону улететь в пустоту
   875→      if (d.x <= 0) {
   876→        d.x = 0
   877→        d.vx = Math.abs(d.vx)
   878→      } else if (d.x + d.w >= this.worldW) {
   879→        d.x = this.worldW - d.w
   880→        d.vx = -Math.abs(d.vx)
   881→      }
   882→      if (d.y <= 0) {
   883→        d.y = 0
   884→        d.vy = Math.abs(d.vy)
   885→      } else if (d.y + d.h >= this.worldH) {
   886→        d.y = this.worldH - d.h
   887→        d.vy = -Math.abs(d.vy)
   888→      }
   889→
   890→      if (p.invuln <= 0 && rectsOverlap(p, d)) {
   891→        if (p.vy > 2 && p.y + p.h - d.y < 20) {
   892→          d.dead = true
   893→          p.vy = -9.5
   894→          this.score += SCORE_DRONE
   895→          this.shake = 6
   896→          this.spawnParticles(d.x + d.w / 2, d.y + d.h / 2, 12, 'poof')
   897→          this.sfx.stomp()
   898→        } else {
   899→          this.damage(DMG_DRONE, p.x < d.x + d.w / 2 ? -1 : 1)
   900→        }
   901→      }
   902→    }
   903→  }

  1014→  private lampDrain(): number {

  1018→    const pcy = p.y + p.h / 2
  1019→    for (const l of this.lamps) {
  1020→      if (!l.on) continue

  1022→      const dy = pcy - l.y
  1023→      if (dy > 0 && dy < LAMP_CONE_LEN) {
  1024→        const halfW = 16 + dy * 0.34

  1026→          drain += DRAIN_LAMP * (1 - (dy / LAMP_CONE_LEN) * 0.45)
  1027→        }
  1028→      }
  1029→    }
  1030→    return Math.min(drain, DRAIN_LAMP * 2)

  1092→      (DRAIN_BASE * mul + this.lampDrain() * (0.8 + mul * 0.2)) * STEP

  1100→  private damage(amount: number, knockDir: number) {

  1108→    this.spawnParticles(p.x + p.w / 2, p.y + p.h / 2, 10, 'spark')

  1117→  private updateParticles() {
  1118→    for (let i = this.particles.length - 1; i >= 0; i--) {
  1119→      const q = this.particles[i]
  1120→      q.life--
  1121→      q.x += q.vx
  1122→      q.y += q.vy
  1123→      q.vy += q.grav
  1124→      q.vx *= 0.98
  1125→      if (q.life <= 0) this.particles.splice(i, 1)
  1126→    }
  1127→  }

  1129→  spawnParticles(x: number, y: number, n: number, kind: Particle['kind']) {
  1130→    const colors: Record<Particle['kind'], string[]> = {
  1131→      spark: ['#ffd94d', '#ffb63d', '#fff3b0'],
  1132→      poof: ['#c9b6d8', '#9c7fae', '#e6d9f0'],
  1133→      star: ['#ffe873', '#a4f04a', '#7ee081'],
  1134→      dust: ['#c9b08a', '#a8916e'],
  1135→    }
  1136→    for (let i = 0; i < n; i++) {
  1137→      const a = Math.random() * Math.PI * 2
  1138→      const sp = 1 + Math.random() * 2.6
  1139→      this.particles.push({
  1140→        x,
  1141→        y,
  1142→        vx: Math.cos(a) * sp,
  1143→        vy: Math.sin(a) * sp - 1.2,
  1144→        life: 22 + Math.random() * 18,
  1145→        maxLife: 40,
  1146→        size: 2 + Math.random() * 3,
  1147→        color: colors[kind][Math.floor(Math.random() * colors[kind].length)],
  1148→        kind,
  1149→        grav: kind === 'poof' ? 0 : kind === 'dust' ? 0.08 : 0.14,
  1150→      })
  1151→    }
