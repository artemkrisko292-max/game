   491→   491→  private loop = (t: number) => {
   492→   492→    if (this.destroyed) return
   493→   493→    this.raf = requestAnimationFrame(this.loop)
   494→   494→    let dt = (t - this.lastT) / 1000
   495→   495→    this.lastT = t
   496→   496→    if (dt > 0.1) dt = 0.1
   497→   497→    if (this.phase === 'playing') {
   498→   498→      this.acc += dt
   499→   499→      let steps = 0
   500→   500→      while (this.acc >= STEP && steps < 4) {
   501→   501→        this.step()
   502→   502→        this.acc -= STEP
   503→   503→        steps++
   504→   504→      }
   505→   505→      if (steps === 4) this.acc = 0
   506→   506→    }
   507→   507→    render(this.ctx, this)
   508→   508→  }

   510→   510→  // ---------- Шаг симуляции ----------
   511→   511→
   512→   512→  private step() {
   513→   513→    const p = this.player
   514→   514→    const inp = this.input
   515→   515→    this.frame++
   516→   516→    if (this.shake > 0) this.shake *= 0.85
   517→   517→
   518→   518→    // Прыжок: буфер + койот-время
   519→   519→    const jumpPressed = inp.jump && !this.prevJump
   520→   520→    this.prevJump = inp.jump
   521→   521→    if (jumpPressed) p.buffer = BUFFER
   522→   522→    if (p.buffer > 0) p.buffer--
   523→   523→    if (p.coyote > 0) p.coyote--
   524→   524→
   525→   525→    // Горизонталь
   526→   526→    const acc = p.onGround ? MOVE_ACC : AIR_ACC
   527→   527→    if (inp.left && !inp.right) {
   528→   528→      p.vx -= acc
   529→   529→      p.facing = -1
   530→   530→    } else if (inp.right && !inp.left) {
   531→   531→      p.vx += acc
   532→   532→      p.facing = 1
   533→   533→    } else {
   534→   534→      p.vx *= p.onGround ? GROUND_FRICTION : AIR_FRICTION
   535→   535→      if (Math.abs(p.vx) < 0.05) p.vx = 0
   536→   536→    }
   537→   537→    p.vx = Math.max(-MOVE_MAX, Math.min(MOVE_MAX, p.vx))
   538→   538→
   539→   539→    // Прыжок
   540→   540→    if (p.buffer > 0 && (p.onGround || p.coyote > 0)) {
   541→   541→      p.vy = JUMP_V
   542→   542→      p.onGround = false
   543→   543→      p.coyote = 0
   544→   544→      p.buffer = 0
   545→   545→      p.squash = 1.18
   546→   546→      this.sfx.jump()
   547→   547→      this.spawnParticles(p.x + p.w / 2, p.y + p.h, 4, 'dust')
   548→   548→    }
   549→   549→    if (!inp.jump && p.vy < JUMP_CUT) p.vy = JUMP_CUT
   550→   550→
   551→   551→    // Гравитация
   552→   552→    p.vy = Math.min(p.vy + GRAV, 14)
   553→   553→
   554→   554→    // Движущиеся платформы
   555→   555→    for (const pl of this.plats) {
   556→   556→      pl.phase += pl.speed
   557→   557→      const nx = pl.cx + Math.sin(pl.phase) * pl.amp
   558→   558→      pl.dx = nx - pl.x
   559→   559→      pl.x = nx
   560→   560→    }
   561→   561→
   562→   562→    // Движение и коллизии
   563→   563→    this.moveX(p, p.vx)
   564→   564→    p.x = Math.max(2, Math.min(this.worldW - p.w - 2, p.x))
   565→   565→
   566→   566→    const wasOnGround = p.onGround
   567→   567→    const prevFeet = p.y + p.h
   568→   568→    this.moveY(p, p.vy, prevFeet)
   569→   569→
   570→   570→    if (p.onGround) {
   571→   571→      if (!wasOnGround) {
   572→   572→        p.squash = 0.78
   573→   573→        this.spawnParticles(p.x + p.w / 2, p.y + p.h, 5, 'dust')
   574→   574→        this.sfx.land()
   575→   575→      }
   576→   576→      p.coyote = COYOTE
   577→   577→      const underL = this.solidAt(p.x + 4, p.y + p.h + 3)
   578→   578→      const underR = this.solidAt(p.x + p.w - 4, p.y + p.h + 3)
   579→   579→      if (underL && underR) {
   580→   580→        p.safeX = p.x
   581→   581→        p.safeY = p.y
   582→   582→      }
   583→   583→    }
   584→   584→    p.squash += (1 - p.squash) * 0.18
   585→   585→    p.anim += Math.abs(p.vx) * 0.09
   586→   586→    if (p.invuln > 0) p.invuln--
   587→   587→    if (p.blink > 0) p.blink--
   588→   588→    else if (Math.random() < 0.012) p.blink = 8
   589→   589→
   590→   590→    // Провал за нижнюю границу мира (люки)
   591→   591→    if (p.y > this.worldH + 60) {
   592→   592→      this.damage(DMG_FALL, 0)
   593→   593→      p.x = p.safeX
   594→   594→      p.y = p.safeY
   595→   595→      p.vx = 0
   596→   596→      p.vy = 0
   597→   597→    }
   598→   598→
   599→   599→    this.updateVampires()
   600→   600→    this.updatePhantoms()
   601→   601→    this.updatePickups()
   602→   602→    this.updateSwitches()
   603→   603→    this.updateHazards()
   604→   604→    this.updateExit()
   605→   605→    this.updateEnergy()
   606→   606→    this.updateParticles()
   607→   607→    this.updateCamera()
   608→   608→  }

   612→   612→  solidAt(px: number, py: number): boolean {
   613→   613→    const c = Math.floor(px / TILE)
   614→   614→    const r = Math.floor(py / TILE)
   615→   615→    if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return false
   616→   616→    return this.grid[r * this.cols + c] === 1
   617→   617→  }

   619→   619→  private rectHitsSolid(x: number, y: number, w: number, h: number): boolean {
   620→   620→    const c0 = Math.floor(x / TILE)
   621→   621→    const c1 = Math.floor((x + w - EPS) / TILE)
   622→   622→    const r0 = Math.floor(y / TILE)
   623→   623→    const r1 = Math.floor((y + h - EPS) / TILE)
   624→   624→    for (let r = r0; r <= r1; r++) {
   625→   625→      for (let c = c0; c <= c1; c++) {
   626→   626→        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) continue
   627→   627→        if (this.grid[r * this.cols + c] === 1) return true
   628→   628→      }
   629→   629→    }
   630→   630→    return false
   631→   631→  }

   633→   633→  private moveX(e: Box & { vx: number }, dx: number): boolean {
   634→   634→    e.x += dx
   635→   635→    if (!this.rectHitsSolid(e.x, e.y, e.w, e.h)) return false
   636→   636→    if (dx > 0) {
   637→   637→      e.x = Math.floor((e.x + e.w) / TILE) * TILE - e.w - EPS
   638→   638→    } else if (dx < 0) {
   639→   639→      e.x = Math.floor(e.x / TILE) * TILE + TILE + EPS
   640→   640→    }
   641→   641→    e.vx = 0
   642→   642→    return true
   643→   643→  }

   645→   645→  private moveY(
   646→   646→    e: Box & { vy: number; onGround: boolean },
   647→   647→    dy: number,
   648→   648→    prevFeet: number,
   649→   649→  ): void {
   650→   650→    e.y += dy
   651→   651→    if (dy >= 0) {
   652→   652→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   653→   653→        e.y = Math.floor((e.y + e.h) / TILE) * TILE - e.h - EPS
   654→   654→        e.vy = 0
   655→   655→        e.onGround = true
   656→   656→      }
   657→   657→      for (const pl of this.plats) {
   658→   658→        const feet = e.y + e.h
   659→   659→        if (
   660→   660→          e.vy >= 0 &&
   661→   661→          feet >= pl.y &&
   662→   662→          prevFeet <= pl.y + 6 &&
   663→   663→          e.x + e.w > pl.x &&
   664→   664→          e.x < pl.x + pl.w
   665→   665→        ) {
   666→   666→          e.y = pl.y - e.h
   667→   667→          e.vy = 0
   668→   668→          e.onGround = true
   669→   669→          e.x += pl.dx
   670→   670→        }
   671→   671→      }
   672→   672→    } else {
   673→   673→      if (this.rectHitsSolid(e.x, e.y, e.w, e.h)) {
   674→   674→        e.y = Math.floor(e.y / TILE) * TILE + TILE + EPS
   675→   675→        e.vy = 0
   676→   676→      }
   677→   677→    }
   678→   678→  }

   926→   926→function circleRect(
   927→   927→  cx: number,
   928→   928→  cy: number,
   929→   929→  r: number,
   930→   930→  rect: Box,
   931→   931→): boolean {
   932→   932→  const nx = Math.max(rect.x, Math.min(cx, rect.x + rect.w))
   933→   933→  const ny = Math.max(rect.y, Math.min(cy, rect.y + rect.h))
   934→   934→  const dx = cx - nx
   935→   935→  const dy = cy - ny
   936→   936→  return dx * dx + dy * dy <= r * r
   937→   937→}
