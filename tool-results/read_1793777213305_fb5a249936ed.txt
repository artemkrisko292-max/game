    38→interface Progress {
    39→  unlocked: number // сколько уровней открыто (1…20)
    40→  stars: number[]
    41→  best: number[]
    42→}
    43→
    44→const EMPTY_PROGRESS: Progress = {
    45→  unlocked: 1,
    46→  stars: Array(LEVELS.length).fill(0),
    47→  best: Array(LEVELS.length).fill(0),
    48→}
    49→
    50→function loadProgress(): Progress {
    51→  try {
    52→    const raw = localStorage.getItem(PROGRESS_KEY)
    53→    if (!raw) return EMPTY_PROGRESS
    54→    const p = JSON.parse(raw) as Partial<Progress>
    55→    const stars = Array(LEVELS.length).fill(0)
    56→    const best = Array(LEVELS.length).fill(0)
    57→    ;(p.stars ?? []).forEach((v, i) => (stars[i] = Number(v) || 0))
    58→    ;(p.best ?? []).forEach((v, i) => (best[i] = Number(v) || 0))
    59→    return {
    60→      unlocked: Math.min(LEVELS.length, Math.max(1, Number(p.unlocked) || 1)),
    61→      stars,
    62→      best,
    63→    }
    64→  } catch {
    65→    return EMPTY_PROGRESS
    66→  }
    67→}
    68→
    69→function starsFromResult(r: LevelResult): number {
    70→  const got = r.bulbs + r.switches + r.fuses
    71→  const tot = r.bulbsTotal + r.switchesTotal + r.fusesTotal
    72→  const pct = tot > 0 ? got / tot : 1
    73→  return pct >= 0.9 ? 3 : pct >= 0.55 ? 2 : 1
    74→}
    75→
    76→function saveLevelProgress(
    77→  levelIndex: number,
    78→  result: LevelResult | null,
    79→  summary?: RunSummary | null,
    80→) {
    81→  try {
    82→    const p = loadProgress()
    83→    const stars = result ? starsFromResult(result) : (summary?.stars ?? 1)
    84→    const score = result ? result.score : (summary?.score ?? 0)
    85→    p.stars[levelIndex] = Math.max(p.stars[levelIndex] ?? 0, stars)
    86→    p.best[levelIndex] = Math.max(p.best[levelIndex] ?? 0, score)
    87→    p.unlocked = Math.min(
    88→      LEVELS.length,
    89→      Math.max(p.unlocked, levelIndex + 2),
    90→    )
    91→    localStorage.setItem(PROGRESS_KEY, JSON.stringify(p))
    92→  } catch {
    93→    /* localStorage недоступен */
    94→  }
    95→}

   209→export default function EnergoHeroGame() {
   210→  const canvasRef = useRef<HTMLCanvasElement>(null)
   211→  const gameRef = useRef<EnergoGame | null>(null)
   212→  const musicRef = useRef<Music | null>(null)
   213→  const [phase, setPhase] = useState<Phase>('menu')
   214→  const [snap, setSnap] = useState<Snapshot | null>(null)
   215→  const [musicOn, setMusicOn] = useState(true)
   216→  const [sfxOn, setSfxOn] = useState(true)
   217→  const [isTouch, setIsTouch] = useState(false)
   218→  const [best, setBest] = useState(0)
   219→  const [progress, setProgress] = useState<Progress>(EMPTY_PROGRESS)
   220→  const [menuView, setMenuView] = useState<'main' | 'levels' | 'howto'>('main')
   221→
   222→  useEffect(() => {
   223→    const canvas = canvasRef.current
   224→    if (!canvas) return
   225→    const dpr = Math.min(2, window.devicePixelRatio || 1)
   226→    canvas.width = VIEW_W * dpr
   227→    canvas.height = VIEW_H * dpr
   228→    const ctx = canvas.getContext('2d')
   229→    if (!ctx) return
   230→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   231→
   232→    const music = new Music()
   233→    musicRef.current = music
   234→
   235→    let mOn = true
   236→    let sOn = true
   237→    try {
   238→      mOn = localStorage.getItem(PREF_MUSIC) !== 'off'
   239→      sOn = localStorage.getItem(PREF_SFX) !== 'off'
   240→    } catch {
   241→      /* ignore */
   242→    }
   243→    music.enabled = mOn
   244→
   245→    const game = new EnergoGame(canvas, {
   246→      onPhase: (s) => {
   247→        setPhase(s.phase)
   248→        setSnap(s)
   249→        if (s.phase === 'menu' || s.phase === 'intro') {
   250→          if (music.enabled) music.start()
   251→          music.setDuck(0.55)
   252→        } else if (s.phase === 'playing') {
   253→          music.setDuck(1)
   254→        } else if (s.phase === 'paused' || s.phase === 'complete') {
   255→          music.setDuck(0.3)
   256→        } else if (s.phase === 'gameover' || s.phase === 'victory') {
   257→          music.stop()
   258→        }
   259→        if (s.phase === 'complete' && s.result) {
   260→          saveLevelProgress(s.levelIndex, s.result)
   261→          setProgress(loadProgress())
   262→        }
   263→        if (s.phase === 'victory' && s.summary) {
   264→          saveLevelProgress(s.levelIndex, null, s.summary)
   265→          setProgress(loadProgress())
   266→          try {
   267→            setBest(
   268→              Number(localStorage.getItem('energohero-best') ?? '0') || 0,
   269→            )
   270→          } catch {
   271→            /* ignore */
   272→          }
   273→        }
   274→      },
   275→    })
   276→    game.sfx.muted = !sOn
   277→    gameRef.current = game
   278→    // Отладочный хук (используется автотестами: __energo.loadLevel(n))
   279→    ;(window as unknown as { __energo?: EnergoGame }).__energo = game
   280→
   281→    // Отложенная инициализация некритичного UI-состояния
   282→    const raf = requestAnimationFrame(() => {
   283→      setIsTouch(
   284→        window.matchMedia('(pointer: coarse)').matches ||
   285→          'ontouchstart' in window ||
   286→          window.innerWidth < 768,
   287→      )
   288→      setProgress(loadProgress())
   289→      setMusicOn(mOn)
   290→      setSfxOn(sOn)
   291→      try {
   292→        setBest(Number(localStorage.getItem('energohero-best') ?? '0') || 0)
   293→      } catch {
   294→        /* ignore */
   295→      }
   296→    })
   297→
   298→    // Музыка стартует после первого жеста пользователя (политика автоплея)
   299→    const kick = () => {
   300→      if (music.enabled) music.start()
   301→    }
   302→    window.addEventListener('pointerdown', kick, { once: true })
   303→    window.addEventListener('keydown', kick, { once: true })
   304→
   305→    return () => {
   306→      cancelAnimationFrame(raf)
   307→      window.removeEventListener('pointerdown', kick)
   308→      window.removeEventListener('keydown', kick)
   309→      music.stop()
   310→      game.destroy()
   311→      gameRef.current = null
   312→    }
   313→  }, [])
   314→
   315→  const act = useCallback(
   316→    (
   317→      action:
   318→        | 'start'
   319→        | 'begin'
   320→        | 'retry'
   321→        | 'next'
   322→        | 'menu'
   323→        | 'pause'
   324→        | { select: number },
   325→    ) => {
   326→      const g = gameRef.current
   327→      if (!g) return
   328→      if (typeof action === 'object') {
   329→        g.selectLevel(action.select)
   330→        return
   331→      }
   332→      switch (action) {
   333→        case 'start':
   334→          g.startRun()
   335→          break
   336→        case 'begin':
   337→          g.beginLevel()
   338→          break
   339→        case 'retry':
   340→          g.retryLevel()
   341→          break
   342→        case 'next':
   343→          g.nextLevel()
   344→          break
   345→        case 'menu':
   346→          g.toMenu()
   347→          break
   348→        case 'pause':
   349→          g.togglePause()
   350→          break
   351→      }
   352→    },
   353→    [],
   354→  )
   355→
   356→  // Enter — основное действие на оверлеях
   357→  useEffect(() => {
   358→    const onKey = (e: KeyboardEvent) => {
   359→      if (e.code !== 'Enter' && e.code !== 'NumpadEnter' && e.key !== 'Enter')
   360→        return
   361→      if (phase === 'menu') act('start')
   362→      else if (phase === 'intro') act('begin')
   363→      else if (phase === 'complete') act('next')
   364→      else if (phase === 'gameover') act('retry')
   365→      else if (phase === 'victory') act('start')
   366→      else if (phase === 'paused') act('pause')
   367→    }
   368→    window.addEventListener('keydown', onKey)
   369→    return () => window.removeEventListener('keydown', onKey)
   370→  }, [phase, act])
   371→
   372→  const hold = useCallback(
   373→    (k: 'left' | 'right' | 'jump') => ({
   374→      onPointerDown: (e: React.PointerEvent) => {
   375→        e.preventDefault()
   376→        gameRef.current?.setTouch(k, true)
   377→      },
   378→      onPointerUp: () => gameRef.current?.setTouch(k, false),
   379→      onPointerLeave: () => gameRef.current?.setTouch(k, false),
   380→      onPointerCancel: () => gameRef.current?.setTouch(k, false),
   381→    }),
   382→    [],
   383→  )
   384→
   385→  const toggleMusic = useCallback(() => {
   386→    setMusicOn((v) => {
   387→      const nv = !v
   388→      musicRef.current?.setEnabled(nv)
   389→      try {
   390→        localStorage.setItem(PREF_MUSIC, nv ? 'on' : 'off')
   391→      } catch {
   392→        /* ignore */
   393→      }
   394→      return nv
   395→    })
   396→  }, [])
   397→
   398→  const toggleSfx = useCallback(() => {
   399→    setSfxOn((v) => {
   400→      const nv = !v
   401→      if (gameRef.current) gameRef.current.sfx.muted = !nv
   402→      try {
   403→        localStorage.setItem(PREF_SFX, nv ? 'on' : 'off')
   404→      } catch {
   405→        /* ignore */
   406→      }
   407→      return nv
   408→    })
   409→  }, [])
   410→
   411→  const playing = phase === 'playing' || phase === 'paused'
   412→
   413→  return (
   414→    <div className="ah-grid-bg flex min-h-screen w-full flex-col items-center justify-center gap-3 p-2 text-[#efe7d2] sm:p-4">
   415→      <main className="w-full max-w-[1080px]">
   416→        <div className="relative aspect-[16/9] max-md:aspect-[4/5] w-full overflow-hidden rounded-md border-2 border-[#2a2f28] bg-[#0b0d0c] shadow-[0_0_0_6px_rgba(255,255,255,0.03),0_24px_70px_rgba(0,0,0,0.6)]">
   417→          <canvas
   418→            ref={canvasRef}
   419→            className="absolute inset-0 h-full w-full object-contain"
   420→            aria-label="Игровое поле ЭнергоГероя"
   421→          />
   422→
   423→          {/* Кнопки музыки, звука и паузы */}
   424→          <div className="absolute right-2 top-2 z-20 flex gap-1.5">
   425→            <CanvasIconBtn
   426→              label={musicOn ? 'Выключить музыку' : 'Включить музыку'}
   427→              on={musicOn}
   428→              onClick={toggleMusic}
   429→            >
   430→              <Music2 className="h-4 w-4" />
   431→            </CanvasIconBtn>
   432→            <CanvasIconBtn
   433→              label={sfxOn ? 'Выключить звук' : 'Включить звук'}

   459→                  best={best}
   460→                  unlocked={progress.unlocked}
   461→                  onContinue={() => act({ select: progress.unlocked - 1 })}

   463→                  onLevels={() => setMenuView('levels')}
   464→                  onHowto={() => setMenuView('howto')}
   465→                />

   467→              {menuView === 'levels' && (
   468→                <MenuLevels
   469→                  progress={progress}
   470→                  onPick={(i) => {
   471→                    setMenuView('main')
   472→                    act({ select: i })

   474→                  onClose={() => setMenuView('main')}

   494→                onToggleMusic={toggleMusic}

  1212→function ScoreCount({ value }: { value: number }) {
  1213→  const [v, setV] = useState(0)
  1214→  useEffect(() => {
  1215→    let raf = 0
  1216→    const t0 = performance.now()
  1217→    const dur = 700
  1218→    const tick = (t: number) => {
  1219→      const k = Math.min(1, (t - t0) / dur)
  1220→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
  1221→      if (k < 1) raf = requestAnimationFrame(tick)
  1222→    }
  1223→    raf = requestAnimationFrame(tick)
  1224→    return () => cancelAnimationFrame(raf)
  1225→  }, [value])
  1226→  return <>{v}</>
  1227→}
