    38→    38→interface Progress {
    39→    39→  unlocked: number // сколько уровней открыто (1…20)
    40→    40→  stars: number[]
    41→    41→  best: number[]
    42→    42→}

    44→    44→const EMPTY_PROGRESS: Progress = {
    45→    45→  unlocked: 1,

    47→    47→  best: Array(LEVELS.length).fill(0),

    49→    49→
    50→    50→function loadProgress(): Progress {
    51→    51→  try {
    52→    52→    const raw = localStorage.getItem(PROGRESS_KEY)
    53→    53→    if (!raw) return EMPTY_PROGRESS
    54→    54→    const p = JSON.parse(raw) as Partial<Progress>
    55→    55→    const stars = Array(LEVELS.length).fill(0)
    56→    56→    const best = Array(LEVELS.length).fill(0)
    57→    57→    ;(p.stars ?? []).forEach((v, i) => (stars[i] = Number(v) || 0))
    58→    58→    ;(p.best ?? []).forEach((v, i) => (best[i] = Number(v) || 0))
    59→    59→    return {
    60→    60→      unlocked: Math.min(LEVELS.length, Math.max(1, Number(p.unlocked) || 1)),
    61→    61→      stars,
    62→    62→      best,
    63→    63→    }
    64→    64→  } catch {
    65→    65→    return EMPTY_PROGRESS
    66→    66→  }
    67→    67→}

    69→    69→function starsFromResult(r: LevelResult): number {
    70→    70→  const got = r.bulbs + r.switches + r.fuses
    71→    71→  const tot = r.bulbsTotal + r.switchesTotal + r.fusesTotal
    72→    72→  const pct = tot > 0 ? got / tot : 1
    73→    73→  return pct >= 0.9 ? 3 : pct >= 0.55 ? 2 : 1
    74→    74→}

    76→    76→function saveLevelProgress(
    77→    77→  levelIndex: number,
    78→    78→  result: LevelResult | null,
    79→    79→  summary?: RunSummary | null,
    80→    80→) {
    81→    81→  try {
    82→    82→    const p = loadProgress()
    83→    83→    const stars = result ? starsFromResult(result) : (summary?.stars ?? 1)
    84→    84→    const score = result ? result.score : (summary?.score ?? 0)
    85→    85→    p.stars[levelIndex] = Math.max(p.stars[levelIndex] ?? 0, stars)
    86→    86→    p.best[levelIndex] = Math.max(p.best[levelIndex] ?? 0, score)
    87→    87→    p.unlocked = Math.min(
    88→    88→      LEVELS.length,
    89→    89→      Math.max(p.unlocked, levelIndex + 2),
    90→    90→    )
    91→    91→    localStorage.setItem(PROGRESS_KEY, JSON.stringify(p))
    92→    92→  } catch {
    93→    93→    /* localStorage недоступен */
    94→    94→  }
    95→    95→}

   840→   840→  return (

   850→   850→        {rank}

   852→   852→
   853→   853→      <h2 className="text-2xl font-black tracking-wide text-[#221e17] sm:text-3xl">

  1212→  1212→function ScoreCount({ value }: { value: number }) {
  1213→  1213→  const [v, setV] = useState(0)
  1214→  1214→  useEffect(() => {
  1215→  1215→    let raf = 0
  1216→  1216→    const t0 = performance.now()
  1217→  1217→    const dur = 700
  1218→  1218→    const tick = (t: number) => {
  1219→  1219→      const k = Math.min(1, (t - t0) / dur)
  1220→  1220→      setV(Math.round(value * (1 - Math.pow(1 - k, 3))))
  1221→  1221→      if (k < 1) raf = requestAnimationFrame(tick)
  1222→  1222→    }
  1223→  1223→    raf = requestAnimationFrame(tick)
  1224→  1224→    return () => cancelAnimationFrame(raf)
  1225→  1225→  }, [value])
  1226→  1226→  return <>{v}</>
  1227→  1227→}
